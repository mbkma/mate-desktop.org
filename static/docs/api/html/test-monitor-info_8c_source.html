27static const char *
28owner_name (MateBackgroundOwner owner)
29{
30 switch (owner) {

32 return "none";
33 case MATE_BACKGROUND_OWNER_CLIENT:
34 return "client";

42static void
43print_all (MateMonitorInfo *info)
44{
45 GdkDisplay *display = gdk_display_get_default ();
46 int n = gdk_display_get_n_monitors (display);
47
48 g_print ("protocol: %s\ncapabilities:%s%s%s\n",
49 mate_monitor_info_has_protocol (info) ? "yes" : "no",
50 mate_monitor_info_has_capability (info, MATE_MONITOR_CAPABILITY_WALLPAPER) ? " wallpaper" : "",
51 mate_monitor_info_has_capability (info, MATE_MONITOR_CAPABILITY_PER_OUTPUT_WALLPAPER) ? " per-output" : "",
52 mate_monitor_info_has_capability (info, MATE_MONITOR_CAPABILITY_SPANNED_WALLPAPER) ? " spanned" : "");
53 for (int i = 0; i < n; i++) {
54 GdkMonitor *monitor = gdk_display_get_monitor (display, i);
55 gchar *id = mate_monitor_info_get_id (info, monitor);
56 gchar *name = mate_monitor_info_get_display_name (info, monitor);
57 gchar *mirror = mate_monitor_info_get_mirror_of (info, monitor);
58 GdkRectangle geometry, workarea;
59
60 gdk_monitor_get_geometry (monitor, &geometry);
61 mate_monitor_info_get_workarea (info, monitor, &workarea);
62 g_print ("monitor %d\n"
63 " id: %s\n name: %s\n internal: %s\n primary: %s\n"
64 " mirror of: %s\n"
65 " geometry: %d,%d %dx%d\n workarea: %d,%d %dx%d\n"
66 " background: %s\n",
67 i, id, name,
68 mate_monitor_info_is_internal (info, monitor) ? "yes" : "no",
69 mate_monitor_info_is_primary (info, monitor) ? "yes" : "no",
70 mirror ? mirror : "-",
71 geometry.x, geometry.y, geometry.width, geometry.height,
72 workarea.x, workarea.y, workarea.width, workarea.height,
73 owner_name (mate_monitor_info_get_background_owner (info, monitor)));
74 g_free (id);
75 g_free (name);
76 g_free (mirror);
77 }
78}

80static void
81on_changed (MateMonitorInfo *info, GdkMonitor *monitor, gpointer data)
82{
83 g_print ("--- changed\n");
84 print_all (info);

87int
88main (int argc, char **argv)
89{
90 gboolean watch = argc > 1 && g_strcmp0 (argv[1], "--watch") == 0;
91 MateMonitorInfo *info;
92
93 gtk_init (&argc, &argv);
94 info = mate_monitor_info_get_default ();
95 print_all (info);
96 if (!watch)
97 return 0;
98 g_signal_connect (info, "changed", G_CALLBACK (on_changed), NULL);
99 gtk_main ();
100 return 0;
