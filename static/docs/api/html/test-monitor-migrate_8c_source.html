59 gtk_init (&argc, &argv);

62 id = mate_monitor_info_get_id (info, monitor);
63 old_id = mate_monitor_info_get_fallback_id (info, monitor);
64 if (g_strcmp0 (id, old_id) == 0) {
65 g_print ("skip: id %s is the fallback id\n", id);
66 return 77;

68 bg_id = mate_bg_get_monitor_id (monitor);
69 ok &= check (g_strcmp0 (bg_id, id) == 0, "mate_bg_get_monitor_id () is the protocol's id");
70
71 old_settings = at (old_id);
72 new_settings = at (id);
73 g_settings_set_string (old_settings, "picture-filename", "/old.png");
74 g_settings_set_boolean (old_settings, "use-shared", FALSE);
75 g_object_unref (mate_bg_get_monitor_settings (monitor));
76 value = g_settings_get_string (new_settings, "picture-filename");
77 ok &= check (g_strcmp0 (value, "/old.png") == 0 &&
78 !g_settings_get_boolean (new_settings, "use-shared"),
79 "the fallback id's settings were copied");
80 g_free (value);
81
82 g_settings_set_string (new_settings, "picture-filename", "/new.png");
83 g_settings_set_string (old_settings, "picture-filename", "/older.png");
84 mate_monitor_info_migrate_settings (info, monitor, MATE_BG_MONITOR_SCHEMA,
85 MATE_BG_MONITOR_PATH_PREFIX);
86 value = g_settings_get_string (new_settings, "picture-filename");
87 ok &= check (g_strcmp0 (value, "/new.png") == 0, "settings already at the id are not overwritten");
88 g_free (value);
89 value = g_settings_get_string (old_settings, "picture-filename");
90 ok &= check (g_strcmp0 (value, "/older.png") == 0, "the fallback id's settings stay");
91 g_free (value);
92
93 g_object_unref (old_settings);
94 g_object_unref (new_settings);
95 g_free (id);
96 g_free (old_id);
97 g_free (bg_id);
