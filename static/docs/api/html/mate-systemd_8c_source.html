46static void
47on_start_transient_unit_cb (GObject *source,
48 GAsyncResult *res,
49 gpointer user_data)
50{
51 g_autoptr (GTask) task = G_TASK (user_data);
52 g_autoptr (GVariant) reply = NULL;
53 GError *error = NULL;
54 StartSystemdScopeData *task_data = g_task_get_task_data (task);
55
56 reply = g_dbus_connection_call_finish (G_DBUS_CONNECTION (source),
57 res, &error);
58 if (error && !g_error_matches (error, G_IO_ERROR, G_IO_ERROR_CANCELLED))
59 {
60 g_warning ("Could not create transient scope for PID %d: %s",
61 task_data->pid, error->message);
62 g_task_return_error (task, error);
63 return;
64 }
65
66 g_debug ("Created transient scope for PID %d", task_data->pid);
67
68 g_task_return_boolean (task, TRUE);
69}

71static void
72start_systemd_scope (GDBusConnection *connection, GTask *task)
73{
74 GVariantBuilder builder;
75 g_autofree char *unit_name = NULL;
76 StartSystemdScopeData *task_data = g_task_get_task_data (task);
77
78 g_assert (task_data != NULL);
79
80 /* This needs to be unique, hopefully the pid will be enough. */
81 unit_name = g_strdup_printf ("app-mate-%s-%d.scope", task_data->name, task_data->pid);
82
83 g_variant_builder_init (&builder, G_VARIANT_TYPE ("(ssa(sv)a(sa(sv)))"));
84 g_variant_builder_add (&builder, "s", unit_name);
85 g_variant_builder_add (&builder, "s", "fail");
86
87 /* Note that mate-session ships a drop-in to control further defaults. */
88 g_variant_builder_open (&builder, G_VARIANT_TYPE ("a(sv)"));
89 if (task_data->description)
90 g_variant_builder_add (&builder,
91 "(sv)",
92 "Description",
93 g_variant_new_string (task_data->description));
94 g_variant_builder_add (&builder,
95 "(sv)",
96 "PIDs",
97 g_variant_new_fixed_array (G_VARIANT_TYPE_UINT32, &task_data->pid, 1, 4));
98
99 /* Default to let systemd garbage collect failed applications we launched. */
100 g_variant_builder_add (&builder,
101 "(sv)",
102 "CollectMode",
103 g_variant_new_string ("inactive-or-failed"));
104 g_variant_builder_close (&builder);
105
106 g_variant_builder_open (&builder, G_VARIANT_TYPE ("a(sa(sv))"));
107 g_variant_builder_close (&builder);
108
109 g_dbus_connection_call (connection,
110 "org.freedesktop.systemd1",
111 "/org/freedesktop/systemd1",
112 "org.freedesktop.systemd1.Manager",
113 "StartTransientUnit",
114 g_variant_builder_end (&builder),
115 G_VARIANT_TYPE ("(o)"),
116 G_DBUS_CALL_FLAGS_NO_AUTO_START,
117 1000,
118 g_task_get_cancellable (task),
119 on_start_transient_unit_cb,
120 task);
121}

123static void
124on_bus_gotten_cb (GObject *source,
125 GAsyncResult *res,
126 gpointer user_data)
127{
128 g_autoptr(GTask) task = G_TASK (user_data);
129 g_autoptr(GDBusConnection) connection = NULL;
130 GError *error = NULL;
131
132 connection = g_bus_get_finish (res, &error);
133 if (error)
134 {
135 if (!g_error_matches (error, G_IO_ERROR, G_IO_ERROR_CANCELLED))
136 g_warning ("Could not get session bus: %s", error->message);
137
138 g_task_return_error (task, error);
139 return;
140 }
141
142 start_systemd_scope (connection, g_steal_pointer (&task));
143}

145/* Is there a systemd user instance we can ask to create the scope? Being in a
146 * logind session is not enough on its own, the user instance can be masked. */
147static gboolean
148user_instance_available (void)
149{
150 g_autofree char *private_socket = NULL;
151 uid_t uid;
152
153 if (sd_pid_get_owner_uid (getpid (), &uid) < 0)
154 return FALSE;
155
156 private_socket = g_build_filename (g_get_user_runtime_dir (), "systemd", "private", NULL);
157
158 return g_file_test (private_socket, G_FILE_TEST_EXISTS);
159}

163static gchar *
164systemd_unit_name_escape (const gchar *in)
165{
166 /* Adapted from systemd source */
167 GString * const str = g_string_sized_new (strlen (in));
168
169 for (; *in; in++)
170 {
171 if (g_ascii_isalnum (*in) || *in == ':' || *in == '_' || *in == '.')
172 g_string_append_c (str, *in);
173 else
174 g_string_append_printf (str, "\\x%02x", *in);
175 }
176 return g_string_free (str, FALSE);
177}

212void
213mate_start_systemd_scope (const char *name,
214 gint32 pid,
215 const char *description,
216 GDBusConnection *connection,
217 GCancellable *cancellable,
218 GAsyncReadyCallback callback,
219 gpointer user_data)
220{
221 g_autoptr(GTask) task = NULL;
222
223#ifdef HAVE_SYSTEMD
224 g_autofree char *own_unit = NULL;
225 g_autofree char *name_tmp = NULL;
226 StartSystemdScopeData *task_data;
227 gint res;
228
229 task = g_task_new (NULL, cancellable, callback, user_data);
230 task_data = g_new0 (StartSystemdScopeData, 1);
231
232 task_data->pid = pid;
233
234 /* Fall back to a "well-known" name, that is unlikely to be used. */
235 if (name == NULL)
236 name = "org.freedesktop.Unknown";
237
238 /* Consider the following cases:
239 * 1. If we get a path, use the binary name only
240 * 2. If the ID ends in .desktop, remove .desktop to get an application ID
241 * 3. Use the name as-is
242 */
243 if (name[0] == '/')
244 name_tmp = g_path_get_basename (name);
245 else if (g_str_has_suffix (name, ".desktop"))
246 name_tmp = g_strndup (name, strlen (name) - 8);
247 else
248 name_tmp = g_strdup (name);
249
250 /* Escape the above, so that it can be extracted from the unit name. */
251 task_data->name = systemd_unit_name_escape (name_tmp);
252
253 task_data->description = g_strdup (description);
254 if (task_data->description == NULL)
255 {
256 const char *app_name = g_get_application_name();
257
258 if (app_name)
259 task_data->description = g_strdup_printf ("Application launched by %s",
260 app_name);
261 }
262
263 g_task_set_task_data (task, task_data, (GDestroyNotify) start_systemd_scope_data_free);
264
265 /* MATE is not started as a systemd user unit, it runs in the logind
266 * session scope, so sd_pid_get_user_unit() returns -ENODATA even where the
267 * user instance is perfectly usable. Only give up if there is none. */
268 res = sd_pid_get_user_unit (getpid (), &own_unit);
269 if (res == -ENODATA && user_instance_available ())
270 res = 0;
271
272 if (res == -ENODATA)
273 {
274 g_debug ("Not systemd managed, will not move PID %d into transient scope\n", pid);
275 g_task_return_boolean (task, TRUE);
276
277 return;
278 }
279 if (res < 0)
280 {
281 g_warning ("Error fetching user unit for own pid: %d\n", -res);
282 g_task_return_new_error (task,
283 G_IO_ERROR,
284 g_io_error_from_errno (-res),
285 "Error fetching user unit for own pid: %d", -res);
286
287 return;
288 }
289
290 if (connection == NULL)
291 g_bus_get (G_BUS_TYPE_SESSION, cancellable, on_bus_gotten_cb, g_steal_pointer (&task));
292 else
293 start_systemd_scope (connection, g_steal_pointer (&task));
294#else
295 g_debug ("Not creating transient scope for PID %d. Systemd support not compiled in.", pid);
296
297 task = g_task_new (NULL, cancellable, callback, user_data);
298 g_task_return_boolean (task, TRUE);
299#endif
300}
