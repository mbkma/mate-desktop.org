92static gboolean
93emit_changed (gpointer data)
94{
95 ProtocolBackend *self = data;
96 GArray *heads = g_array_new (FALSE, FALSE, sizeof (OutputHead));
97 GList *l;
98
99 self->emit_id = 0;
100
101 for (l = self->heads; l != NULL; l = l->next)
102 {
103 ProtocolHead *ph = l->data;
104 OutputHead head;
105 guint i;
106
107 memset (&head, 0, sizeof (OutputHead));
108 head.name = g_strdup (ph->name);
109 head.description = g_strdup (ph->description ? ph->description : ph->name);
110 head.enabled = ph->enabled;
111 head.x = ph->x;
112 head.y = ph->y;
113 head.transform = ph->transform & 0x3;
114 head.scale = ph->scale > 0.0 ? ph->scale : 1.0;
115 head.adaptive_sync_supported = ph->adaptive_sync ? 1 : -1;
116 head.adaptive_sync = ph->adaptive_sync;
117 head.modes = g_array_new (FALSE, FALSE, sizeof (Mode));
118 head.id = g_strdup (ph->id);
119 head.display_name = g_strdup (ph->display_name);
120 head.internal = ph->internal;
121 head.primary = ph->primary;
122 head.mirror_of = ph->mirror_of && *ph->mirror_of ? g_strdup (ph->mirror_of) : NULL;
123
124 if (ph->enabled && ph->current)
125 {
126 head.width = ph->current->mode.width;
127 head.height = ph->current->mode.height;
128 head.rate = ph->current->mode.rate;
129 head.preferred = ph->current->mode.preferred;
130 g_array_append_val (head.modes, ph->current->mode);
131 }
132
133 for (i = 0; i < ph->modes->len; i++)
134 {
135 ProtocolMode *pm = g_ptr_array_index (ph->modes, i);
136
137 if (pm->mode.width <= 0 || pm->mode.height <= 0)
138 continue;
139 if (display_head_has_mode (&head, pm->mode.width, pm->mode.height, pm->mode.rate))
140 continue;
141 g_array_append_val (head.modes, pm->mode);
142 }
143
144 g_array_append_val (heads, head);
145 }
146
147 self->base.changed (heads, NULL, self->base.changed_data);
148 return G_SOURCE_REMOVE;
149}

529static ProtocolMode *
530find_mode (ProtocolHead *ph, int width, int height, int rate)
531{
532 ProtocolMode *best = NULL;
533 guint i;
534
535 for (i = 0; i < ph->modes->len; i++)
536 {
537 ProtocolMode *pm = g_ptr_array_index (ph->modes, i);
538
539 if (pm->mode.width != width || pm->mode.height != height)
540 continue;
541 if (pm->mode.rate == rate)
542 return pm;
543 if (rate <= 0 && (best == NULL || pm->mode.rate > best->mode.rate))
544 best = pm;
545 }
546
547 return best;
548}

589static void
590protocol_apply (DisplayBackend *backend,
591 GArray *heads,
592 gboolean temporary,
593 DisplayBackendAppliedFunc done,
594 gpointer data)
595{
596 ProtocolBackend *self = (ProtocolBackend *) backend;
597 ProtocolApply *call = g_new0 (ProtocolApply, 1);
598 gboolean primary_set = FALSE;
599 guint i;
600
601 call->done = done;
602 call->data = data;
603 call->configuration = zwlr_output_manager_v1_create_configuration (self->manager, self->serial);
604 call->mate = mate_output_manager_v1_get_configuration (self->mate, call->configuration);
605 zwlr_output_configuration_v1_add_listener (call->configuration, &configuration_listener, call);
606
607 for (i = 0; i < heads->len; i++)
608 {
609 OutputHead *head = &g_array_index (heads, OutputHead, i);
610 ProtocolHead *ph = find_head (self, head->name);
611 struct zwlr_output_configuration_head_v1 *ch;
612 ProtocolMode *pm;
613 int width = head->width, height = head->height;
614
615 if (ph == NULL)
616 continue;
617
618 if (!head->enabled)
619 {
620 zwlr_output_configuration_v1_disable_head (call->configuration, ph->proxy);
621 continue;
622 }
623
624 ch = zwlr_output_configuration_v1_enable_head (call->configuration, ph->proxy);
625
626 if (width <= 0 || height <= 0)
627 display_head_get_preferred_size (head, &width, &height);
628 pm = find_mode (ph, width, height, head->rate);
629 if (pm)
630 zwlr_output_configuration_head_v1_set_mode (ch, pm->proxy);
631 else if (width > 0 && height > 0)
632 zwlr_output_configuration_head_v1_set_custom_mode (ch, width, height, head->rate);
633
634 zwlr_output_configuration_head_v1_set_position (ch, head->x, head->y);
635 zwlr_output_configuration_head_v1_set_transform (ch, (head->transform & 0x3) | (ph->transform & 0x4));
636 zwlr_output_configuration_head_v1_set_scale (ch, wl_fixed_from_double (head->scale > 0.0 ? head->scale : 1.0));
637 if (head->adaptive_sync_supported == 1 &&
638 zwlr_output_configuration_head_v1_get_version (ch) >=
639 ZWLR_OUTPUT_CONFIGURATION_HEAD_V1_SET_ADAPTIVE_SYNC_SINCE_VERSION)
640 zwlr_output_configuration_head_v1_set_adaptive_sync (ch, head->adaptive_sync
641 ? ZWLR_OUTPUT_HEAD_V1_ADAPTIVE_SYNC_STATE_ENABLED
642 : ZWLR_OUTPUT_HEAD_V1_ADAPTIVE_SYNC_STATE_DISABLED);
643
644 if (head->primary && !primary_set)
645 {
646 mate_output_configuration_v1_set_primary (call->mate, ch);
647 primary_set = TRUE;
648 }
649 }
650
651 if (temporary)
652 mate_output_configuration_v1_set_temporary (call->mate);
653
654 zwlr_output_configuration_v1_apply (call->configuration);
655 wl_display_flush (self->display);
656}

658static void

660{

662

679DisplayBackend *
680display_backend_protocol_new (DisplayBackendChangedFunc changed, gpointer data)
681{
682 GdkDisplay *gdk_display = gdk_display_get_default ();
683 ProtocolBackend *self;
684
685 if (gdk_display == NULL || !GDK_IS_WAYLAND_DISPLAY (gdk_display))
686 return NULL;
687
688 self = g_new0 (ProtocolBackend, 1);
689 self->display = gdk_wayland_display_get_wl_display (gdk_display);
690 self->registry = wl_display_get_registry (self->display);
691 wl_registry_add_listener (self->registry, &registry_listener, self);
692 wl_display_roundtrip (self->display);
693
694 if (self->wlr_name == 0 || self->mate_name == 0)
695 {
696 wl_registry_destroy (self->registry);
697 g_free (self);
698 return NULL;
699 }
700
701 self->base.name = "wlr-output-management";
702 self->base.features = DISPLAY_BACKEND_PRIMARY | DISPLAY_BACKEND_IDENTITY |
703 DISPLAY_BACKEND_MIRROR | DISPLAY_BACKEND_TEMPORARY;
704 self->base.changed = changed;
705 self->base.changed_data = data;
706 self->base.refresh = protocol_refresh;
707 self->base.apply = protocol_apply;
708 self->base.free = protocol_free;
709
710 self->mate = wl_registry_bind (self->registry, self->mate_name,
711 &mate_output_manager_v1_interface, MATE_OUTPUT_MANAGER_VERSION);
712 mate_output_manager_v1_add_listener (self->mate, &mate_manager_listener, self);
713 self->manager = wl_registry_bind (self->registry, self->wlr_name,
714 &zwlr_output_manager_v1_interface, self->wlr_version);
715 zwlr_output_manager_v1_add_listener (self->manager, &manager_listener, self);
716 wl_display_roundtrip (self->display);
717 wl_display_roundtrip (self->display);
718
719 return &self->base;
720}
