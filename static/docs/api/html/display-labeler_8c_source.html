48void
49display_labeler_get_color (guint index, guint count, GdkRGBA *color)
50{
51 double r, g, b;
52
53 if (count == 0)
54 count = 1;
55
56 gtk_hsv_to_rgb (2.0 / 3 / count * (index % count), 1.0 / 3, 1.0, &r, &g, &b);
57 color->red = r;
58 color->green = g;
59 color->blue = b;
60 color->alpha = 1.0;
61}

75static gboolean
76on_label_draw (GtkWidget *widget, cairo_t *cr, gpointer data)
77{
78 GdkRGBA *color = g_object_get_data (G_OBJECT (widget), "color");
79 int width = gtk_widget_get_allocated_width (widget);
80 int height = gtk_widget_get_allocated_height (widget);
81
82 cairo_set_source_rgb (cr, 0, 0, 0);
83 cairo_rectangle (cr, EDGE_THICKNESS / 2.0, EDGE_THICKNESS / 2.0,
84 width - EDGE_THICKNESS, height - EDGE_THICKNESS);
85 cairo_set_line_width (cr, EDGE_THICKNESS);
86 cairo_stroke (cr);
87
88 gdk_cairo_set_source_rgba (cr, color);
89 cairo_rectangle (cr, EDGE_THICKNESS, EDGE_THICKNESS,
90 width - EDGE_THICKNESS * 2, height - EDGE_THICKNESS * 2);
91 cairo_fill (cr);
92
93 return FALSE;
94}

96static GdkMonitor *
97find_monitor (OutputHead *head)
98{
99 GdkDisplay *display = gdk_display_get_default ();
100 int i, n = gdk_display_get_n_monitors (display);
101
102 for (i = 0; i < n; i++)
103 {
104 GdkMonitor *monitor = gdk_display_get_monitor (display, i);
105 GdkRectangle geometry;
106
107 gdk_monitor_get_geometry (monitor, &geometry);
108 if (geometry.x == head->x && geometry.y == head->y && geometry.width > 0)
109 return monitor;
110 }
111
112 return NULL;
113}

115static GtkWidget *
116make_label (OutputHead *head, GdkMonitor *monitor, gboolean mirror, guint index, guint count)
117{
118 GtkWidget *window;
119 GtkWidget *label;
120 GtkCssProvider *provider;
121 GdkRGBA *color = g_new (GdkRGBA, 1);
122 char *markup;
123
124 window = gtk_window_new (GTK_WINDOW_TOPLEVEL);
125 gtk_window_set_decorated (GTK_WINDOW (window), FALSE);
126 gtk_window_set_accept_focus (GTK_WINDOW (window), FALSE);
127 gtk_window_set_skip_taskbar_hint (GTK_WINDOW (window), TRUE);
128 gtk_window_set_title (GTK_WINDOW (window), display_head_get_label (head));
129 gtk_widget_set_app_paintable (window, TRUE);
130 gtk_container_set_border_width (GTK_CONTAINER (window), PADDING + EDGE_THICKNESS);
131
132 display_labeler_get_color (index, count, color);
133 g_object_set_data_full (G_OBJECT (window), "color", color, g_free);
134 g_signal_connect (window, "draw", G_CALLBACK (on_label_draw), NULL);
135
136#ifdef HAVE_GTK_LAYER_SHELL
137 if (gtk_layer_is_supported ())
138 {
139 gtk_layer_init_for_window (GTK_WINDOW (window));
140 gtk_layer_set_namespace (GTK_WINDOW (window), "mate-display-label");
141 gtk_layer_set_layer (GTK_WINDOW (window), GTK_LAYER_SHELL_LAYER_OVERLAY);
142 gtk_layer_set_keyboard_mode (GTK_WINDOW (window), GTK_LAYER_SHELL_KEYBOARD_MODE_NONE);
143 gtk_layer_set_anchor (GTK_WINDOW (window), GTK_LAYER_SHELL_EDGE_TOP, TRUE);
144 gtk_layer_set_anchor (GTK_WINDOW (window), GTK_LAYER_SHELL_EDGE_LEFT, TRUE);
145 gtk_layer_set_monitor (GTK_WINDOW (window), monitor);
146 }
147#endif
148
149 if (mirror)
150 markup = g_markup_printf_escaped ("<b>%s</b>", _("Mirror Screens"));
151 else
152 markup = g_markup_printf_escaped ("<b>%s</b>\n<small>%s</small>",
153 display_head_get_label (head), head->name);
154
155 label = gtk_label_new (NULL);
156 gtk_label_set_markup (GTK_LABEL (label), markup);
157 g_free (markup);
158
159 provider = gtk_css_provider_new ();
160 gtk_css_provider_load_from_data (provider, "label { color: black; }", -1, NULL);
161 gtk_style_context_add_provider (gtk_widget_get_style_context (label),
162 GTK_STYLE_PROVIDER (provider),
163 GTK_STYLE_PROVIDER_PRIORITY_APPLICATION);
164 g_object_unref (provider);
165
166 gtk_container_add (GTK_CONTAINER (window), label);
167
168 gtk_widget_show_all (window);
169
170 return window;
171}

190static void
191place_labels (DisplayLabeler *labeler)
192{
193 guint i, count;
194 gboolean missing = FALSE;
195
196 destroy_windows (labeler);
197
198 if (labeler->heads == NULL)
199 return;
200
201 count = labeler->heads->len;
202
203 for (i = 0; i < labeler->heads->len; i++)
204 {
205 OutputHead *head = &g_array_index (labeler->heads, OutputHead, i);
206 GdkMonitor *monitor;
207
208 if (!head->enabled)
209 continue;
210
211 monitor = find_monitor (head);
212 if (monitor == NULL)
213 missing = TRUE;
214 else
215 g_ptr_array_add (labeler->windows, make_label (head, monitor, labeler->mirror, i, count));
216
217 if (labeler->mirror)
218 break;
219 }
220
221 if (missing && labeler->retries_left > 0 && labeler->retry == 0)
222 {
223 labeler->retries_left--;
224 labeler->retry = g_timeout_add (RETRY_MS, retry_place, labeler);
225 }
226}

228static void
229on_monitors_changed (GdkDisplay *display, GdkMonitor *monitor, gpointer data)
230{
231 DisplayLabeler *labeler = data;
232
233 if (labeler->heads == NULL)
234 return;
235
236 labeler->retries_left = RETRIES;

240DisplayLabeler *
241display_labeler_new (void)
242{
243 DisplayLabeler *labeler = g_new0 (DisplayLabeler, 1);
244 GdkDisplay *display = gdk_display_get_default ();
245
246 labeler->windows = g_ptr_array_new_with_free_func ((GDestroyNotify) gtk_widget_destroy);
247 labeler->added_id = g_signal_connect (display, "monitor-added",
248 G_CALLBACK (on_monitors_changed), labeler);
249 labeler->removed_id = g_signal_connect (display, "monitor-removed",
250 G_CALLBACK (on_monitors_changed), labeler);
251
252 return labeler;
253}

255void
256display_labeler_show (DisplayLabeler *labeler, GArray *heads, gboolean mirror)
257{
258 display_heads_free (labeler->heads);
259 labeler->heads = display_heads_copy (heads);
260 labeler->mirror = mirror;
261 labeler->retries_left = RETRIES;
262
263 if (labeler->retry)
264 {
265 g_source_remove (labeler->retry);
266 labeler->retry = 0;
267 }
268
269 place_labels (labeler);
270}

272void
273display_labeler_hide (DisplayLabeler *labeler)
274{
275 if (labeler->retry)
276 {
277 g_source_remove (labeler->retry);
278 labeler->retry = 0;
279 }
280
281 display_heads_free (labeler->heads);
282 labeler->heads = NULL;
283 destroy_windows (labeler);
284}
