59static gchar *
60variant_dup_string (GVariant *variant)
61{
62 if (variant == NULL)
63 return g_strdup ("");
64
65 if (g_variant_is_of_type (variant, G_VARIANT_TYPE_STRING) ||
66 g_variant_is_of_type (variant, G_VARIANT_TYPE_OBJECT_PATH))
67 return g_variant_dup_string (variant, NULL);
68
69 if (g_variant_is_of_type (variant, G_VARIANT_TYPE_STRING_ARRAY)) {
70 const gchar **strv;
71 gchar *joined;
72
73 strv = g_variant_get_strv (variant, NULL);
74 joined = g_strjoinv (", ", (gchar **) strv);
75 g_free (strv);
76 return joined;
77 }
78
79 return g_strdup ("");
80}

114static void
115gvc_mpris_player_update_metadata (GvcMprisPlayer *player,
116 GVariant *metadata)
117{
118 GVariantIter iter;
119 const gchar *key;
120 GVariant *value;
121
122 if (metadata == NULL || !g_variant_is_of_type (metadata, G_VARIANT_TYPE_VARDICT))
123 return;
124
125 g_clear_pointer (&player->priv->title, g_free);
126 g_clear_pointer (&player->priv->artist, g_free);
127 g_clear_pointer (&player->priv->album, g_free);
128 g_clear_pointer (&player->priv->art_url, g_free);
129 g_clear_pointer (&player->priv->track_id, g_free);
130 player->priv->length = 0;
131
132 player->priv->title = g_strdup ("");
133 player->priv->artist = g_strdup ("");
134 player->priv->album = g_strdup ("");
135 player->priv->art_url = g_strdup ("");
136 player->priv->track_id = g_strdup ("");
137
138 g_variant_iter_init (&iter, metadata);
139 while (g_variant_iter_next (&iter, "{&sv}", &key, &value)) {
140 if (g_str_equal (key, "xesam:title")) {
141 g_free (player->priv->title);
142 player->priv->title = variant_dup_string (value);
143 } else if (g_str_equal (key, "xesam:artist")) {
144 g_free (player->priv->artist);
145 player->priv->artist = variant_dup_string (value);
146 } else if (g_str_equal (key, "xesam:album")) {
147 g_free (player->priv->album);
148 player->priv->album = variant_dup_string (value);
149 } else if (g_str_equal (key, "mpris:artUrl")) {
150 g_free (player->priv->art_url);
151 player->priv->art_url = variant_dup_string (value);
152 } else if (g_str_equal (key, "mpris:trackid") &&
153 g_variant_is_of_type (value, G_VARIANT_TYPE_OBJECT_PATH)) {
154 g_free (player->priv->track_id);
155 player->priv->track_id = variant_dup_string (value);
156 } else if (g_str_equal (key, "mpris:length") &&
157 g_variant_is_of_type (value, G_VARIANT_TYPE_INT64)) {
158 player->priv->length = g_variant_get_int64 (value);
159 }
160
161 g_variant_unref (value);
162 }

183static void
184gvc_mpris_player_read_initial_state (GvcMprisPlayer *player)
185{
186 GVariant *metadata;
187 gchar *value;
188
189 value = get_cached_string (player->priv->server_proxy, "Identity");
190 if (value != NULL && value[0] != '\0') {
191 g_free (player->priv->identity);
192 player->priv->identity = value;
193 } else {
194 gchar *name = g_strdup (player->priv->bus_name + strlen ("org.mpris.MediaPlayer2."));
195 g_free (value);
196 if (name[0] != '\0')
197 name[0] = g_ascii_toupper (name[0]);
198 g_free (player->priv->identity);
199 player->priv->identity = name;
200 }
201
202 g_free (player->priv->desktop_entry);
203 player->priv->desktop_entry = get_cached_string (player->priv->server_proxy, "DesktopEntry");
204
205 g_free (player->priv->playback_status);
206 player->priv->playback_status = get_cached_string (player->priv->player_proxy, "PlaybackStatus");
207 if (player->priv->playback_status == NULL || player->priv->playback_status[0] == '\0') {
208 g_free (player->priv->playback_status);
209 player->priv->playback_status = g_strdup ("Unknown");
210 }
211
212 g_free (player->priv->loop_status);
213 player->priv->loop_status = get_cached_string (player->priv->player_proxy, "LoopStatus");
214 if (player->priv->loop_status == NULL)
215 player->priv->loop_status = g_strdup ("None");
216
217 player->priv->shuffle = get_cached_boolean (player->priv->player_proxy, "Shuffle");
218
219 metadata = g_dbus_proxy_get_cached_property (player->priv->player_proxy, "Metadata");
220 gvc_mpris_player_update_metadata (player, metadata);
221 if (metadata != NULL)
222 g_variant_unref (metadata);
223
224 gvc_mpris_player_update_capabilities (player);
225}

227static void
228on_proxy_properties_changed (GDBusProxy *proxy,
229 GVariant *changed_properties,
230 const gchar **invalidated_properties,
231 GvcMprisPlayer *player)
232{
233 gboolean metadata_changed = FALSE;
234 gboolean status_changed = FALSE;
235 gboolean capabilities_changed = FALSE;
236 gboolean player_iface;
237 GVariantIter iter;
238 const gchar *key;
239 GVariant *value;
240
241 player_iface = proxy == player->priv->player_proxy;
242
243 g_variant_iter_init (&iter, changed_properties);
244 while (g_variant_iter_next (&iter, "{&sv}", &key, &value)) {
245 if (player_iface && g_str_equal (key, "Metadata")) {
246 gvc_mpris_player_update_metadata (player, value);
247 metadata_changed = TRUE;
248 } else if (player_iface && g_str_equal (key, "PlaybackStatus")) {
249 g_free (player->priv->playback_status);
250 player->priv->playback_status = variant_dup_string (value);
251 status_changed = TRUE;
252 } else if (player_iface && g_str_equal (key, "Shuffle")) {
253 player->priv->shuffle = g_variant_get_boolean (value);
254 capabilities_changed = TRUE;
255 } else if (player_iface && g_str_equal (key, "LoopStatus")) {
256 g_free (player->priv->loop_status);
257 player->priv->loop_status = variant_dup_string (value);
258 capabilities_changed = TRUE;
259 } else if (!player_iface && g_str_equal (key, "Identity")) {
260 g_free (player->priv->identity);
261 player->priv->identity = variant_dup_string (value);
262 metadata_changed = TRUE;
263 } else if (!player_iface && g_str_equal (key, "DesktopEntry")) {
264 g_free (player->priv->desktop_entry);
265 player->priv->desktop_entry = variant_dup_string (value);
266 } else if (g_str_has_prefix (key, "Can")) {
267 capabilities_changed = TRUE;
268 }
269
270 g_variant_unref (value);
271 }
272
273 if (capabilities_changed) {
274 gvc_mpris_player_update_capabilities (player);
275 g_signal_emit (player, signals[CAPABILITIES_CHANGED], 0);
276 }
277 if (metadata_changed)
278 g_signal_emit (player, signals[METADATA_CHANGED], 0);
279 if (status_changed)
280 g_signal_emit (player, signals[STATUS_CHANGED], 0);
281}

283static void
284on_proxy_ready (GObject *source_object,
285 GAsyncResult *res,
286 gpointer user_data)
287{
288 GvcMprisPlayer *player = GVC_MPRIS_PLAYER (user_data);
289 GDBusProxy *proxy;
290 GError *error = NULL;
291
292 proxy = g_dbus_proxy_new_for_bus_finish (res, &error);
293 if (error != NULL) {
294 g_debug ("Unable to create MPRIS proxy for %s: %s",
295 player->priv->bus_name,
296 error->message);
297 g_error_free (error);
298 } else if (g_str_equal (g_dbus_proxy_get_interface_name (proxy), MPRIS_SERVER_IFACE)) {
299 player->priv->server_proxy = proxy;
300 g_signal_connect (proxy,
301 "g-properties-changed",
302 G_CALLBACK (on_proxy_properties_changed),
303 player);
304 } else {
305 player->priv->player_proxy = proxy;
306 g_signal_connect (proxy,
307 "g-properties-changed",
308 G_CALLBACK (on_proxy_properties_changed),
309 player);
310 }
311
312 player->priv->pending_proxies--;
313
314 if (player->priv->pending_proxies == 0 &&
315 player->priv->server_proxy != NULL &&
316 player->priv->player_proxy != NULL) {
317 gvc_mpris_player_read_initial_state (player);
318 player->priv->ready = TRUE;
319 g_signal_emit (player, signals[READY], 0);
320 }
321
322 g_object_unref (player);
323}
