93static void
94epub_document_make_tree_entry (linknode *list_data, LinksCBStruct *user_data)
95{
96 GtkTreeIter tree_iter;
97 EvLink *link = NULL;
98 EvLinkDest *ev_dest = NULL;
99 EvLinkAction *ev_action;
100 char *title_markup;
101
102 if (g_strrstr (list_data->pagelink, "#") == NULL) {
103 ev_dest = ev_link_dest_new_page (list_data->page);
104 } else {
105 gchar *hlink_uri = g_strdup_printf ("epub:///%s", list_data->pagelink);
106 ev_dest = ev_link_dest_new_hlink (hlink_uri, list_data->page);
107 g_free (hlink_uri);
108 }
109
110 ev_action = ev_link_action_new_dest (ev_dest);
111 link = ev_link_new (list_data->linktext, ev_action);
112
113 gtk_tree_store_append (GTK_TREE_STORE (user_data->model), &tree_iter, user_data->parent);
114 title_markup = g_markup_escape_text (list_data->linktext, -1);
115
116 gtk_tree_store_set (GTK_TREE_STORE (user_data->model), &tree_iter,
117 EV_DOCUMENT_LINKS_COLUMN_MARKUP, title_markup,
118 EV_DOCUMENT_LINKS_COLUMN_LINK, link,
119 EV_DOCUMENT_LINKS_COLUMN_EXPAND, (list_data->children != NULL),
120 -1);
121
122 if (list_data->children) {
123 LinksCBStruct child_struct;
124 child_struct.parent = &tree_iter;
125 child_struct.model = user_data->model;
126 g_list_foreach (list_data->children, (GFunc) epub_document_make_tree_entry, &child_struct);
127 }
128
129 g_free (title_markup);
130 g_object_unref (link);
131}

133static GtkTreeModel *
134epub_document_links_get_links_model (EvDocumentLinks *document_links)
135{
136 GtkTreeModel *model = NULL;
137 EpubDocument *epub_document = EPUB_DOCUMENT (document_links);
138
139 g_return_val_if_fail (EPUB_IS_DOCUMENT (document_links), NULL);
140
141 model = (GtkTreeModel *) gtk_tree_store_new (EV_DOCUMENT_LINKS_COLUMN_NUM_COLUMNS,
142 G_TYPE_STRING,
143 G_TYPE_OBJECT,
144 G_TYPE_BOOLEAN,
145 G_TYPE_STRING);
146
147 LinksCBStruct link_struct;
148 link_struct.model = model;
149
150 EvLink *link = ev_link_new (epub_document->docTitle,
151 ev_link_action_new_dest (ev_link_dest_new_page (0)));
152 GtkTreeIter parent;
153 link_struct.parent = &parent;
154
155 gtk_tree_store_append (GTK_TREE_STORE (model), &parent, NULL);
156 gtk_tree_store_set (GTK_TREE_STORE (model), &parent,
157 EV_DOCUMENT_LINKS_COLUMN_MARKUP, epub_document->docTitle,
158 EV_DOCUMENT_LINKS_COLUMN_LINK, link,
159 EV_DOCUMENT_LINKS_COLUMN_EXPAND, TRUE,
160 -1);
161 g_object_unref (link);
162
163 if (epub_document->index)
164 g_list_foreach (epub_document->index, (GFunc) epub_document_make_tree_entry, &link_struct);
165
166 return model;
167}

220static EvDocumentInfo *
221epub_document_get_info (EvDocument *document)
222{
223 EpubDocument *epub_document = EPUB_DOCUMENT (document);
224 EvDocumentInfo *info = g_new0 (EvDocumentInfo, 1);
225
226 info->fields_mask = EV_DOCUMENT_INFO_TITLE |
227 EV_DOCUMENT_INFO_FORMAT |
228 EV_DOCUMENT_INFO_AUTHOR |
229 EV_DOCUMENT_INFO_SUBJECT |
230 EV_DOCUMENT_INFO_LAYOUT |
231 EV_DOCUMENT_INFO_PERMISSIONS |
232 EV_DOCUMENT_INFO_N_PAGES;
233
234 if (epub_document->gepub_doc) {
235 info->title = gepub_doc_get_metadata (epub_document->gepub_doc, GEPUB_META_TITLE);
236 info->author = gepub_doc_get_metadata (epub_document->gepub_doc, GEPUB_META_AUTHOR);
237 info->subject = gepub_doc_get_metadata (epub_document->gepub_doc, GEPUB_META_DESC);
238 }
239
240 if (!info->title)
241 info->title = g_strdup (_("Unknown"));
242 if (!info->author)
243 info->author = g_strdup (_("Unknown"));
244 if (!info->subject)
245 info->subject = g_strdup (_("Unknown"));
246
247 info->format = g_strdup ("epub");
248 info->layout = EV_DOCUMENT_LAYOUT_SINGLE_PAGE;
249 info->permissions = EV_DOCUMENT_PERMISSIONS_OK_TO_COPY;
250 info->n_pages = epub_document_get_n_pages (document);
251
252 return info;
253}

286static GList *
287build_index_from_gepub_toc (EpubDocument *epub_document)
288{
289 GList *toc = gepub_doc_get_toc (epub_document->gepub_doc);
290 GList *index = NULL;
291 guint first = 0, last = 0;
292
293 for (GList *l = toc; l; l = l->next) {
294 GepubNavPoint *np = l->data;
295 linknode *node = g_new0 (linknode, 1);
296 node->linktext = g_strdup (np->label);
297 node->pagelink = g_strdup (np->content);
298 node->children = NULL;
299
300 gint chapter = gepub_doc_resource_uri_to_chapter (
301 epub_document->gepub_doc, np->content);
302 node->page = (chapter >= 0) ? chapter : 0;
303
304 if (index)
305 last = node->page;
306 else
307 first = node->page;
308
309 index = g_list_prepend (index, node);
310 }
311
312 if (first < last)
313 index = g_list_reverse (index);
314
315 return index;
316}

318static gboolean
319epub_document_load (EvDocument *document,
320 const char *uri,
321 GError **error)
322{
323 EpubDocument *epub_document = EPUB_DOCUMENT (document);
324 GError *err = NULL;
325
326 gchar *filepath = g_filename_from_uri (uri, NULL, &err);
327 if (!filepath) {
328 g_propagate_error (error, err);
329 return FALSE;
330 }
331
332 epub_document->archivename = filepath;
333
334 epub_document->gepub_doc = gepub_doc_new (filepath, &err);
335 if (!epub_document->gepub_doc) {
336 g_propagate_error (error, err);
337 return FALSE;
338 }
339
340 epub_document->docTitle = gepub_doc_get_metadata (
341 epub_document->gepub_doc, GEPUB_META_TITLE);
342 if (!epub_document->docTitle)
343 epub_document->docTitle = g_strdup (_("Unknown"));
344
345 epub_document->index = build_index_from_gepub_toc (epub_document);
346
347 epub_document->n_pages = gepub_doc_get_n_chapters (epub_document->gepub_doc);
348 if (epub_document->n_pages == 0) {
349 g_set_error_literal (error,
350 EV_DOCUMENT_ERROR,
351 EV_DOCUMENT_ERROR_INVALID,
352 _("epub document has no pages"));
353 return FALSE;
354 }
355
356 /* Pre-compute page URIs so get_page is thread-safe */
357 epub_document->page_paths = g_new0 (gchar *, epub_document->n_pages);
358 for (gint i = 0; i < epub_document->n_pages; i++) {
359 gepub_doc_set_chapter (epub_document->gepub_doc, i);
360 gchar *path = gepub_doc_get_current_path (epub_document->gepub_doc);
361 if (path) {
362 epub_document->page_paths[i] = g_strdup_printf ("epub:///%s", path);
363 g_free (path);
364 }
365 }
366
367 return TRUE;
368}

383{
