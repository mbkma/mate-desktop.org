84static void
85configure_volume_box_layout (GvcStreamAppletIcon *icon,
86 GtkOrientation orientation)
87{
88 if (icon->priv->volume_box == NULL ||
89 icon->priv->volume_image == NULL)
90 return;
91
92 gtk_orientable_set_orientation (GTK_ORIENTABLE (icon->priv->volume_box),
93 orientation);
94
95 if (orientation == GTK_ORIENTATION_HORIZONTAL) {
96 gtk_box_reorder_child (GTK_BOX (icon->priv->volume_box),
97 GTK_WIDGET (icon->priv->volume_image),
98 0);
99 gtk_box_reorder_child (GTK_BOX (icon->priv->volume_box),
100 icon->priv->bar,
101 1);
102 } else {
103 gtk_box_reorder_child (GTK_BOX (icon->priv->volume_box),
104 icon->priv->bar,
105 0);
106 gtk_box_reorder_child (GTK_BOX (icon->priv->volume_box),
107 GTK_WIDGET (icon->priv->volume_image),
108 1);
109 }

112static void
113configure_dock_layout (GvcStreamAppletIcon *icon,
114 gboolean volume_first)
115{
116 gboolean vertical_panel;
117
118 vertical_panel = icon->priv->orient == MATE_PANEL_APPLET_ORIENT_LEFT ||
119 icon->priv->orient == MATE_PANEL_APPLET_ORIENT_RIGHT;
120
121 if (icon->priv->player_widget_visible && vertical_panel) {
122 gtk_orientable_set_orientation (GTK_ORIENTABLE (icon->priv->dock_box),
123 GTK_ORIENTATION_HORIZONTAL);
124 gvc_channel_bar_set_orientation (GVC_CHANNEL_BAR (icon->priv->bar),
125 GTK_ORIENTATION_VERTICAL);
126 configure_volume_box_layout (icon, GTK_ORIENTATION_VERTICAL);
127
128 if (icon->priv->player_separator != NULL)
129 gtk_orientable_set_orientation (GTK_ORIENTABLE (icon->priv->player_separator),
130 GTK_ORIENTATION_VERTICAL);
131
132 if (volume_first) {
133 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box), icon->priv->volume_box, 0);
134 if (icon->priv->player_separator != NULL)
135 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box),
136 icon->priv->player_separator,
137 1);
138 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box),
139 icon->priv->player_widget,
140 2);
141 } else {
142 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box),
143 icon->priv->player_widget,
144 0);
145 if (icon->priv->player_separator != NULL)
146 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box),
147 icon->priv->player_separator,
148 1);
149 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box), icon->priv->volume_box, 2);
150 }
151
152 gtk_widget_set_size_request (icon->priv->dock, -1, -1);
153 return;
154 }
155
156 gtk_orientable_set_orientation (GTK_ORIENTABLE (icon->priv->dock_box),
157 GTK_ORIENTATION_VERTICAL);
158
159 if (icon->priv->player_separator != NULL)
160 gtk_orientable_set_orientation (GTK_ORIENTABLE (icon->priv->player_separator),
161 GTK_ORIENTATION_HORIZONTAL);
162
163 if (icon->priv->player_widget_visible && volume_first) {
164 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box), icon->priv->volume_box, 0);
165 if (icon->priv->player_separator != NULL)
166 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box),
167 icon->priv->player_separator,
168 1);
169 if (icon->priv->player_widget != NULL)
170 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box),
171 icon->priv->player_widget,
172 2);
173 } else {
174 if (icon->priv->player_widget != NULL)
175 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box),
176 icon->priv->player_widget,
177 0);
178 if (icon->priv->player_separator != NULL)
179 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box),
180 icon->priv->player_separator,
181 1);
182 gtk_box_reorder_child (GTK_BOX (icon->priv->dock_box), icon->priv->volume_box, 2);
183 }
184
185 if (icon->priv->player_widget_visible) {
186 gvc_channel_bar_set_orientation (GVC_CHANNEL_BAR (icon->priv->bar),
187 GTK_ORIENTATION_HORIZONTAL);
188 configure_volume_box_layout (icon, GTK_ORIENTATION_HORIZONTAL);
189 gtk_widget_set_size_request (icon->priv->dock, 336, -1);
190 return;
191 }
192
193 gtk_widget_set_size_request (icon->priv->dock, -1, -1);
194 switch (icon->priv->orient) {
195 case MATE_PANEL_APPLET_ORIENT_LEFT:
196 case MATE_PANEL_APPLET_ORIENT_RIGHT:
197 gvc_channel_bar_set_orientation (GVC_CHANNEL_BAR (icon->priv->bar), GTK_ORIENTATION_HORIZONTAL);
198 configure_volume_box_layout (icon, GTK_ORIENTATION_HORIZONTAL);
199 break;
200 case MATE_PANEL_APPLET_ORIENT_UP:
201 case MATE_PANEL_APPLET_ORIENT_DOWN:
202 default:
203 gvc_channel_bar_set_orientation (GVC_CHANNEL_BAR (icon->priv->bar), GTK_ORIENTATION_VERTICAL);
204 configure_volume_box_layout (icon, GTK_ORIENTATION_VERTICAL);
205 }
206}

208static gboolean
209popup_dock (GvcStreamAppletIcon *icon, guint time)
210{
211 GtkAllocation allocation;
212 GtkAllocation anchor;
213 GdkDisplay *display;
214 GdkScreen *screen;
215 int x, y;
216 GdkMonitor *monitor_num;
217 GdkRectangle monitor;
218 GtkRequisition dock_req;
219
220 screen = gtk_widget_get_screen (GTK_WIDGET (icon));
221 gtk_widget_get_allocation (GTK_WIDGET (icon), &allocation);
222 gdk_window_get_origin (gtk_widget_get_window (GTK_WIDGET (icon)), &allocation.x, &allocation.y);
223 gtk_widget_set_state_flags (GTK_WIDGET (icon), GTK_STATE_FLAG_CHECKED, FALSE);
224
225 /* position roughly */
226 gtk_window_set_screen (GTK_WINDOW (icon->priv->dock), screen);
227 configure_dock_layout (icon, FALSE);
228
229 display = gdk_screen_get_display (screen);
230 monitor_num = gdk_display_get_monitor_at_window (display, gtk_widget_get_window (GTK_WIDGET (icon)));
231 gdk_monitor_get_geometry (monitor_num, &monitor);
232
233 gtk_container_foreach (GTK_CONTAINER (icon->priv->dock), (GtkCallback) gtk_widget_show_all, NULL);
234 gtk_widget_get_preferred_size (icon->priv->dock, &dock_req, NULL);
235
236#if defined(ENABLE_WAYLAND)
237 if (GDK_IS_WAYLAND_DISPLAY (display) &&
238 gtk_layer_is_layer_window (GTK_WINDOW (icon->priv->dock)))
239 {
240 gboolean top, bottom, left, right;
241 GtkWidget *toplevel;
242 toplevel = gtk_widget_get_toplevel (GTK_WIDGET (icon));
243
244 /* Layer-shell margins are relative to the selected monitor. */
245 gtk_layer_set_monitor (GTK_WINDOW (icon->priv->dock), monitor_num);
246
247 top = bottom = left = right = FALSE;
248 if (GTK_IS_WINDOW (toplevel) && gtk_layer_is_layer_window (GTK_WINDOW (toplevel))) {
249 top = gtk_layer_get_anchor (GTK_WINDOW (toplevel), GTK_LAYER_SHELL_EDGE_TOP);
250 bottom = gtk_layer_get_anchor (GTK_WINDOW (toplevel), GTK_LAYER_SHELL_EDGE_BOTTOM);
251 left = gtk_layer_get_anchor (GTK_WINDOW (toplevel), GTK_LAYER_SHELL_EDGE_LEFT);
252 right = gtk_layer_get_anchor (GTK_WINDOW (toplevel), GTK_LAYER_SHELL_EDGE_RIGHT);
253 }
254
255 if (!((top && left && right && !bottom) ||
256 (bottom && left && right && !top) ||
257 (left && bottom && top && !right) ||
258 (right && bottom && top && !left))) {
259 /* The panel may use another surface type or anchor pattern. */
260 switch (icon->priv->orient) {
261 case MATE_PANEL_APPLET_ORIENT_DOWN:
262 top = left = right = TRUE;
263 bottom = FALSE;
264 break;
265 case MATE_PANEL_APPLET_ORIENT_UP:
266 bottom = left = right = TRUE;
267 top = FALSE;
268 break;
269 case MATE_PANEL_APPLET_ORIENT_RIGHT:
270 left = bottom = top = TRUE;
271 right = FALSE;
272 break;
273 case MATE_PANEL_APPLET_ORIENT_LEFT:
274 right = bottom = top = TRUE;
275 left = FALSE;
276 break;
277 default:
278 bottom = left = right = TRUE;
279 top = FALSE;
280 break;
281 }
282 }
283
284 /*Set anchors to the edges (will hold to panel edge) and position along the panel
285 *Unset margins and anchors from any other position so as to avoid rendering issues
286 *when orientation changes as when the panel is moved
287 */
288
289 if (top && left && right && !bottom)
290 {
291 configure_dock_layout (icon, TRUE);
292 gtk_widget_get_preferred_size (icon->priv->dock, NULL, &dock_req);
293 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_TOP, TRUE);
294 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_LEFT, TRUE);
295 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_BOTTOM, FALSE);
296 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_RIGHT, FALSE);
297 gtk_layer_set_margin (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_LEFT,
298 CLAMP (allocation.x - monitor.x, 0,
299 MAX (0, monitor.width - dock_req.width)));
300 gtk_layer_set_margin (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_TOP, 0);
301 }
302 else if (bottom && left && right && !top)
303 {
304 configure_dock_layout (icon, FALSE);
305 gtk_widget_get_preferred_size (icon->priv->dock, NULL, &dock_req);
306 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_BOTTOM, TRUE);
307 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_LEFT, TRUE);
308 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_TOP, FALSE);
309 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_RIGHT, FALSE);
310 gtk_layer_set_margin (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_LEFT,
311 CLAMP (allocation.x - monitor.x, 0,
312 MAX (0, monitor.width - dock_req.width)));
313 gtk_layer_set_margin (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_TOP, 0);
314 }
315 else if (left && bottom && top && !right)
316 {
317 configure_dock_layout (icon, TRUE);
318 gtk_widget_get_preferred_size (icon->priv->dock, NULL, &dock_req);
319 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_LEFT, TRUE);
320 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_TOP, TRUE);
321 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_RIGHT, FALSE);
322 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_BOTTOM, FALSE);
323 gtk_layer_set_margin (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_TOP,
324 CLAMP (allocation.y - monitor.y, 0,
325 MAX (0, monitor.height - dock_req.height)));
326 gtk_layer_set_margin (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_LEFT, 0);
327 }
328 else if (right && bottom && top && !left)
329 {
330 configure_dock_layout (icon, FALSE);
331 gtk_widget_get_preferred_size (icon->priv->dock, NULL, &dock_req);
332 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_RIGHT, TRUE);
333 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_TOP, TRUE);
334 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_LEFT, FALSE);
335 gtk_layer_set_anchor (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_BOTTOM, FALSE);
336 gtk_layer_set_margin (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_TOP,
337 CLAMP (allocation.y - monitor.y, 0,
338 MAX (0, monitor.height - dock_req.height)));
339 gtk_layer_set_margin (GTK_WINDOW (icon->priv->dock), GTK_LAYER_SHELL_EDGE_LEFT, 0);
340 }
341 /* Discard a size retained from a previous popup layout. */
342 gtk_window_resize (GTK_WINDOW (icon->priv->dock), 1, 1);
343 gtk_widget_show_all (icon->priv->dock);
344
345 /* Grab focus */
346 gtk_grab_add (icon->priv->dock);
347 gtk_widget_grab_focus (icon->priv->dock);
348
349 return TRUE;
350 } else if (GDK_IS_WAYLAND_DISPLAY (display)) {
351 GtkWidget *toplevel = gtk_widget_get_toplevel (GTK_WIDGET (icon));
352
353 /* Ordinary Wayland popups need a parent to be positioned. */
354 if (GTK_IS_WINDOW (toplevel))
355 gtk_window_set_transient_for (GTK_WINDOW (icon->priv->dock), GTK_WINDOW (toplevel));
356 }
357#endif /* wayland support */
358
359 anchor = allocation;
360#if defined(ENABLE_WAYLAND)
361 if (!GDK_IS_WAYLAND_DISPLAY (display))
362#endif
363 {
364 GtkWidget *panel;
365 GtkAllocation panel_allocation;
366 gint panel_x;
367 gint panel_y;
368
369 // Use the panel window's outer edge for the axis perpendicular to the panel on X11. Wayland doesn't have this isssue.
370 panel = gtk_widget_get_toplevel (GTK_WIDGET (icon));
371 if (GTK_IS_WINDOW (panel) && gtk_widget_get_realized (panel)) {
372 gtk_widget_get_allocation (panel, &panel_allocation);
373 gdk_window_get_origin (gtk_widget_get_window (panel), &panel_x, &panel_y);
374
375 if (icon->priv->orient == MATE_PANEL_APPLET_ORIENT_LEFT ||
376 icon->priv->orient == MATE_PANEL_APPLET_ORIENT_RIGHT) {
377 anchor.x = panel_x;
378 anchor.width = panel_allocation.width;
379 } else {
380 anchor.y = panel_y;
381 anchor.height = panel_allocation.height;
382 }
383 }
384 }
385
386 if (icon->priv->orient == MATE_PANEL_APPLET_ORIENT_LEFT || icon->priv->orient == MATE_PANEL_APPLET_ORIENT_RIGHT) {
387 if (anchor.x + anchor.width + dock_req.width <= monitor.x + monitor.width)
388 x = anchor.x + anchor.width;
389 else
390 x = anchor.x - dock_req.width;
391
392 if (allocation.y + dock_req.height <= monitor.y + monitor.height)
393 y = allocation.y;
394 else
395 y = monitor.y + monitor.height - dock_req.height;
396 } else {
397 if (anchor.y + anchor.height + dock_req.height <= monitor.y + monitor.height)
398 y = anchor.y + anchor.height;
399 else
400 y = anchor.y - dock_req.height;
401
402 if (allocation.x + dock_req.width <= monitor.x + monitor.width)
403 x = allocation.x;
404 else
405 x = monitor.x + monitor.width - dock_req.width;
406 }
407
408#if defined(ENABLE_WAYLAND)
409 if (!GDK_IS_WAYLAND_DISPLAY (display))
410#endif
411 {
412 GdkRectangle workarea;
413
414 /* X11 workarea excludes the panel's reserved space. An experimental themed
415 * border management. Might need checks! */
416 gdk_monitor_get_workarea (monitor_num, &workarea);
417 if (icon->priv->orient == MATE_PANEL_APPLET_ORIENT_LEFT ||
418 icon->priv->orient == MATE_PANEL_APPLET_ORIENT_RIGHT) {
419 if (x > allocation.x)
420 x = MAX (x, workarea.x);
421 else
422 x = MIN (x, workarea.x + workarea.width - dock_req.width);
423 } else {
424 if (y > allocation.y)
425 y = MAX (y, workarea.y);
426 else
427 y = MIN (y, workarea.y + workarea.height - dock_req.height);
428 }
429
430 x = CLAMP (x, monitor.x, MAX (monitor.x, monitor.x + monitor.width - dock_req.width));
431 y = CLAMP (y, monitor.y, MAX (monitor.y, monitor.y + monitor.height - dock_req.height));
432 }
433
434 if (icon->priv->orient == MATE_PANEL_APPLET_ORIENT_LEFT ||
435 icon->priv->orient == MATE_PANEL_APPLET_ORIENT_RIGHT)
436 configure_dock_layout (icon, x > allocation.x);
437 else
438 configure_dock_layout (icon, y > allocation.y);
439
440 gtk_window_move (GTK_WINDOW (icon->priv->dock), x, y);
441
442 /* Without this, the popup window appears as a square after changing
443 * the orientation */
444 gtk_window_resize (GTK_WINDOW (icon->priv->dock), 1, 1);
445
446 gtk_widget_show_all (icon->priv->dock);
447
448 /* Grab focus */
449 gtk_grab_add (icon->priv->dock);
450
451 display = gtk_widget_get_display (icon->priv->dock);
452
453 do {
454 GdkSeat *seat = gdk_display_get_default_seat (display);
455 GdkWindow *window = gtk_widget_get_window (icon->priv->dock);
456
457 if (gdk_seat_grab (seat,
458 window,
459 GDK_SEAT_CAPABILITY_ALL,
460 TRUE,
461 NULL,
462 NULL,
463 NULL,
464 NULL) != GDK_GRAB_SUCCESS) {
465 gtk_grab_remove (icon->priv->dock);
466 gtk_widget_hide (icon->priv->dock);
467 break;
468 }
469 } while (0);
470
471 gtk_widget_grab_focus (icon->priv->dock);
472
473 return TRUE;
474}

476static gboolean
477on_applet_icon_button_press (GtkWidget *applet_icon,
478 GdkEventButton *event,
479 GvcStreamAppletIcon *icon)
480{
481 if (event->button == 1) {
482 popup_dock (icon, GDK_CURRENT_TIME);
483 return TRUE;
484 }
485
486 if (event->button == 8 && icon->priv->player != NULL) {
487 gvc_mpris_player_previous (icon->priv->player);
488 return TRUE;
489 }
490
491 if (event->button == 9 && icon->priv->player != NULL) {
492 gvc_mpris_player_next (icon->priv->player);
493 return TRUE;
494 }
495
496 /* Middle click acts as mute/unmute */
497 if (event->button == 2) {
498 gchar *action = NULL;
499
500 if (icon->priv->applet_settings != NULL)
501 action = g_settings_get_string (icon->priv->applet_settings, "middle-click-action");
502
503 if (g_strcmp0 (action, "player") == 0) {
504 if (icon->priv->player != NULL) {
505 gvc_mpris_player_play_pause (icon->priv->player);
506 g_free (action);
507 return TRUE;
508 }
509 } else if (g_strcmp0 (action, "mute-output") != 0 &&
510 g_strcmp0 (action, "mute-all") != 0 &&
511 action != NULL) {
512 g_free (action);
513 return FALSE;
514 }
515
516 g_free (action);
517
518 if (icon->priv->control == NULL)
519 return FALSE;
520
521 gboolean is_muted = mate_mixer_stream_control_get_mute (icon->priv->control);
522
523 mate_mixer_stream_control_set_mute (icon->priv->control, !is_muted);
524 return TRUE;
525 }
526 return FALSE;
527}

572{

864static void
865update_icon (GvcStreamAppletIcon *icon)
866{
867 guint volume = 0;
868 gdouble decibel = 0;
869 guint normal = 0;
870 gboolean muted = FALSE;
871 guint n = 0;
872 gchar *markup;
873 const gchar *description;
874 MateMixerStreamControlFlags flags;
875
876 if (icon->priv->control == NULL) {
877 /* Do not bother creating a tooltip for an unusable icon as it
878 * has no practical use */
879 gtk_widget_set_has_tooltip (GTK_WIDGET (icon), FALSE);
880 return;
881 } else
882 gtk_widget_set_has_tooltip (GTK_WIDGET (icon), TRUE);
883
884 flags = mate_mixer_stream_control_get_flags (icon->priv->control);
885
886 if (flags & MATE_MIXER_STREAM_CONTROL_MUTE_READABLE)
887 muted = mate_mixer_stream_control_get_mute (icon->priv->control);
888
889 if (flags & MATE_MIXER_STREAM_CONTROL_VOLUME_READABLE) {
890 volume = mate_mixer_stream_control_get_volume (icon->priv->control);
891 normal = mate_mixer_stream_control_get_normal_volume (icon->priv->control);
892
893 /* Select an icon, they are expected to be sorted, the lowest index being
894 * the mute icon and the rest being volume increments */
895 if (volume <= 0 || muted)
896 n = 0;
897 else
898 n = CLAMP (3 * volume / normal + 1, 1, 3);
899 }
900 if (flags & MATE_MIXER_STREAM_CONTROL_HAS_DECIBEL)
901 decibel = mate_mixer_stream_control_get_decibel (icon->priv->control);
902
903 gvc_stream_applet_icon_set_icon_from_name (icon, icon->priv->icon_names[n]);
904 icon->priv->current_icon = n;
905
906 description = mate_mixer_stream_control_get_label (icon->priv->control);
907
908 guint volume_percent = (guint) round (100.0 * volume / normal);
909 if (icon->priv->applet_settings != NULL &&
910 !g_settings_get_boolean (icon->priv->applet_settings, "tooltip-show-volume")) {
911 markup = g_strdup_printf ("<b>%s</b>", icon->priv->display_name);
912 } else if (muted) {
913 markup = g_strdup_printf ("<b>%s: %s %u%%</b>\n<small>%s</small>",
914 icon->priv->display_name,
915 _("Muted at"),
916 volume_percent,
917 description);
918 } else if (flags & MATE_MIXER_STREAM_CONTROL_VOLUME_READABLE) {
919 if (flags & MATE_MIXER_STREAM_CONTROL_HAS_DECIBEL) {
920 if (decibel > -MATE_MIXER_INFINITY) {
921 markup = g_strdup_printf ("<b>%s: %u%%</b>\n"
922 "<small>%0.2f dB\n%s</small>",
923 icon->priv->display_name,
924 volume_percent,
925 decibel,
926 description);
927 } else {
928 markup = g_strdup_printf ("<b>%s: %u%%</b>\n"
929 "<small>-&#8734; dB\n%s</small>",
930 icon->priv->display_name,
931 volume_percent,
932 description);
933 }
934 } else {
935 markup = g_strdup_printf ("<b>%s: %u%%</b>\n<small>%s</small>",
936 icon->priv->display_name,
937 volume_percent,
938 description);
939 }
940 } else {
941 markup = g_strdup_printf ("<b>%s</b>\n<small>%s</small>",
942 icon->priv->display_name,
943 description);
944 }
945
946 if (icon->priv->player != NULL &&
947 icon->priv->applet_settings != NULL &&
948 g_settings_get_boolean (icon->priv->applet_settings, "tooltip-show-player")) {
949 const gchar *title;
950 const gchar *artist;
951 const gchar *identity;
952 gchar *identity_escaped;
953 gchar *title_escaped;
954 gchar *artist_escaped;
955 gchar *player_markup;
956 gchar *combined;
957
958 identity = gvc_mpris_player_get_identity (icon->priv->player);
959 title = gvc_mpris_player_get_title (icon->priv->player);
960 artist = gvc_mpris_player_get_artist (icon->priv->player);
961
962 identity_escaped = g_markup_escape_text (identity != NULL ? identity : "", -1);
963 title_escaped = g_markup_escape_text (title != NULL ? title : "", -1);
964 artist_escaped = g_markup_escape_text (artist != NULL ? artist : "", -1);
965
966 if (title_escaped[0] != '\0' && artist_escaped[0] != '\0')
967 player_markup = g_strdup_printf ("\n<small>%s\n%s - %s</small>",
968 identity_escaped,
969 artist_escaped,
970 title_escaped);
971 else if (title_escaped[0] != '\0')
972 player_markup = g_strdup_printf ("\n<small>%s\n%s</small>",
973 identity_escaped,
974 title_escaped);
975 else
976 player_markup = g_strdup_printf ("\n<small>%s</small>",
977 identity_escaped);
978
979 combined = g_strconcat (markup, player_markup, NULL);
980 g_free (markup);
981 g_free (player_markup);
982 g_free (identity_escaped);
983 g_free (title_escaped);
984 g_free (artist_escaped);
985 markup = combined;
986 }
987
988 gtk_widget_set_tooltip_markup (GTK_WIDGET (icon), markup);
989
990 g_free (markup);
991}
