66static void
67caja_background_load_from_file_metadata (CajaFile *file,
68 EelBackground *background)
69{
70 char *color, *image;
71
72 g_assert (EEL_IS_BACKGROUND (background));
73 g_assert (CAJA_IS_FILE (file));
74 g_assert (g_object_get_data (G_OBJECT (background), "eel_background_file") == file);
75
76 color = caja_file_get_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_COLOR, NULL);
77 image = caja_file_get_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_IMAGE, NULL);
78
79 /* if there's none, read the default from the theme */
80 if (color == NULL && image == NULL)
81 caja_background_get_default_settings (&color, &image);
82
83 /* Block the other handler while we are responding to changes
84 * in the metadata so it doesn't try to change the metadata.
85 */
86 g_signal_handlers_block_by_func (background, G_CALLBACK (caja_background_changed_cb), file);
87
88 eel_background_set_color (background, color);
89 /* non-tiled only available for desktop, at least for now */
90 eel_bg_set_placement (background, MATE_BG_PLACEMENT_TILED);
91 eel_background_set_image_uri (background, image);
92
93 /* Unblock the handler. */
94 g_signal_handlers_unblock_by_func (background, G_CALLBACK (caja_background_changed_cb), file);
95
96 g_free (color);
97 g_free (image);
98}

123/* handle the background changed signal */
124static void
125caja_background_changed_cb (EelBackground *background,
126 GdkDragAction action,
127 CajaFile *file)
128{
129 g_assert (EEL_IS_BACKGROUND (background));
130 g_assert (CAJA_IS_FILE (file));
131 g_assert (g_object_get_data (G_OBJECT (background), "eel_background_file") == file);
132
133 char *color = eel_background_get_color (background);
134 char *image = eel_background_get_image_uri (background);
135
136 /* Block the other handler while we are writing metadata so it doesn't
137 * try to change the background.
138 */
139 g_signal_handlers_block_by_func (file, G_CALLBACK (caja_background_settings_notify_cb),
140 background);
141
142 if (action != (GdkDragAction) CAJA_DND_ACTION_SET_AS_FOLDER_BACKGROUND &&
143 action != (GdkDragAction) CAJA_DND_ACTION_SET_AS_GLOBAL_BACKGROUND)
144 {
145 action = (GdkDragAction) GPOINTER_TO_INT (g_object_get_data (G_OBJECT (background),
146 "default_drag_action"));
147 }
148
149 if (action == (GdkDragAction) CAJA_DND_ACTION_SET_AS_GLOBAL_BACKGROUND)
150 {
151 caja_file_set_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_COLOR, NULL, NULL);
152 caja_file_set_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_IMAGE, NULL, NULL);
153
154 g_signal_handlers_block_by_func (caja_preferences,
155 G_CALLBACK (caja_background_theme_notify_cb),
156 background);
157
158 g_settings_set_string (caja_preferences,
159 CAJA_PREFERENCES_BACKGROUND_COLOR, color ? color : "");
160 g_settings_set_string (caja_preferences,
161 CAJA_PREFERENCES_BACKGROUND_URI, image ? image : "");
162
163 g_settings_set_boolean (caja_preferences, CAJA_PREFERENCES_BACKGROUND_SET, TRUE);
164
165 g_signal_handlers_unblock_by_func (caja_preferences,
166 G_CALLBACK (caja_background_theme_notify_cb),
167 background);
168 } else {
169 caja_file_set_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_COLOR, NULL, color);
170 caja_file_set_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_IMAGE, NULL, image);
171 }
172
173 /* Unblock the handler. */
174 g_signal_handlers_unblock_by_func (file, G_CALLBACK (caja_background_settings_notify_cb),
175 background);
176
177 g_free (color);
178 g_free (image);
179}

181/* handle the background reset signal by setting values from the current theme */
182static void
183caja_background_reset_cb (EelBackground *background,
184 CajaFile *file)
185{
186 char *color, *image;
187
188 /* Block the other handler while we are writing metadata so it doesn't
189 * try to change the background.
190 */
191 g_signal_handlers_block_by_func (file, G_CALLBACK (caja_background_settings_notify_cb),
192 background);
193
194 color = caja_file_get_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_COLOR, NULL);
195 image = caja_file_get_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_IMAGE, NULL);
196 if (!color && !image)
197 {
198 g_signal_handlers_block_by_func (caja_preferences,
199 G_CALLBACK (caja_background_theme_notify_cb),
200 background);
201 g_settings_set_boolean (caja_preferences, CAJA_PREFERENCES_BACKGROUND_SET, FALSE);
202 g_signal_handlers_unblock_by_func (caja_preferences,
203 G_CALLBACK (caja_background_theme_notify_cb),
204 background);
205 }
206 else
207 {
208 /* reset the metadata */
209 caja_file_set_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_COLOR, NULL, NULL);
210 caja_file_set_metadata (file, CAJA_METADATA_KEY_LOCATION_BACKGROUND_IMAGE, NULL, NULL);
211 }
212 g_free (color);
213 g_free (image);
214
215 /* Unblock the handler. */
216 g_signal_handlers_unblock_by_func (file, G_CALLBACK (caja_background_settings_notify_cb),
217 background);
218
219 caja_background_settings_notify_cb (file, background);
220}

236/* key routine that hooks up a background and location */
237void
238caja_connect_background_to_file_metadata (GtkWidget *widget,
239 CajaFile *file,
240 GdkDragAction default_drag_action)
241{
242 EelBackground *background;
243 gpointer old_file;
244
245 /* Get at the background object we'll be connecting. */
246 background = eel_get_widget_background (widget);
247
248 /* Check if it is already connected. */
249 old_file = g_object_get_data (G_OBJECT (background), "eel_background_file");
250 if (old_file == file)
251 return;
252
253 /* Disconnect old signal handlers. */
254 if (old_file != NULL)
255 {
256 g_assert (CAJA_IS_FILE (old_file));
257
258 g_signal_handlers_disconnect_by_func (background,
259 G_CALLBACK (caja_background_changed_cb), old_file);
260 g_signal_handlers_disconnect_by_func (background,
261 G_CALLBACK (caja_background_reset_cb), old_file);
262
263 g_object_weak_unref (G_OBJECT (background), caja_background_weak_notify, old_file);
264
265 g_signal_handlers_disconnect_by_func (old_file,
266 G_CALLBACK (caja_background_settings_notify_cb),
267 background);
268
269 caja_file_monitor_remove (old_file, background);
270
271 g_signal_handlers_disconnect_by_func (caja_preferences, caja_background_theme_notify_cb,
272 background);
273 }
274
275 /* Attach the new directory. */
276 caja_file_ref (file);
277 g_object_set_data_full (G_OBJECT (background), "eel_background_file",
278 file, (GDestroyNotify) caja_file_unref);
279
280 g_object_set_data (G_OBJECT (background), "default_drag_action",
281 GINT_TO_POINTER (default_drag_action));
282
283 /* Connect new signal handlers. */
284 if (file != NULL)
285 {
286 g_signal_connect_object (background, "settings_changed",
287 G_CALLBACK (caja_background_changed_cb), file, 0);
288
289 g_signal_connect_object (background, "reset",
290 G_CALLBACK (caja_background_reset_cb), file, 0);
291
292 g_signal_connect_object (file, "changed",
293 G_CALLBACK (caja_background_settings_notify_cb), background, 0);
294
295 g_object_weak_ref (G_OBJECT (background), caja_background_weak_notify, file);
296
297 /* arrange to receive file metadata */
298 caja_file_monitor_add (file, background, CAJA_FILE_ATTRIBUTE_INFO);
299
300 /* arrange for notification when the theme changes */
301 g_signal_connect (caja_preferences, "changed::" CAJA_PREFERENCES_BACKGROUND_SET,
302 G_CALLBACK(caja_background_theme_notify_cb), background);
303 g_signal_connect (caja_preferences, "changed::" CAJA_PREFERENCES_BACKGROUND_COLOR,
304 G_CALLBACK(caja_background_theme_notify_cb), background);
305 g_signal_connect (caja_preferences, "changed::" CAJA_PREFERENCES_BACKGROUND_URI,
306 G_CALLBACK(caja_background_theme_notify_cb), background);
307 }
308
309 /* Update the background based on the file metadata. */
310 caja_background_load_from_file_metadata (file, background);
311}

380static void
381desktop_background_resolve_settings (EelBackground *background)
382{
383 DesktopBackgroundState *state = desktop_background_get_state (background);
384 GtkWidget *toplevel;
385 GdkMonitor *monitor;
386
387 if (state == NULL)
388 {
389 return;
390 }
391
392 state->effective = mate_background_preferences;
393
394 toplevel = gtk_widget_get_toplevel (state->container);
395 monitor = toplevel != NULL ?
396 g_object_get_data (G_OBJECT (toplevel), "caja-desktop-monitor") : NULL;
397 if (monitor == NULL)
398 {
399 desktop_background_state_drop_monitor (state);
400 return;
401 }
402
403 if (state->monitor == NULL || state->monitor_object != monitor)
404 {
405 desktop_background_state_drop_monitor (state);
406 state->monitor = caja_preferences_get_monitor_settings (monitor);
407 if (state->monitor == NULL)
408 {
409 return;
410 }
411
412 state->monitor_object = monitor;
413 state->monitor_handler = g_signal_connect (state->monitor, "changed",
414 G_CALLBACK (desktop_background_monitor_changed_cb),
415 background);
416 }
417
418 if (!g_settings_get_boolean (state->monitor, MATE_BG_KEY_USE_SHARED))
419 {
420 state->effective = state->monitor;
421 }
422}

434/* delayed initializor of desktop background after GSettings changes */
435static gboolean
436desktop_background_prefs_change_event_idle_cb (EelBackground *background)
437{
438 gchar *desktop_color = NULL;
439 gboolean draw_now;
440 DesktopBackgroundState *state = desktop_background_get_state (background);
441
442 if (state != NULL)
443 {
444 state->reload_pending = FALSE;
445 }
446
447 desktop_background_resolve_settings (background);
448 eel_bg_load_from_gsettings (background,
449 desktop_background_effective_settings (background));
450
451 desktop_color = eel_bg_get_desktop_color (background);
452 eel_background_set_color (background, desktop_color);
453
454 draw_now = g_settings_get_boolean (desktop_background_effective_settings (background),
455 MATE_BG_KEY_DRAW_BACKGROUND);
456 if (state != NULL && state->last_draw != draw_now)
457 {
458 state->last_draw = draw_now;
459 g_signal_emit_by_name (background, "appearance_changed");
460 }
461
462 g_free(desktop_color);
463 g_object_unref (background);
464
465 return FALSE; /* remove from the list of event sources */
466}

526void
527caja_connect_desktop_background_to_settings (CajaIconContainer *icon_container)
528{
529 EelBackground *background;
530 DesktopBackgroundState *state;
531
532 background = eel_get_widget_background (GTK_WIDGET (icon_container));
533
534 /* Check if the desktop background is already connected to avoid
535 * accumulating duplicate signal handlers. This function is called
536 * on every begin_loading signal (e.g. during desktop rescans);
537 * without this guard, the "change-event" handler on the global
538 * mate_background_preferences object accumulates on each call,
539 * causing memory growth and CPU spikes when wallpaper settings change.
540 */
541 if (g_object_get_data (G_OBJECT (background),
542 "desktop_bg_connected") != NULL)
543 {
544 return;
545 }
546
547 eel_background_set_desktop (background, TRUE);
548
549 state = g_new0 (DesktopBackgroundState, 1);
550 state->container = GTK_WIDGET (icon_container);
551 state->effective = mate_background_preferences;
552 g_object_set_data_full (G_OBJECT (background), "caja-desktop-background-state",
553 state, desktop_background_state_free);
554
555 g_signal_connect_object (background, "settings_changed",
556 G_CALLBACK (desktop_background_changed_cb), NULL, 0);
557
558 g_signal_connect_object (background, "reset",
559 G_CALLBACK (desktop_background_reset_cb), NULL, 0);
560
561 desktop_background_resolve_settings (background);
562 state->last_draw = g_settings_get_boolean (desktop_background_effective_settings (background),
563 MATE_BG_KEY_DRAW_BACKGROUND);
564 eel_bg_load_from_gsettings (background,
565 desktop_background_effective_settings (background));
566
567 /* Connect to "change-event" signal to receive *groups of changes* before
568 * they are split out into multiple emissions of the "changed" signal.
569 */
570 state->shared_handler = g_signal_connect (mate_background_preferences, "change-event",
571 G_CALLBACK (desktop_background_prefs_change_event_cb),
572 background);
573
574 g_object_set_data (G_OBJECT (background),
575 "desktop_bg_connected",
576 GINT_TO_POINTER (1));
577
578 if (!gtk_widget_get_realized (GTK_WIDGET (icon_container)))
579 {
580 g_signal_connect_object (icon_container, "realize",
581 G_CALLBACK (desktop_background_container_realized_cb),
582 background, 0);
583 }
