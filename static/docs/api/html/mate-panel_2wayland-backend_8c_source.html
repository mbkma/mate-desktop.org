38static void
39wayland_panel_toplevel_apply_anchors (PanelToplevel* toplevel,
40 PanelOrientation orientation,
41 gboolean expand)
42{
43 GtkWindow* window;
44 gboolean anchor[GTK_LAYER_SHELL_EDGE_ENTRY_NUMBER];
45
46 window = GTK_WINDOW (toplevel);
47
48 for (int i = 0; i < GTK_LAYER_SHELL_EDGE_ENTRY_NUMBER; i++)
49 anchor[i] = expand;
50
51 switch (orientation) {
52 case PANEL_ORIENTATION_LEFT:
53 anchor[GTK_LAYER_SHELL_EDGE_LEFT] = TRUE;
54 anchor[GTK_LAYER_SHELL_EDGE_RIGHT] = FALSE;
55 break;
56 case PANEL_ORIENTATION_RIGHT:
57 anchor[GTK_LAYER_SHELL_EDGE_RIGHT] = TRUE;
58 anchor[GTK_LAYER_SHELL_EDGE_LEFT] = FALSE;
59 break;
60 case PANEL_ORIENTATION_TOP:
61 anchor[GTK_LAYER_SHELL_EDGE_TOP] = TRUE;
62 anchor[GTK_LAYER_SHELL_EDGE_BOTTOM] = FALSE;
63 break;
64 case PANEL_ORIENTATION_BOTTOM:
65 anchor[GTK_LAYER_SHELL_EDGE_BOTTOM] = TRUE;
66 anchor[GTK_LAYER_SHELL_EDGE_TOP] = FALSE;
67 break;
68 default:
69 g_warning ("Invalid panel orientation %d", orientation);
70 }
71
72 for (int i = 0; i < GTK_LAYER_SHELL_EDGE_ENTRY_NUMBER; i++)
73 gtk_layer_set_anchor (window, i, anchor[i]);
74}

77static void
78wayland_panel_toplevel_set_anchor (GtkWidget *widget,
79 GtkAllocation *alloc,
80 gpointer data)
81{
82 PanelToplevel* toplevel = data;
83
84 gulong id = GPOINTER_TO_SIZE (g_object_get_data (G_OBJECT (toplevel),
85 "wayland-set-anchor-id"));
86 if (id != 0)
87 g_signal_handler_disconnect (G_OBJECT (toplevel), id);
88 g_object_set_data (G_OBJECT (toplevel), "wayland-set-anchor-id", NULL);
89
90 wayland_panel_toplevel_apply_anchors (toplevel,
91 panel_toplevel_get_orientation (toplevel),
92 panel_toplevel_get_expand (toplevel));
93}

96static void
97wayland_panel_toplevel_set_anchor_default (gpointer data)
98{
99 PanelToplevel* toplevel = data;
100 GtkWindow* window;
101 GtkWidget *widget;
102
103 if (g_object_get_data (G_OBJECT (toplevel), "wayland-set-anchor-id") == NULL) {
104 window = GTK_WINDOW (toplevel);
105 gtk_layer_set_anchor (window, GTK_LAYER_SHELL_EDGE_TOP, TRUE);
106 gtk_layer_set_anchor (window, GTK_LAYER_SHELL_EDGE_BOTTOM, FALSE);
107 gtk_layer_set_anchor (window, GTK_LAYER_SHELL_EDGE_LEFT, TRUE);
108 gtk_layer_set_anchor (window, GTK_LAYER_SHELL_EDGE_RIGHT, FALSE);
109
110 widget = GTK_WIDGET (toplevel);
111 gulong id = g_signal_connect (widget, "size-allocate",
112 G_CALLBACK (wayland_panel_toplevel_set_anchor),
113 toplevel);
114 g_object_set_data (G_OBJECT (toplevel), "wayland-set-anchor-id",
115 GSIZE_TO_POINTER (id));
116 }
117
118 g_object_set_data (G_OBJECT (toplevel), "wayland-set-anchor-default-id", NULL);
119}

136static gboolean
137wayland_panel_toplevel_placement_changed (PanelToplevel* toplevel,
138 PanelOrientation orientation,
139 gboolean expand)
140{
141 gpointer prev = g_object_get_data (G_OBJECT (toplevel),
142 "wayland-panel-placement");
143 PanelOrientation prev_orientation =
144 prev != NULL ? (PanelOrientation) GPOINTER_TO_INT (prev) : orientation;
145 gboolean prev_expand = g_object_get_data (G_OBJECT (toplevel),
146 "wayland-panel-expand") != NULL;
147 gboolean changed = prev != NULL &&
148 (prev_orientation != orientation || prev_expand != expand);
149
150 g_object_set_data (G_OBJECT (toplevel), "wayland-panel-placement",
151 GINT_TO_POINTER (orientation));
152 g_object_set_data (G_OBJECT (toplevel), "wayland-panel-expand",
153 expand ? GINT_TO_POINTER (1) : NULL);
154
155 return changed;
156}

158void
159wayland_panel_toplevel_update_placement (PanelToplevel* toplevel)
160{
161 gboolean expand;
162 PanelOrientation orientation;
163 GdkMonitor *monitor;
164
165 monitor = panel_multimonitor_get_gdk_monitor (panel_toplevel_get_monitor (toplevel));
166 if (monitor != gtk_layer_get_monitor (GTK_WINDOW (toplevel)))
167 gtk_layer_set_monitor (GTK_WINDOW (toplevel), monitor);
168
169 orientation = panel_toplevel_get_orientation (toplevel);
170 expand = panel_toplevel_get_expand (toplevel);
171
172 if (wayland_panel_toplevel_placement_changed (toplevel, orientation, expand)) {
173 /* the panel size is going to change: let it resize in the
174 * two-anchor corner state first */
175 g_object_set_data (G_OBJECT (toplevel), "wayland-placement-orient",
176 GINT_TO_POINTER (orientation));
177 if (expand)
178 g_object_set_data (G_OBJECT (toplevel), "wayland-placement-expand",
179 GINT_TO_POINTER (1));
180 else
181 g_object_set_data (G_OBJECT (toplevel), "wayland-placement-expand", NULL);
182
183 wayland_panel_toplevel_queue_anchor_default (toplevel);
184 return;
185 }
186
187 wayland_panel_toplevel_apply_anchors (toplevel, orientation, expand);
188}

205static void
206wayland_panel_toplevel_monitor_removed (GdkDisplay *display,
207 GdkMonitor *monitor,
208 PanelToplevel *toplevel)
209{
210 if (gdk_display_get_n_monitors (display) == 0) {
211 /* With no monitor left the panel waits hidden: gtk-layer-shell
212 * would otherwise remap it from inside GDK's monitor-added, before
213 * GDK listens to the new wl_output, and GDK would never learn
214 * that monitor's geometry. */
215 g_idle_add (wayland_panel_toplevel_hide_while_no_monitors,
216 g_object_ref (toplevel));
217 return;
218 }
219 /* The compositor closes a layer surface whose output goes away, and
220 * a closed surface gets no frame callbacks, so GTK would never lay
221 * the panel out again: remap it at once. */
222 if (gtk_layer_get_monitor (GTK_WINDOW (toplevel)) == monitor)
223 gtk_layer_set_monitor (GTK_WINDOW (toplevel), NULL);
224}

263 wayland_panel_toplevel_update_placement (toplevel);

266void
267wayland_panel_toplevel_move_resize (PanelToplevel *toplevel,
268 const GdkRectangle *geometry,
269 const GdkRectangle *monitor_geom)
270{
271 GtkWindow* window;
272 gint margin[GTK_LAYER_SHELL_EDGE_ENTRY_NUMBER];
273
274 window = GTK_WINDOW (toplevel);
275
276 margin[GTK_LAYER_SHELL_EDGE_LEFT] = geometry->x - monitor_geom->x;
277 margin[GTK_LAYER_SHELL_EDGE_TOP] = geometry->y - monitor_geom->y;
278 margin[GTK_LAYER_SHELL_EDGE_RIGHT] =
279 monitor_geom->x + monitor_geom->width - (geometry->x + geometry->width);
280 margin[GTK_LAYER_SHELL_EDGE_BOTTOM] =
281 monitor_geom->y + monitor_geom->height - (geometry->y + geometry->height);
282
283 /*
284 * gtk-layer-shell driven layer surfaces never resize the underlying
285 * GDK window themselves; if the widget was reallocated to a different
286 * size (e.g. after an orientation change) the compositor keeps showing
287 * the stale surface footprint. Force the GDK window to the geometry
288 * size and invalidate it, so a correctly sized buffer is committed.
289 */
290 {
291 GdkWindow *gdk = gtk_widget_get_window (GTK_WIDGET (window));
292 if (gdk && (gdk_window_get_width (gdk) != geometry->width ||
293 gdk_window_get_height (gdk) != geometry->height)) {
294 gdk_window_resize (gdk, geometry->width, geometry->height);
295 gdk_window_invalidate_rect (gdk, NULL, FALSE);
296 }
297 }
298
299 for (int i = 0; i < GTK_LAYER_SHELL_EDGE_ENTRY_NUMBER; i++)
300 gtk_layer_set_margin (window, i, margin[i]);
301}

321GtkWindow *
322wayland_panel_toplevel_create_autohide_window (PanelToplevel *toplevel)
323{
324 GtkWindow *window;
325 GdkScreen *screen;
326
327 window = GTK_WINDOW (gtk_window_new (GTK_WINDOW_TOPLEVEL));
328
329 gtk_widget_set_name (GTK_WIDGET (window), "MatePanelWindowHidden");
330 gtk_window_set_title (window, "MatePanelWindowHidden");
331 gtk_window_set_decorated (window, FALSE);
332 gtk_window_set_type_hint (window, GDK_WINDOW_TYPE_HINT_UTILITY);
333 gtk_window_set_skip_taskbar_hint (window, TRUE);
334 gtk_window_set_skip_pager_hint (window, TRUE);
335 gtk_window_set_accept_focus (window, FALSE);
336 gtk_window_set_focus_on_map (window, FALSE);
337 gtk_widget_set_app_paintable (GTK_WIDGET (window), TRUE);
338
339 /* grab crossing events on this surface; they drive the autohide
340 * state machine independently of the (animated) panel surface */
341 gtk_widget_add_events (GTK_WIDGET (window),
342 GDK_ENTER_NOTIFY_MASK |
343 GDK_LEAVE_NOTIFY_MASK |
344 GDK_POINTER_MOTION_MASK);
345
346 /* visually transparent: this window is only an input area */
347 screen = gtk_widget_get_screen (GTK_WIDGET (window));
348 if (gdk_screen_get_rgba_visual (screen))
349 gtk_widget_set_visual (GTK_WIDGET (window),
350 gdk_screen_get_rgba_visual (screen));
351
352 gtk_layer_init_for_window (window);
353 gtk_layer_set_layer (window, GTK_LAYER_SHELL_LAYER_TOP);
354 gtk_layer_set_namespace (window, "panel");
355 gtk_layer_set_exclusive_zone (window, -1);
356 gtk_layer_set_keyboard_interactivity (window, FALSE);
357
358 /* place it with the {left, top} corner margins, like the panel */
359 gtk_layer_set_anchor (window, GTK_LAYER_SHELL_EDGE_LEFT, TRUE);
360 gtk_layer_set_anchor (window, GTK_LAYER_SHELL_EDGE_RIGHT, FALSE);
361 gtk_layer_set_anchor (window, GTK_LAYER_SHELL_EDGE_TOP, TRUE);
362 gtk_layer_set_anchor (window, GTK_LAYER_SHELL_EDGE_BOTTOM, FALSE);
363
364 g_signal_connect (window, "draw",
365 G_CALLBACK (wayland_panel_toplevel_autohide_window_draw),
366 NULL);
367
368 return window;
369}

371void
372wayland_panel_toplevel_autohide_window_show (GtkWindow *window,
373 PanelToplevel *toplevel,
374 const GdkRectangle *geometry,
375 const GdkRectangle *monitor_geom)
376{
377 GdkDisplay *display;
378 int monitor;
379 int n_monitors;
380
381 display = gdk_display_get_default ();
382 n_monitors = gdk_display_get_n_monitors (display);
383 monitor = panel_toplevel_get_monitor (toplevel);
384 if (monitor < 0 || monitor >= n_monitors)
385 monitor = 0;
386
387 gtk_layer_set_monitor (window, gdk_display_get_monitor (display, monitor));
388
389 gtk_widget_set_size_request (GTK_WIDGET (window),
390 geometry->width, geometry->height);
391 gtk_window_resize (window, geometry->width, geometry->height);
392
393 gtk_layer_set_margin (window, GTK_LAYER_SHELL_EDGE_LEFT,
394 geometry->x - monitor_geom->x);
395 gtk_layer_set_margin (window, GTK_LAYER_SHELL_EDGE_TOP,
396 geometry->y - monitor_geom->y);
397
398 if (!gtk_widget_get_visible (GTK_WIDGET (window)))
399 gtk_widget_show (GTK_WIDGET (window));
400}
