25add_head (GArray *heads, const char *name, gboolean enabled, int x, int y,
26 int width, int height, int preferred_width, int preferred_height,

51static void
52test_sizes (void)
53{
54 GArray *heads = g_array_new (FALSE, FALSE, sizeof (OutputHead));
55 int w, h;
56
57 add_head (heads, "A", TRUE, 0, 0, 1920, 1080, 1920, 1080, 0, 1.0);
58 add_head (heads, "B", TRUE, 0, 0, 3840, 2160, 3840, 2160, 1, 2.0);
59 add_head (heads, "C", FALSE, 0, 0, 0, 0, 1366, 768, 0, 1.0);
60 add_head (heads, "D", TRUE, 0, 0, 2560, 1600, 2560, 1600, 0, 1.5);
61
62 display_head_get_size (nth (heads, 0), &w, &h);
63 g_assert_cmpint (w, ==, 1920);
64 g_assert_cmpint (h, ==, 1080);
65 display_head_get_size (nth (heads, 1), &w, &h);
66 g_assert_cmpint (w, ==, 2160);
67 g_assert_cmpint (h, ==, 3840);
68 display_head_get_logical_size (nth (heads, 1), &w, &h);
69 g_assert_cmpint (w, ==, 1080);
70 g_assert_cmpint (h, ==, 1920);
71 display_head_get_size (nth (heads, 2), &w, &h);
72 g_assert_cmpint (w, ==, 1366);
73 g_assert_cmpint (h, ==, 768);
74 display_head_get_logical_size (nth (heads, 3), &w, &h);
75 g_assert_cmpint (w, ==, 1707);
76 g_assert_cmpint (h, ==, 1067);
77
78 display_heads_free (heads);
79}

81static void
82test_place_disabled (void)
83{
84 GArray *heads = g_array_new (FALSE, FALSE, sizeof (OutputHead));
85 int x, y, w, h;
86
87 add_head (heads, "off1", FALSE, 0, 0, 0, 0, 1366, 768, 0, 1.0);
88 add_head (heads, "A", TRUE, 0, 100, 1920, 1080, 1920, 1080, 0, 1.0);
89 add_head (heads, "B", TRUE, 1920, 0, 1280, 1024, 1280, 1024, 0, 1.0);
90 add_head (heads, "off2", FALSE, 1920, 0, 0, 0, 1024, 768, 0, 1.0);
91
92 display_heads_place_disabled (heads);
93
94 g_assert_cmpint (nth (heads, 0)->x, ==, 3200 + 1180 / 6);
95 g_assert_cmpint (nth (heads, 0)->y, ==, 0);
96 g_assert_cmpint (nth (heads, 3)->x, ==, 3200 + 1180 / 6 + 1366 + 1180 / 6);
97 g_assert_cmpint (nth (heads, 3)->y, ==, 0);
98 g_assert_cmpint (nth (heads, 1)->x, ==, 0);
99 g_assert_cmpint (nth (heads, 1)->y, ==, 100);
100 g_assert_cmpint (nth (heads, 2)->x, ==, 1920);
101
102 g_assert_true (display_heads_get_frame (heads, &x, &y, &w, &h));
103 g_assert_cmpint (x, ==, 0);
104 g_assert_cmpint (y, ==, 0);
105 g_assert_cmpint (w, ==, nth (heads, 3)->x + 1024);
106 g_assert_cmpint (h, ==, 1180);
107
108 display_heads_free (heads);
109}
