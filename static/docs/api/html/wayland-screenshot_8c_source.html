76static gboolean
77is_shm_format_supported (enum wl_shm_format format)
78{
79 return format == WL_SHM_FORMAT_ARGB8888 ||
80 format == WL_SHM_FORMAT_XRGB8888 ||
81 format == WL_SHM_FORMAT_ABGR8888 ||
82 format == WL_SHM_FORMAT_XBGR8888 ||
83 format == WL_SHM_FORMAT_BGR888;
84}
85
86static gint
87get_bpp_from_format (enum wl_shm_format format)
88{
89 switch (format)
90 {
91 case WL_SHM_FORMAT_ARGB8888:
92 case WL_SHM_FORMAT_XRGB8888:
93 case WL_SHM_FORMAT_ABGR8888:
94 case WL_SHM_FORMAT_XBGR8888:
95 return 4;
96 case WL_SHM_FORMAT_BGR888:
97 return 3;
98 default:
99 return 0;
100 }
101}

103static GdkPixbuf *
104convert_buffer_to_pixbuf (OutputData *output)
105{
106 guint8 *data = output->shm_data;
107 enum wl_shm_format format = output->format;
108 gboolean has_alpha = TRUE;
109 GdkPixbuf *pixbuf;
110
111 if (format == WL_SHM_FORMAT_ARGB8888 || format == WL_SHM_FORMAT_XRGB8888)
112 {
113 for (int y = 0; y < output->height; y++)
114 {
115 for (int x = 0; x < output->width; x++)
116 {
117 gint offset = y * output->stride + x * 4;
118 guint32 *px = (guint32 *)(gpointer)(data + offset);
119 guint8 blue = *px & 0xFF;
120 guint8 green = (*px >> 8) & 0xFF;
121 guint8 red = (*px >> 16) & 0xFF;
122 guint8 alpha = (*px >> 24) & 0xFF;
123 data[offset + 0] = red;
124 data[offset + 1] = green;
125 data[offset + 2] = blue;
126 data[offset + 3] = alpha;
127 }
128 }
129 }
130 else if (format == WL_SHM_FORMAT_ABGR8888 || format == WL_SHM_FORMAT_XBGR8888)
131 {
132 for (int y = 0; y < output->height; y++)
133 {
134 for (int x = 0; x < output->width; x++)
135 {
136 gint offset = y * output->stride + x * 4;
137 guint32 *px = (guint32 *)(gpointer)(data + offset);
138 guint8 red = *px & 0xFF;
139 guint8 green = (*px >> 8) & 0xFF;
140 guint8 blue = (*px >> 16) & 0xFF;
141 guint8 alpha = (*px >> 24) & 0xFF;
142 data[offset + 0] = red;
143 data[offset + 1] = green;
144 data[offset + 2] = blue;
145 data[offset + 3] = alpha;
146 }
147 }
148 }
149 else if (format == WL_SHM_FORMAT_BGR888)
150 {
151 has_alpha = FALSE;
152 for (int y = 0; y < output->height; y++)
153 {
154 for (int x = 0; x < output->width; x++)
155 {
156 gint offset = y * output->stride + x * 3;
157 guint8 *px = (guint8 *)(gpointer)(data + offset);
158 guint8 blue = px[2];
159 guint8 green = px[1];
160 guint8 red = px[0];
161 data[offset + 0] = red;
162 data[offset + 1] = green;
163 data[offset + 2] = blue;
164 }
165 }
166 }
167 else
168 {
169 g_warning ("Wayland screenshot: unsupported pixel format: 0x%x", format);
170 return NULL;
171 }
172
173 pixbuf = gdk_pixbuf_new_from_data (data, GDK_COLORSPACE_RGB, has_alpha, 8,
174 output->width, output->height,
175 output->stride, NULL, NULL);
176 return pixbuf;
177}

182handle_global (void *data, struct wl_registry *registry, uint32_t name,
183 const char *interface, uint32_t version)

196 cd->image_copy_capture_manager = wl_registry_bind (registry, name,
197 &ext_image_copy_capture_manager_v1_interface, 1);
198}

210/* --- wlr-screencopy frame listener --- */

212static void
213handle_frame_buffer (void *data, struct zwlr_screencopy_frame_v1 *frame,
214 uint32_t format, uint32_t width, uint32_t height, uint32_t stride)
215{
216 OutputData *output = data;
217 char template[] = "/tmp/mate-screenshot-buffer-XXXXXX";
218 int fd;
219
220 output->format = format;
221 output->width = width;
222 output->height = height;
223 output->stride = stride;
224 output->size = stride * height;
225
226 g_info ("Wayland screenshot: wlr frame buffer %dx%d stride=%d format=0x%x size=%d",
227 width, height, stride, format, output->size);
228
229 fd = mkstemp (template);
230 if (fd == -1)
231 {
232 g_warning ("Wayland screenshot: failed to create temp file");
233 output->capture_failed = TRUE;
234 return;
235 }
236 ftruncate (fd, output->size);
237 unlink (template);
238
239 output->shm_data = mmap (NULL, output->size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
240 if (output->shm_data == MAP_FAILED)
241 {
242 g_warning ("Wayland screenshot: failed to mmap buffer");
243 output->shm_data = NULL;
244 close (fd);
245 output->capture_failed = TRUE;
246 return;
247 }
248
249 output->pool = wl_shm_create_pool (output->client_data->shm, fd, output->size);
250 output->buffer = wl_shm_pool_create_buffer (output->pool, 0, width, height, stride, format);
251 close (fd);
252 wl_shm_pool_destroy (output->pool);
253 output->pool = NULL;
254
255 zwlr_screencopy_frame_v1_copy (frame, output->buffer);
256}

362static void
363handle_ext_session_done (void *data, struct ext_image_copy_capture_session_v1 *session)
364{
365 OutputData *output = data;
366 char template[] = "/tmp/mate-screenshot-buffer-XXXXXX";
367 int fd;
368
369 if (output->image_copy_capture_frame != NULL)
370 return;
371
372 output->stride = output->width * get_bpp_from_format (output->format);
373 output->size = output->stride * output->height;
374
375 if (output->size <= 0 || !output->has_format)
376 {
377 g_warning ("Wayland screenshot: ext session done but invalid size=%d or no format", output->size);
378 output->capture_failed = TRUE;
379 return;
380 }
381
382 fd = mkstemp (template);
383 if (fd == -1)
384 {
385 g_warning ("Wayland screenshot: failed to create temp file");
386 output->capture_failed = TRUE;
387 return;
388 }
389 ftruncate (fd, output->size);
390 unlink (template);
391
392 output->shm_data = mmap (NULL, output->size, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
393 if (output->shm_data == MAP_FAILED)
394 {
395 g_warning ("Wayland screenshot: failed to mmap buffer");
396 output->shm_data = NULL;
397 close (fd);
398 output->capture_failed = TRUE;
399 return;
400 }
401
402 output->pool = wl_shm_create_pool (output->client_data->shm, fd, output->size);
403 output->buffer = wl_shm_pool_create_buffer (output->pool, 0,
404 output->width, output->height,
405 output->stride, output->format);
406 close (fd);
407 wl_shm_pool_destroy (output->pool);
408 output->pool = NULL;
409
410 if (output->buffer == NULL)
411 {
412 g_warning ("Wayland screenshot: failed to create wl_buffer");
413 output->capture_failed = TRUE;
414 return;
415 }
416
417 output->image_copy_capture_frame =
418 ext_image_copy_capture_session_v1_create_frame (session);
419 ext_image_copy_capture_frame_v1_add_listener (output->image_copy_capture_frame,
420 &ext_frame_listener, output);
421 ext_image_copy_capture_frame_v1_attach_buffer (output->image_copy_capture_frame,
422 output->buffer);
423 ext_image_copy_capture_frame_v1_damage_buffer (output->image_copy_capture_frame,
424 0, 0, INT32_MAX, INT32_MAX);
425 ext_image_copy_capture_frame_v1_capture (output->image_copy_capture_frame);
426}

478static GdkPixbuf *
479compose_screenshot (GList *outputs)
480{
481 gint total_width = 0;
482 gint total_height = 0;
483
484 /* Calculate the virtual framebuffer size using GDK logical coordinates */
485 for (GList *elem = outputs; elem; elem = elem->next)
486 {
487 OutputData *od = elem->data;
488 GdkRectangle geometry;
489
490 gdk_monitor_get_geometry (od->monitor, &geometry);
491
492 gint sx = geometry.x + geometry.width;
493 gint sy = geometry.y + geometry.height;
494 if (sx > total_width)
495 total_width = sx;
496 if (sy > total_height)
497 total_height = sy;
498 }
499
500 if (total_width == 0 || total_height == 0)
501 return NULL;
502
503 /* Create the destination pixbuf for the virtual framebuffer */
504 GdkPixbuf *dest = gdk_pixbuf_new (GDK_COLORSPACE_RGB, FALSE, 8,
505 total_width, total_height);
506
507 /* Composite each output into the destination */
508 for (GList *elem = outputs; elem; elem = elem->next)
509 {
510 OutputData *od = elem->data;
511 GdkRectangle geometry;
512 GdkPixbuf *raw;
513 gdouble scale;
514 gdouble composite_scale;
515
516 if (od->capture_failed)
517 continue;
518
519 raw = convert_buffer_to_pixbuf (od);
520 if (raw == NULL)
521 continue;
522
523 /* Handle Y-inverted buffer from wlr-screencopy */
524 if (od->y_inverted)
525 {
526 GdkPixbuf *flipped = gdk_pixbuf_rotate_simple (raw, GDK_PIXBUF_ROTATE_UPSIDEDOWN);
527 g_object_unref (raw);
528 raw = flipped;
529 }
530
531 /* The capture buffer is in physical pixels; the destination uses the
532 * same GDK logical coordinates as the rest of the application, so we
533 * downscale each output by its own scale factor. */
534 gdk_monitor_get_geometry (od->monitor, &geometry);
535 scale = (gdouble) od->width / (gdouble) geometry.width;
536 if (scale < 1)
537 scale = 1;
538
539 composite_scale = 1.0 / scale;
540
541 g_info ("Wayland screenshot: compositing output %dx%d at (%d,%d) scale %.2f",
542 od->width, od->height, geometry.x, geometry.y, scale);
543
544 gdk_pixbuf_composite (raw, dest,
545 geometry.x, geometry.y,
546 geometry.width, geometry.height,
547 geometry.x, geometry.y,
548 composite_scale, composite_scale,
549 GDK_INTERP_BILINEAR, 255);
550 g_object_unref (raw);
551 }
552
553 return dest;
554}

556static GdkPixbuf *
557capture_screenshot (G_GNUC_UNUSED gboolean capture_all_monitors, gboolean show_mouse)
558{
559 ClientData client_data = { 0 };
560 GList *outputs = NULL;
561 GdkPixbuf *screenshot = NULL;
562 gboolean failure = FALSE;
563 int n_monitors;
564
565 client_data.display = gdk_wayland_display_get_wl_display (gdk_display_get_default ());
566 client_data.registry = wl_display_get_registry (client_data.display);
567 wl_registry_add_listener (client_data.registry, &registry_listener, &client_data);
568 wl_display_roundtrip (client_data.display);
569
570 if (client_data.shm == NULL)
571 {
572 g_warning ("Wayland screenshot: wl_shm not available");
573 free_client_data (&client_data);
574 return NULL;
575 }
576
577 if (client_data.output_image_capture_source_manager == NULL &&
578 client_data.image_copy_capture_manager == NULL &&
579 client_data.screencopy_manager == NULL)
580 {
581 g_warning ("Wayland screenshot: no capture protocol available");
582 free_client_data (&client_data);
583 return NULL;
584 }
585
586 n_monitors = gdk_display_get_n_monitors (gdk_display_get_default ());
587 for (int i = 0; i < n_monitors; i++)
588 {
589 OutputData *output;
590 GdkMonitor *monitor = gdk_display_get_monitor (gdk_display_get_default (), i);
591 struct wl_output *wl_output = gdk_wayland_monitor_get_wl_output (monitor);
592
593 if (wl_output == NULL)
594 {
595 g_warning ("Wayland screenshot: no wl_output for monitor %d", i);
596 continue;
597 }
598
599 output = g_new0 (OutputData, 1);
600 output->monitor = monitor;
601 output->client_data = &client_data;
602 outputs = g_list_append (outputs, output);
603
604 if (client_data.image_copy_capture_manager != NULL &&
605 client_data.output_image_capture_source_manager != NULL)
606 {
607 struct ext_image_capture_source_v1 *source =
608 ext_output_image_capture_source_manager_v1_create_source (
609 client_data.output_image_capture_source_manager, wl_output);
610
611 guint32 options = show_mouse ? EXT_IMAGE_COPY_CAPTURE_MANAGER_V1_OPTIONS_PAINT_CURSORS : 0;
612 output->image_copy_capture_session =
613 ext_image_copy_capture_manager_v1_create_session (
614 client_data.image_copy_capture_manager, source, options);
615
616 ext_image_copy_capture_session_v1_add_listener (
617 output->image_copy_capture_session, &ext_session_listener, output);
618
619 ext_image_capture_source_v1_destroy (source);
620 }
621 else if (client_data.screencopy_manager != NULL)
622 {
623 output->frame = zwlr_screencopy_manager_v1_capture_output (
624 client_data.screencopy_manager, show_mouse ? 1 : 0, wl_output);
625 zwlr_screencopy_frame_v1_add_listener (output->frame, &frame_listener, output);
626 }
627 }
628
629 /* Dispatch events until all outputs are captured or failed */
630 while (TRUE)
631 {
632 gboolean done = TRUE;
633 for (GList *elem = outputs; elem; elem = elem->next)
634 {
635 OutputData *od = elem->data;
636 if (!od->capture_done && !od->capture_failed)
637 done = FALSE;
638 if (od->capture_failed)
639 failure = TRUE;
640 }
641 if (done)
642 break;
643 wl_display_dispatch (client_data.display);
644 }
645
646 if (!failure && outputs != NULL)
647 {
648 screenshot = compose_screenshot (outputs);
649 }
650
651 free_client_data (&client_data);
652 g_list_free_full (outputs, free_output_data);
653
654 return screenshot;
655}

687GdkPixbuf *
688wayland_screenshot_capture_screen (gboolean include_pointer)
689{
690 if (!wayland_initialized)
