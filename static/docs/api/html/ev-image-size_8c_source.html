30static EvImageSizeResult
31jpeg_size_from_header (const guchar *data,
32 gsize len,
33 gint *width,
34 gint *height)
35{
36 /* See https://www.w3.org/Graphics/JPEG/itu-t81.pdf page 39 (section
37 * B.2.2) */
38 gsize pos = 2;
39
40 while (1) {
41 guchar marker;
42
43 if (pos + 2 > len)
44 return EV_IMAGE_SIZE_NEED_MORE;
45 if (data[pos] != 0xff)
46 return EV_IMAGE_SIZE_UNKNOWN;
47 marker = data[pos + 1];
48 if (marker == 0xff) {
49 /* Fill byte */
50 pos++;
51 continue;
52 }
53 pos += 2;
54 /* Markers without a payload */
55 if (marker == 0x01 || (marker >= 0xd0 && marker <= 0xd7))
56 continue;
57 /* End of image or start of scan before any frame header */
58 if (marker == 0xd9 || marker == 0xda)
59 return EV_IMAGE_SIZE_UNKNOWN;
60
61 if (pos + 2 > len)
62 return EV_IMAGE_SIZE_NEED_MORE;
63
64 /* Start of frame markers, except DHT, JPG and DAC */
65 if (marker >= 0xc0 && marker <= 0xcf &&
66 marker != 0xc4 && marker != 0xc8 && marker != 0xcc) {
67 if (pos + 7 > len)
68 return EV_IMAGE_SIZE_NEED_MORE;
69 *height = READ_BE16 (data + pos + 3);
70 *width = READ_BE16 (data + pos + 5);
71 return EV_IMAGE_SIZE_FOUND;
72 }
73
74 if (READ_BE16 (data + pos) < 2)
75 return EV_IMAGE_SIZE_UNKNOWN;
76 pos += READ_BE16 (data + pos);
77 }
78}

80static EvImageSizeResult
81webp_size_from_header (const guchar *data,
82 gsize len,
83 gint *width,
84 gint *height)
85{
86 /* See https://developers.google.com/speed/webp/docs/riff_container
87 * and https://datatracker.ietf.org/doc/html/rfc6386 (lossy) as well as
88 * https://developers.google.com/speed/webp/docs/webp_lossless_bitstream_specification
89 * (lossless). */
90 const guchar *chunk = data + 12;
91
92 if (len < 30)
93 return EV_IMAGE_SIZE_NEED_MORE;
94
95 if (memcmp (chunk, "VP8 ", 4) == 0) {
96 /* Lossy: 3-byte frame tag, then the 9d 01 2a start code */
97 if (memcmp (chunk + 11, "\x9d\x01\x2a", 3) != 0)
98 return EV_IMAGE_SIZE_UNKNOWN;
99 *width = READ_LE16 (chunk + 14) & 0x3fff;
100 *height = READ_LE16 (chunk + 16) & 0x3fff;
101 return EV_IMAGE_SIZE_FOUND;
102 } else if (memcmp (chunk, "VP8L", 4) == 0) {
103 /* Lossless: signature byte, then 14-bit width-1 and height-1 */
104 const guchar *b = chunk + 9;
105 if (chunk[8] != 0x2f)
106 return EV_IMAGE_SIZE_UNKNOWN;
107 *width = 1 + (b[0] | (b[1] & 0x3f) << 8);
108 *height = 1 + (b[1] >> 6 | b[2] << 2 | (b[3] & 0x0f) << 10);
109 return EV_IMAGE_SIZE_FOUND;
110 } else if (memcmp (chunk, "VP8X", 4) == 0) {
111 /* Extended: 24-bit canvas width-1 and height-1 */
112 *width = 1 + READ_LE24 (chunk + 12);
113 *height = 1 + READ_LE24 (chunk + 15);
114 return EV_IMAGE_SIZE_FOUND;
115 }
116
117 return EV_IMAGE_SIZE_UNKNOWN;
118}

120static EvImageSizeResult
121png_size_from_header (const guchar *data,
122 gsize len,
123 gint *width,
124 gint *height)
125{
126 /* The IHDR chunk always comes first, right after the signature.
127 * See https://www.w3.org/TR/png/#11IHDR */
128 if (len < 24)
129 return EV_IMAGE_SIZE_NEED_MORE;
130 if (memcmp (data + 12, "IHDR", 4) != 0)
131 return EV_IMAGE_SIZE_UNKNOWN;
132 *width = READ_BE32 (data + 16);
133 *height = READ_BE32 (data + 20);
134 return EV_IMAGE_SIZE_FOUND;
135}

137static EvImageSizeResult
138gif_size_from_header (const guchar *data,
139 gsize len,
140 gint *width,
141 gint *height)
142{
143 /* The logical screen size follows the signature. See
144 * https://www.w3.org/Graphics/GIF/spec-gif89a.txt ("Logical Screen
145 * Descriptor" section, number 18). */
146 *width = READ_LE16 (data + 6);
147 *height = READ_LE16 (data + 8);
148 return EV_IMAGE_SIZE_FOUND;
149}

166EvImageSizeResult
167ev_image_size_from_header (const guchar *data,
168 gsize len,
169 gint *width,
170 gint *height)
171{
172 EvImageSizeResult ret = EV_IMAGE_SIZE_UNKNOWN;
173
174 if (len < 12)
175 return EV_IMAGE_SIZE_NEED_MORE;
176
177 if (data[0] == 0xff && data[1] == 0xd8) {
178 ret = jpeg_size_from_header (data, len, width, height);
179 } else if (memcmp (data, "\x89PNG\r\n\x1a\n", 8) == 0) {
180 ret = png_size_from_header (data, len, width, height);
181 } else if (memcmp (data, "GIF87a", 6) == 0 || memcmp (data, "GIF89a", 6) == 0) {
182 ret = gif_size_from_header (data, len, width, height);
183 } else if (memcmp (data, "RIFF", 4) == 0 && memcmp (data + 8, "WEBP", 4) == 0) {
184 ret = webp_size_from_header (data, len, width, height);
185 }
186
187 if (ret == EV_IMAGE_SIZE_FOUND && (*width <= 0 || *height <= 0))
188 return EV_IMAGE_SIZE_UNKNOWN;
189
190 return ret;
191}
