107static gboolean
108gpm_brightness_helper_strtoint (const gchar *text, gint *value)
109{
110 gchar *endptr = NULL;
111 gint64 value_raw;
112
113 if (text == NULL)
114 return FALSE;
115
116 value_raw = g_ascii_strtoll (text, &endptr, 10);
117
118 if (endptr == text)
119 return FALSE;
120 if (errno == ERANGE || value_raw > G_MAXINT || value_raw < G_MININT)
121 return FALSE;
122
123 *value = (gint) value_raw;
124 return TRUE;
125}

130static gint
131gpm_brightness_helper_get_value (const gchar *argument)
132{
133 gboolean ret;
134 GError *error = NULL;
135 gchar *stdout_data = NULL;
136 gint exit_status = 0;
137 gint value = -1;
138 gchar *command = NULL;
139
140 /* get the data */
141 command = g_strdup_printf (LIBEXECDIR "/mate-power-backlight-helper --%s", argument);
142 ret = g_spawn_command_line_sync (command,
143 &stdout_data, NULL, &exit_status, &error);
144 if (!ret) {
145 g_error ("failed to get value: %s", error->message);
146 g_error_free (error);
147 goto out;
148 }
149 g_debug ("executing %s retval: %i", command, exit_status);
150
151 /* parse for a number */
152 ret = gpm_brightness_helper_strtoint (stdout_data, &value);
153 if (!ret)
154 goto out;
155out:
156 g_free (command);
157 g_free (stdout_data);
158 return value;
159}

205static gboolean
206gpm_brightness_output_get_internal (GpmBrightness *brightness, RROutput output, guint *cur)
207{
208 unsigned long nitems;
209 unsigned long bytes_after;
210 guint *prop;
211 Atom actual_type;
212 int actual_format;
213 gboolean ret = FALSE;
214
215 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
216
217 if (brightness->priv->backlight == None)
218 return FALSE;
219
220 if (XRRGetOutputProperty (brightness->priv->dpy, output, brightness->priv->backlight,
221 0, 4, False, False, None,
222 &actual_type, &actual_format,
223 &nitems, &bytes_after, ((unsigned char **)&prop)) != Success) {
224 g_debug ("failed to get property");
225 return FALSE;
226 }
227 if (actual_type == XA_INTEGER && nitems == 1 && actual_format == 32) {
228 memcpy (cur, prop, sizeof (guint));
229 ret = TRUE;
230 }
231 XFree (prop);
232 return ret;
233}

238static gboolean
239gpm_brightness_output_set_internal (GpmBrightness *brightness, RROutput output, guint value)
240{
241 GdkDisplay *display;
242
243 gboolean ret = TRUE;
244
245 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
246
247 /* don't abort on error */
248 display = gdk_display_get_default ();
249 gdk_x11_display_error_trap_push (display);
250 XRRChangeOutputProperty (brightness->priv->dpy, output, brightness->priv->backlight, XA_INTEGER, 32,
251 PropModeReplace, (unsigned char *) &value, 1);
252 XFlush (brightness->priv->dpy);
253 gdk_display_flush (display);
254 if (gdk_x11_display_error_trap_pop (display)) {
255 g_warning ("failed to XRRChangeOutputProperty for brightness %u", value);
256 ret = FALSE;
257 }
258 /* we changed the hardware */
259 if (ret)
260 brightness->priv->hw_changed = TRUE;
261 return ret;
262}

267static gboolean
268gpm_brightness_setup_display (GpmBrightness *brightness)
269{
270 gint major, minor;
271
272 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
273
274 /* get the display */
275 brightness->priv->dpy = GDK_DISPLAY_XDISPLAY (gdk_display_get_default());
276 if (!brightness->priv->dpy) {
277 g_error ("Cannot open display");
278 return FALSE;
279 }
280 /* is XRandR new enough? */
281 if (!XRRQueryVersion (brightness->priv->dpy, &major, &minor)) {
282 g_debug ("RandR extension missing");
283 return FALSE;
284 }
285 if (major < 1 || (major == 1 && minor < 3)) {
286 g_debug ("RandR version %d.%d too old", major, minor);
287 return FALSE;
288 }
289 /* Can we support "Backlight" */
290 brightness->priv->backlight = XInternAtom (brightness->priv->dpy, "Backlight", True);
291 if (brightness->priv->backlight == None) {
292 /* Do we support "BACKLIGHT" (legacy) */
293 brightness->priv->backlight = XInternAtom (brightness->priv->dpy, "BACKLIGHT", True);
294 if (brightness->priv->backlight == None) {
295 g_debug ("No outputs have backlight property");
296 return FALSE;
297 }
298 }
299 return TRUE;
300}

305static gboolean
306gpm_brightness_output_get_limits (GpmBrightness *brightness, RROutput output,
307 guint *min, guint *max)
308{
309 XRRPropertyInfo *info;
310 gboolean ret = TRUE;
311
312 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
313
314 info = XRRQueryOutputProperty (brightness->priv->dpy, output, brightness->priv->backlight);
315 if (info == NULL) {
316 g_debug ("could not get output property");
317 return FALSE;
318 }
319 if (!info->range || info->num_values != 2) {
320 g_debug ("was not range");
321 ret = FALSE;
322 goto out;
323 }
324 *min = info->values[0];
325 *max = info->values[1];
326out:
327 XFree (info);
328 return ret;
329}

334static gboolean
335gpm_brightness_output_get_percentage (GpmBrightness *brightness, RROutput output)
336{
337 guint cur;
338 gboolean ret;
339 guint min, max;
340 guint percentage;
341
342 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
343
344 ret = gpm_brightness_output_get_internal (brightness, output, &cur);
345 if (!ret)
346 return FALSE;
347 ret = gpm_brightness_output_get_limits (brightness, output, &min, &max);
348 if (!ret || min == max)
349 return FALSE;
350 g_debug ("hard value=%u, min=%u, max=%u", cur, min, max);
351 percentage = egg_discrete_to_percent (cur, (max-min)+1);
352 g_debug ("percentage %u", percentage);
353 brightness->priv->shared_value = percentage;
354 return TRUE;
355}

360static gboolean
361gpm_brightness_output_down (GpmBrightness *brightness, RROutput output)
362{
363 guint cur;
364 guint step;
365 gboolean ret;
366 guint min, max;
367
368 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
369
370 ret = gpm_brightness_output_get_internal (brightness, output, &cur);
371 if (!ret)
372 return FALSE;
373 ret = gpm_brightness_output_get_limits (brightness, output, &min, &max);
374 if (!ret || min == max)
375 return FALSE;
376 g_debug ("hard value=%u, min=%u, max=%u", cur, min, max);
377 if (cur == min) {
378 g_debug ("already min");
379 return TRUE;
380 }
381 step = gpm_brightness_get_step ((max-min)+1);
382 if (cur < step) {
383 g_debug ("truncating to %u", min);
384 cur = min;
385 } else {
386 cur -= step;
387 }
388 ret = gpm_brightness_output_set_internal (brightness, output, cur);
389 return ret;
390}

427static gboolean
428gpm_brightness_output_set (GpmBrightness *brightness, RROutput output)
429{
430 guint cur;
431 gboolean ret;
432 guint min, max;
433 gint i;
434 gint shared_value_abs;
435 guint step;
436
437 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
438
439 ret = gpm_brightness_output_get_internal (brightness, output, &cur);
440 if (!ret)
441 return FALSE;
442 ret = gpm_brightness_output_get_limits (brightness, output, &min, &max);
443 if (!ret || min == max)
444 return FALSE;
445
446 shared_value_abs = egg_discrete_from_percent (brightness->priv->shared_value, (max-min)+1);
447 g_debug ("percent=%u, absolute=%i", brightness->priv->shared_value, shared_value_abs);
448
449 g_debug ("hard value=%u, min=%u, max=%u", cur, min, max);
450 if (shared_value_abs > (gint) max)
451 shared_value_abs = max;
452 if (shared_value_abs < (gint) min)
453 shared_value_abs = min;
454 if ((gint) cur == shared_value_abs) {
455 g_debug ("already set %u", cur);
456 return TRUE;
457 }
458
459 /* step the correct way */
460 if ((gint) cur < shared_value_abs) {
461 /* some adaptors have a large number of steps */
462 step = gpm_brightness_get_step (shared_value_abs - cur);
463 g_debug ("using step of %u", step);
464
465 /* going up */
466 for (i=cur; i<=shared_value_abs; i+=step) {
467 ret = gpm_brightness_output_set_internal (brightness, output, i);
468 if (!ret)
469 break;
470 if ((gint) cur != shared_value_abs)
471 g_usleep (1000 * GPM_BRIGHTNESS_DIM_INTERVAL);
472 }
473 } else {
474 /* some adaptors have a large number of steps */
475 step = gpm_brightness_get_step (cur - shared_value_abs);
476 g_debug ("using step of %u", step);
477
478 /* going down */
479 for (i=cur; i>=shared_value_abs; i-=step) {
480 ret = gpm_brightness_output_set_internal (brightness, output, i);
481 if (!ret)
482 break;
483 if ((gint) cur != shared_value_abs)
484 g_usleep (1000 * GPM_BRIGHTNESS_DIM_INTERVAL);
485 }
486 }
487 return TRUE;
488}

493static gboolean
494gpm_brightness_foreach_resource (GpmBrightness *brightness, GpmXRandROp op, XRRScreenResources *resources)
495{
496 gint i;
497 gboolean ret;
498 gboolean success_any = FALSE;
499 RROutput output;
500
501 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
502
503 /* do for each output */
504 for (i=0; i<resources->noutput; i++) {
505 output = resources->outputs[i];
506 g_debug ("resource %i of %i", i+1, resources->noutput);
507 if (op==ACTION_BACKLIGHT_GET) {
508 ret = gpm_brightness_output_get_percentage (brightness, output);
509 } else if (op==ACTION_BACKLIGHT_INC) {
510 ret = gpm_brightness_output_up (brightness, output);
511 } else if (op==ACTION_BACKLIGHT_DEC) {
512 ret = gpm_brightness_output_down (brightness, output);
513 } else if (op==ACTION_BACKLIGHT_SET) {
514 ret = gpm_brightness_output_set (brightness, output);
515 } else {
516 ret = FALSE;
517 g_warning ("op not known");
518 }
519 if (ret) {
520 success_any = TRUE;
521 }
522 }
523 return success_any;
524}

529static gboolean
530gpm_brightness_foreach_screen (GpmBrightness *brightness, GpmXRandROp op)
531{
532 guint i;
533 guint length;
534 XRRScreenResources *resource;
535 gboolean ret;
536 gboolean success_any = FALSE;
537
538 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
539
540 /* Return immediately if we can't use XRandR */
541 if (!brightness->priv->has_extension)
542 return FALSE;
543
544 /* do for each screen */
545 length = brightness->priv->resources->len;
546 for (i=0; i<length; i++) {
547 resource = (XRRScreenResources *) g_ptr_array_index (brightness->priv->resources, i);
548 g_debug ("using resource %p", resource);
549 ret = gpm_brightness_foreach_resource (brightness, op, resource);
550 if (ret)
551 success_any = TRUE;
552 }
553 XSync (brightness->priv->dpy, False);
554 return success_any;
555}

587static void
588gpm_brightness_update_cache (GpmBrightness *brightness)
589{
590 guint length;
591 Window root;
592 GdkScreen *gscreen;
593 GdkDisplay *display;
594 XRRScreenResources *resource;
595
596 g_return_if_fail (GPM_IS_BRIGHTNESS (brightness));
597
598 /* invalidate and remove all the previous entries */
599 length = brightness->priv->resources->len;
600 if (length > 0)
601 g_ptr_array_set_size (brightness->priv->resources, 0);
602
603 display = gdk_display_get_default ();
604 gscreen = gdk_display_get_default_screen (display);
605
606 /* if we have not setup the changed on the monitor, set it here */
607 if (g_object_get_data (G_OBJECT (gscreen), "gpk-set-monitors-changed") == NULL) {
608 g_debug ("watching ::monitors_changed on %p", gscreen);
609 g_object_set_data (G_OBJECT (gscreen), "gpk-set-monitors-changed", (gpointer) "true");
610 g_signal_connect (G_OBJECT (gscreen), "monitors_changed",
611 G_CALLBACK (gpm_brightness_monitors_changed), brightness);
612 }
613
614 root = RootWindow (brightness->priv->dpy, 0);
615
616 gdk_x11_display_error_trap_push (display);
617 resource = XRRGetScreenResourcesCurrent (brightness->priv->dpy, root);
618 if (gdk_x11_display_error_trap_pop (display) || resource == NULL) {
619 g_warning ("failed to XRRGetScreenResourcesCurrent");
620 }
621
622 if (resource != NULL) {
623 g_debug ("adding resource %p", resource);
624 g_ptr_array_add (brightness->priv->resources, resource);
625 }
626}

635static gboolean
636gpm_brightness_trust_cache (GpmBrightness *brightness)
637{
638 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
639 /* only return the cached value if the cache is trusted and we have change events */
640 if (brightness->priv->cache_trusted && brightness->priv->has_changed_events) {
641 g_debug ("using cache for value %u (okay)", brightness->priv->cache_percentage);
642 return TRUE;
643 }
644
645 /* can we trust that if we set a value 5 minutes ago, will it still be valid now?
646 * if we have multiple things setting policy on the workstation, e.g. fast user switching
647 * or kpowersave, then this will be invalid -- this logic may be insane */
648 if (GPM_SOLE_SETTER_USE_CACHE && brightness->priv->cache_trusted) {
649 g_debug ("using cache for value %u (probably okay)", brightness->priv->cache_percentage);
650 return TRUE;
651 }
652 return FALSE;
653}

662gboolean
663gpm_brightness_set (GpmBrightness *brightness, guint percentage, gboolean *hw_changed)
664{
665 gboolean ret = FALSE;
666 gboolean trust_cache;
667
668 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
669
670 /* can we check the new value with the cache? */
671 trust_cache = gpm_brightness_trust_cache (brightness);
672 if (trust_cache && percentage == brightness->priv->cache_percentage) {
673 g_debug ("not setting the same value %u", percentage);
674 return TRUE;
675 }
676
677 /* set the value we want */
678#ifdef HAVE_X11
679 brightness->priv->shared_value = percentage;
680#endif
681
682 /* reset to not-changed */
683 brightness->priv->hw_changed = FALSE;
684
685#ifdef HAVE_X11
686 ret = gpm_brightness_foreach_screen (brightness, ACTION_BACKLIGHT_SET);
687#endif
688
689 /* legacy fallback */
690 if (!ret) {
691 if (brightness->priv->extension_levels < 0)
692 brightness->priv->extension_levels = gpm_brightness_helper_get_value ("get-max-brightness");
693 brightness->priv->extension_current = egg_discrete_from_percent (percentage, brightness->priv->extension_levels+1);
694 ret = gpm_brightness_helper_set_value ("set-brightness", brightness->priv->extension_current);
695 }
696
697 /* did the hardware have to be modified? */
698 if (ret && hw_changed != NULL)
699 *hw_changed = brightness->priv->hw_changed;
700
701 /* we did something to the hardware, so untrusted */
702 if (ret)
703 brightness->priv->cache_trusted = FALSE;
704
705 return ret;
706}

716gboolean
717gpm_brightness_get (GpmBrightness *brightness, guint *percentage)
718{
719 gboolean ret = FALSE;
720 gboolean trust_cache;
721 guint percentage_local;
722
723 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
724 g_return_val_if_fail (percentage != NULL, FALSE);
725
726 /* can we use the cache? */
727 trust_cache = gpm_brightness_trust_cache (brightness);
728 if (trust_cache) {
729 *percentage = brightness->priv->cache_percentage;
730 return TRUE;
731 }
732
733 /* get the brightness from hardware -- slow */
734#ifdef HAVE_X11
735 ret = gpm_brightness_foreach_screen (brightness, ACTION_BACKLIGHT_GET);
736 percentage_local = brightness->priv->shared_value;
737#endif
738
739 /* legacy fallback */
740 if (!ret) {
741 if (brightness->priv->extension_levels < 0)
742 brightness->priv->extension_levels = gpm_brightness_helper_get_value ("get-max-brightness");
743 brightness->priv->extension_current = gpm_brightness_helper_get_value ("get-brightness");
744 percentage_local = egg_discrete_to_percent (brightness->priv->extension_current, brightness->priv->extension_levels+1);
745 ret = TRUE;
746 }
747
748 /* valid? */
749 if (percentage_local > 100) {
750 g_warning ("percentage value of %u will be truncated", percentage_local);
751 percentage_local = 100;
752 }
753
754 /* a new value is always trusted if the method and checks succeed */
755 if (ret) {
756 brightness->priv->cache_percentage = percentage_local;
757 brightness->priv->cache_trusted = TRUE;
758 *percentage = percentage_local;
759 } else {
760 brightness->priv->cache_trusted = FALSE;
761 }
762 return ret;
763}

773gboolean
774gpm_brightness_up (GpmBrightness *brightness, gboolean *hw_changed)
775{
776 gboolean ret = FALSE;
777 guint step;
778
779 g_return_val_if_fail (GPM_IS_BRIGHTNESS (brightness), FALSE);
780
781 /* reset to not-changed */
782 brightness->priv->hw_changed = FALSE;
783
784#ifdef HAVE_X11
785 ret = gpm_brightness_foreach_screen (brightness, ACTION_BACKLIGHT_INC);
786#endif
787
788 /* did the hardware have to be modified? */
789 if (ret && hw_changed != NULL)
790 *hw_changed = brightness->priv->hw_changed;
791
792 /* we did something to the hardware, so untrusted */
793 if (ret)
794 brightness->priv->cache_trusted = FALSE;
795
796 /* legacy fallback */
797 if (!ret) {
798 if (brightness->priv->extension_levels < 0)
799 brightness->priv->extension_levels = gpm_brightness_helper_get_value ("get-max-brightness");

829{

834
835 /* reset to not-changed */
836 brightness->priv->hw_changed = FALSE;
837

854 brightness->priv->extension_current = gpm_brightness_helper_get_value ("get-brightness");

857 if (brightness->priv->extension_current > 0) {
858 step = gpm_brightness_get_step (brightness->priv->extension_levels);
859 brightness->priv->extension_current -= step;
860 if (brightness->priv->extension_current < 0)
861 brightness->priv->extension_current = 0;

863 }
864 if (hw_changed != NULL)
865 *hw_changed = ret;
866 brightness->priv->cache_trusted = FALSE;

868 }
869out:
