112static gint64
113egg_idletime_xsyncvalue_to_int64 (XSyncValue value)
114{
115 return ((guint64) XSyncValueHigh32 (value)) << 32 | (guint64) XSyncValueLow32 (value);

121static void
122egg_idletime_xsync_alarm_set (EggIdletime *idletime, EggIdletimeAlarm *alarm, EggIdletimeAlarmType alarm_type)
123{
124 XSyncAlarmAttributes attr;
125 XSyncValue delta;
126 unsigned int flags;
127 XSyncTestType test;
128
129 /* just remove it */
130 if (alarm_type == EGG_IDLETIME_ALARM_TYPE_DISABLED) {
131 if (alarm->xalarm) {
132 XSyncDestroyAlarm (idletime->priv->dpy, alarm->xalarm);
133 alarm->xalarm = None;
134 }
135 return;
136 }
137
138 /* which way do we do the test? */
139 if (alarm_type == EGG_IDLETIME_ALARM_TYPE_POSITIVE)
140 test = XSyncPositiveTransition;
141 else
142 test = XSyncNegativeTransition;
143
144 XSyncIntToValue (&delta, 0);
145
146 attr.trigger.counter = idletime->priv->idle_counter;
147 attr.trigger.value_type = XSyncAbsolute;
148 attr.trigger.test_type = test;
149 attr.trigger.wait_value = alarm->timeout;
150 attr.delta = delta;
151
152 flags = XSyncCACounter | XSyncCAValueType | XSyncCATestType | XSyncCAValue | XSyncCADelta;
153
154 if (alarm->xalarm)
155 XSyncChangeAlarm (idletime->priv->dpy, alarm->xalarm, flags, &attr);
156 else
157 alarm->xalarm = XSyncCreateAlarm (idletime->priv->dpy, flags, &attr);
158}

179static void
180egg_idletime_set_reset_alarm (EggIdletime *idletime, XSyncAlarmNotifyEvent *alarm_event)
181{
182 EggIdletimeAlarm *alarm;
183 int overflow;
184 XSyncValue add;
185 gint64 current, reset_threshold;
186
187 alarm = egg_idletime_alarm_find_id (idletime, 0);
188
189 if (!idletime->priv->reset_set) {
190 /* don't match on the current value because
191 * XSyncNegativeComparison means less or equal. */
192 XSyncIntToValue (&add, -1);
193 XSyncValueAdd (&alarm->timeout, alarm_event->counter_value, add, &overflow);
194
195 /* set the reset alarm to fire the next time
196 * idletime->priv->idle_counter < the current counter value */
197 egg_idletime_xsync_alarm_set (idletime, alarm, EGG_IDLETIME_ALARM_TYPE_NEGATIVE);
198
199 /* don't try to set this again if multiple timers are going off in sequence */
200 idletime->priv->reset_set = TRUE;
201
202 current = egg_idletime_get_time (idletime);
203 reset_threshold = egg_idletime_xsyncvalue_to_int64 (alarm->timeout);
204 if (current < reset_threshold) {
205 /* We've missed the alarm already */
206 egg_idletime_alarm_reset_all (idletime);
207 }
208 }
209}

230static GdkFilterReturn
231egg_idletime_event_filter_cb (GdkXEvent *gdkxevent, GdkEvent *event, gpointer data)
232{
233 EggIdletimeAlarm *alarm;
234 XEvent *xevent = (XEvent *) gdkxevent;
235 EggIdletime *idletime = (EggIdletime *) data;
236 XSyncAlarmNotifyEvent *alarm_event;
237
238 /* no point continuing */
239 if (xevent->type != idletime->priv->sync_event + XSyncAlarmNotify)
240 return GDK_FILTER_CONTINUE;
241
242 alarm_event = (XSyncAlarmNotifyEvent *) xevent;
243
244 /* did we match one of our alarms? */
245 alarm = egg_idletime_alarm_find_event (idletime, alarm_event);
246 if (alarm == NULL)
247 return GDK_FILTER_CONTINUE;
248
249 /* are we the reset alarm? */
250 if (alarm->id == 0) {
251 egg_idletime_alarm_reset_all (idletime);
252 goto out;
253 }
254
255 /* emit */
256 g_signal_emit (alarm->idletime, signals [SIGNAL_ALARM_EXPIRED], 0, alarm->id);
257
258 /* we need the first alarm to go off to set the reset alarm */
259 egg_idletime_set_reset_alarm (idletime, alarm_event);
260out:
261 /* don't propagate */
262 return GDK_FILTER_REMOVE;
263}

324static void
325egg_idletime_wayland_notification_idled_cb (void *data,
326 struct ext_idle_notification_v1 *notification)
327{
328 EggIdletime *idletime = EGG_IDLETIME (data);
329 EggIdletimeWaylandNotification *wn;
330 guint i;
331
332 for (i = 0; i < idletime->priv->wayland_notifications->len; i++) {
333 wn = g_ptr_array_index (idletime->priv->wayland_notifications, i);
334 if (wn->notification == notification) {
335 g_signal_emit (idletime, signals [SIGNAL_ALARM_EXPIRED], 0, wn->id);
336 break;
337 }
338 }
339 idletime->priv->wayland_is_idle = TRUE;
340 if (idletime->priv->idle_timer)
341 g_timer_start (idletime->priv->idle_timer);
342}

348egg_idletime_wayland_notification_resumed_cb (void *data,

396gint64
397egg_idletime_get_time (EggIdletime *idletime)
398{
399 g_return_val_if_fail (EGG_IS_IDLETIME (idletime), 0);
400
401#ifdef HAVE_X11
402 if (gdk_display_get_default () != NULL &&
403 GDK_IS_X11_DISPLAY (gdk_display_get_default ()) &&
404 idletime->priv->idle_counter) {
405 XSyncValue value;
406 XSyncQueryCounter (idletime->priv->dpy, idletime->priv->idle_counter, &value);
407 return egg_idletime_xsyncvalue_to_int64 (value);
408 }
409#endif
410#ifdef HAVE_WAYLAND
411 if (gdk_display_get_default () != NULL &&
412 GDK_IS_WAYLAND_DISPLAY (gdk_display_get_default ()) &&
413 idletime->priv->wayland_is_idle &&
414 idletime->priv->idle_timer) {
415 return (gint64)(g_timer_elapsed (idletime->priv->idle_timer, NULL) * 1000.0);
416 }
417#endif
418 return 0;
419}

461static void
462egg_idletime_init (EggIdletime *idletime)
463{
464 idletime->priv = egg_idletime_get_instance_private (idletime);
465
466 idletime->priv->reset_set = FALSE;
467 idletime->priv->array = g_ptr_array_new ();
468
469#ifdef HAVE_WAYLAND
470 idletime->priv->wayland_notifications = NULL;
471 idletime->priv->wayland_is_idle = FALSE;
472 idletime->priv->idle_timer = NULL;
473#endif
474
475#ifdef HAVE_X11
476 if (gdk_display_get_default () != NULL &&
477 GDK_IS_X11_DISPLAY (gdk_display_get_default ())) {
478 int sync_error;
479 int ncounters;
480 XSyncSystemCounter *counters;
481 EggIdletimeAlarm *alarm;
482 gint i;
483
484 idletime->priv->idle_counter = None;
485 idletime->priv->sync_event = 0;
486 idletime->priv->dpy = GDK_DISPLAY_XDISPLAY (gdk_display_get_default());
487
488 /* get the sync event */
489 if (!XSyncQueryExtension (idletime->priv->dpy, &idletime->priv->sync_event, &sync_error)) {
490 g_warning ("No Sync extension.");
491 return;
492 }
493
494 /* gtk_init should do XSyncInitialize for us */
495 counters = XSyncListSystemCounters (idletime->priv->dpy, &ncounters);
496 for (i=0; i < ncounters && !idletime->priv->idle_counter; i++) {
497 if (strcmp(counters[i].name, "IDLETIME") == 0)
498 idletime->priv->idle_counter = counters[i].counter;
499 }
500 XSyncFreeSystemCounterList (counters);
501
502 /* arh. we don't have IDLETIME support */
503 if (!idletime->priv->idle_counter) {
504 g_warning ("No idle counter.");
505 return;
506 }
507
508 /* catch the timer alarm */
509 gdk_window_add_filter (NULL, egg_idletime_event_filter_cb, idletime);
510
511 /* create a reset alarm */
512 alarm = egg_idletime_alarm_new (idletime, 0);
513 g_ptr_array_add (idletime->priv->array, alarm);
514 return;
515 }
516#endif
517#ifdef HAVE_WAYLAND
518 if (gdk_display_get_default () != NULL &&
519 GDK_IS_WAYLAND_DISPLAY (gdk_display_get_default ())) {
520 struct wl_display *display;
521 struct wl_registry *registry;
522
523 idletime->priv->wayland_notifications = g_ptr_array_new_with_free_func (g_free);
524 idletime->priv->idle_timer = g_timer_new ();
525
526 display = gdk_wayland_display_get_wl_display (gdk_display_get_default ());
527 if (display == NULL) {
528 g_warning ("Cannot get Wayland display");
529 return;
530 }
531
532 registry = wl_display_get_registry (display);
533 if (registry == NULL) {
534 g_warning ("Cannot get Wayland registry");
535 return;
536 }
537
538 wl_registry_add_listener (registry, &egg_idletime_wayland_registry_listener, idletime);
539 wl_display_roundtrip (display);
540
541 if (idletime->priv->idle_notifier == NULL) {
542 g_debug ("Compositor does not support ext_idle_notifier_v1");
543 }
544 if (idletime->priv->seat == NULL) {
545 g_warning ("No wl_seat available");
546 }
547 return;
548 }
549#endif
550 g_debug ("No idle tracking available for current display");
551}

556gboolean
557egg_idletime_alarm_set (EggIdletime *idletime, guint id, guint timeout)
558{
559 g_return_val_if_fail (EGG_IS_IDLETIME (idletime), FALSE);
560 g_return_val_if_fail (id != 0, FALSE);
561 g_return_val_if_fail (timeout != 0, FALSE);
562
563#ifdef HAVE_X11
564 if (gdk_display_get_default () != NULL &&
565 GDK_IS_X11_DISPLAY (gdk_display_get_default ())) {
566 EggIdletimeAlarm *alarm;
567
568 /* see if we already created an alarm with this ID */
569 alarm = egg_idletime_alarm_find_id (idletime, id);
570 if (alarm == NULL) {
571 /* create a new alarm */
572 alarm = egg_idletime_alarm_new (idletime, id);
573
574 /* add to array */
575 g_ptr_array_add (idletime->priv->array, alarm);
576 }
577
578 /* set the timeout */
579 XSyncIntToValue (&alarm->timeout, (gint)timeout);
580
581 /* set, and start the timer */
582 egg_idletime_xsync_alarm_set (idletime, alarm, EGG_IDLETIME_ALARM_TYPE_POSITIVE);
583 return TRUE;
584 }
585#endif
586#ifdef HAVE_WAYLAND
587 if (gdk_display_get_default () != NULL &&
588 GDK_IS_WAYLAND_DISPLAY (gdk_display_get_default ())) {
589 EggIdletimeWaylandNotification *wn;
590
591 if (idletime->priv->idle_notifier == NULL) {
592 g_warning ("EggIdletime: idle_notifier not available");
593 return FALSE;
594 }
595 if (idletime->priv->seat == NULL) {
596 g_warning ("EggIdletime: no Wayland seat available");
597 return FALSE;
598 }
599
600 /* see if we already created a notification with this ID */
601 wn = egg_idletime_alarm_wayland_find_id (idletime, id);
602 if (wn != NULL) {
603 if (wn->notification)
604 ext_idle_notification_v1_destroy (wn->notification);
605 wn->timeout = timeout;
606 } else {
607 wn = g_new0 (EggIdletimeWaylandNotification, 1);
608 wn->id = id;
609 wn->timeout = timeout;
610 g_ptr_array_add (idletime->priv->wayland_notifications, wn);
611 }
612
613 wn->notification = ext_idle_notifier_v1_get_input_idle_notification (
614 idletime->priv->idle_notifier, timeout, idletime->priv->seat);
615 ext_idle_notification_v1_add_listener (wn->notification,
616 &egg_idletime_wayland_notification_listener, idletime);
617
618 return TRUE;
619 }
620#endif
621 return FALSE;
622}

627gboolean
628egg_idletime_alarm_remove (EggIdletime *idletime, guint id)
629{
630 g_return_val_if_fail (EGG_IS_IDLETIME (idletime), FALSE);
631
632#ifdef HAVE_X11
633 if (gdk_display_get_default () != NULL &&
634 GDK_IS_X11_DISPLAY (gdk_display_get_default ())) {
635 EggIdletimeAlarm *alarm;
636
637 alarm = egg_idletime_alarm_find_id (idletime, id);
638 if (alarm == NULL)
639 return FALSE;
640 return egg_idletime_alarm_free (idletime, alarm);
641 }
642#endif
643#ifdef HAVE_WAYLAND
644 if (gdk_display_get_default () != NULL &&
645 GDK_IS_WAYLAND_DISPLAY (gdk_display_get_default ())) {
646 EggIdletimeWaylandNotification *wn;
647
648 wn = egg_idletime_alarm_wayland_find_id (idletime, id);
649 if (wn == NULL)
650 return FALSE;
651
652 if (wn->notification)
653 ext_idle_notification_v1_destroy (wn->notification);
654 g_ptr_array_remove (idletime->priv->wayland_notifications, wn);
655 g_free (wn);
656 return TRUE;
657 }
658#endif
659 return FALSE;
660}

665void
666egg_idletime_alarm_reset_all (EggIdletime *idletime)
667{
668 g_return_if_fail (EGG_IS_IDLETIME (idletime));
669
670#ifdef HAVE_X11
671 if (gdk_display_get_default () != NULL &&
672 GDK_IS_X11_DISPLAY (gdk_display_get_default ())) {
673 guint i;
674 EggIdletimeAlarm *alarm;
675
676 if (!idletime->priv->reset_set)
677 return;
678
679 /* reset all the alarms (except the reset alarm) to their timeouts */
680 for (i=1; i<idletime->priv->array->len; i++) {
681 alarm = g_ptr_array_index (idletime->priv->array, i);
682 egg_idletime_xsync_alarm_set (idletime, alarm, EGG_IDLETIME_ALARM_TYPE_POSITIVE);
683 }
684
685 /* set the reset alarm to be disabled */
686 alarm = g_ptr_array_index (idletime->priv->array, 0);
687 egg_idletime_xsync_alarm_set (idletime, alarm, EGG_IDLETIME_ALARM_TYPE_DISABLED);
688
689 /* emit signal so say we've reset all timers */
690 g_signal_emit (idletime, signals [SIGNAL_RESET], 0);
691
692 /* we need to be reset again on the next event */
693 idletime->priv->reset_set = FALSE;
694 return;
695 }
696#endif
697#ifdef HAVE_WAYLAND
698 if (gdk_display_get_default () != NULL &&
699 GDK_IS_WAYLAND_DISPLAY (gdk_display_get_default ())) {
700 g_signal_emit (idletime, signals [SIGNAL_RESET], 0);
701 return;
702 }
703#endif
704}

709static void
710egg_idletime_finalize (GObject *object)
711{
712 guint i;
713 EggIdletime *idletime;
714
715 g_return_if_fail (object != NULL);
716 g_return_if_fail (EGG_IS_IDLETIME (object));
717
718 idletime = EGG_IDLETIME (object);
719 idletime->priv = egg_idletime_get_instance_private (idletime);
720
721#ifdef HAVE_X11
722 if (gdk_display_get_default () != NULL &&
723 GDK_IS_X11_DISPLAY (gdk_display_get_default ())) {
724 EggIdletimeAlarm *alarm;
725
726 /* free all counters, including reset counter */
727 for (i=0; i<idletime->priv->array->len; i++) {
