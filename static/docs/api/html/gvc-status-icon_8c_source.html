75static GSettings *
76new_applet_settings (void)
77{
78 GSettingsSchemaSource *source;
79 GSettingsSchema *schema;
80
81 source = g_settings_schema_source_get_default ();
82 if (source == NULL)
83 return NULL;
84
85 schema = g_settings_schema_source_lookup (source,
86 "org.mate.volume-control",
87 TRUE);
88 if (schema == NULL)
89 return NULL;
90
91 g_settings_schema_unref (schema);
92
93 return g_settings_new ("org.mate.volume-control");

107static gboolean
108is_recording_application_control (MateMixerStreamControl *control)
109{
110 MateMixerAppInfo *app_info;
111 const gchar *app_id;
112
113 if (mate_mixer_stream_control_get_role (control) != MATE_MIXER_STREAM_CONTROL_ROLE_APPLICATION)
114 return FALSE;
115
116 app_info = mate_mixer_stream_control_get_app_info (control);
117 if (app_info == NULL)
118 return TRUE;
119
120 app_id = mate_mixer_app_info_get_id (app_info);
121 if (app_id == NULL)
122 return TRUE;
123
124 return strcmp (app_id, "org.mate.VolumeControl") != 0 &&
125 strcmp (app_id, "org.gnome.VolumeControl") != 0 &&
126 strcmp (app_id, "org.PulseAudio.pavucontrol") != 0;
127}

129static void
130update_icon_input (GvcStatusIcon *status_icon)
131{
132 MateMixerStreamControl *control = NULL;
133 gboolean show = FALSE;
134
135 /* Enable the input icon when any application is recording, regardless of
136 * which input stream it uses */
137 if (status_icon->priv->input != NULL) {
138 const gchar *stream_name;
139 const GList *inputs = mate_mixer_stream_list_controls (status_icon->priv->input);
140
141 control = mate_mixer_stream_get_default_control (status_icon->priv->input);
142
143 stream_name = mate_mixer_stream_get_name (status_icon->priv->input);
144 if (stream_name != NULL && g_str_has_suffix (stream_name, ".monitor"))
145 inputs = NULL;
146
147 while (inputs != NULL) {
148 MateMixerStreamControl *input = MATE_MIXER_STREAM_CONTROL (inputs->data);
149
150 if (is_recording_application_control (input)) {
151 if (G_UNLIKELY (control == NULL))
152 control = input;
153 show = TRUE;
154 break;
155 }
156 inputs = inputs->next;
157 }
158 }
159
160 if (!show && status_icon->priv->context != NULL) {
161 const GList *streams;
162
163 streams = mate_mixer_context_list_streams (status_icon->priv->context);
164 while (streams != NULL && !show) {
165 MateMixerStream *stream = MATE_MIXER_STREAM (streams->data);
166 const gchar *name;
167 const GList *inputs;
168
169 if (stream == status_icon->priv->input ||
170 mate_mixer_stream_get_direction (stream) != MATE_MIXER_DIRECTION_INPUT) {
171 streams = streams->next;
172 continue;
173 }
174
175 name = mate_mixer_stream_get_name (stream);
176 if (name != NULL && g_str_has_suffix (name, ".monitor")) {
177 streams = streams->next;
178 continue;
179 }
180
181 inputs = mate_mixer_stream_list_controls (stream);
182 while (inputs != NULL) {
183 MateMixerStreamControl *input = MATE_MIXER_STREAM_CONTROL (inputs->data);
184
185 if (is_recording_application_control (input)) {
186 control = mate_mixer_stream_get_default_control (stream);
187 if (control == NULL)
188 control = input;
189 show = TRUE;
190 break;
191 }
192
193 inputs = inputs->next;
194 }
195
196 streams = streams->next;
197 }
198 }
199
200 gvc_stream_status_icon_set_control (status_icon->priv->icon_input, control);
201 gtk_status_icon_set_visible (GTK_STATUS_ICON (status_icon->priv->icon_input), show);
202}

289static void
290update_active_player (GvcStatusIcon *status_icon)
291{
292 GvcMprisPlayer *player;
293 gboolean show_player;
294
295 show_player = settings_get_boolean (status_icon, "show-player-controls", TRUE);
296
297 player = NULL;
298 if (status_icon->priv->selected_player != NULL &&
299 gvc_mpris_player_is_ready (status_icon->priv->selected_player))
300 player = status_icon->priv->selected_player;
301 if (player == NULL)
302 player = gvc_mpris_manager_get_best_player (status_icon->priv->mpris_manager);
303
304 if (player == status_icon->priv->active_player) {
305 if (player != NULL) {
306 gvc_player_widget_set_player (GVC_PLAYER_WIDGET (status_icon->priv->player_widget),
307 show_player ? player : NULL);
308 gvc_stream_status_icon_set_player_widget_visible (status_icon->priv->icon_output,
309 show_player);
310 }
311 update_player_selector (status_icon);
312 return;
313 }
314
315 g_clear_object (&status_icon->priv->active_player);
316 if (player != NULL)
317 status_icon->priv->active_player = g_object_ref (player);
318
319 gvc_player_widget_set_player (GVC_PLAYER_WIDGET (status_icon->priv->player_widget),
320 show_player ? status_icon->priv->active_player : NULL);
321 gvc_stream_status_icon_set_mpris_player (status_icon->priv->icon_output,
322 status_icon->priv->active_player);
323 gvc_stream_status_icon_set_player_widget_visible (status_icon->priv->icon_output,
324 status_icon->priv->active_player != NULL &&
325 show_player);
326 update_player_selector (status_icon);
327}

329static void
330remember_player (GvcStatusIcon *status_icon,
331 GvcMprisPlayer *player)
332{
333 const gchar *desktop_entry;
334 gchar **known;
335 gboolean found = FALSE;
336 gint i;
337
338 if (status_icon->priv->applet_settings == NULL)
339 return;
340
341 desktop_entry = gvc_mpris_player_get_desktop_entry (player);
342 if (desktop_entry == NULL || desktop_entry[0] == '\0')
343 return;
344
345 known = g_settings_get_strv (status_icon->priv->applet_settings, "known-players");
346 for (i = 0; known != NULL && known[i] != NULL; i++) {
347 if (g_strcmp0 (known[i], desktop_entry) == 0) {
348 found = TRUE;
349 break;
350 }
351 }
352
353 if (!found) {
354 GPtrArray *array;
355
356 array = g_ptr_array_new_with_free_func (g_free);
357 for (i = 0; known != NULL && known[i] != NULL; i++)
358 g_ptr_array_add (array, g_strdup (known[i]));
359 g_ptr_array_add (array, g_strdup (desktop_entry));
360 g_ptr_array_add (array, NULL);
361 g_settings_set_strv (status_icon->priv->applet_settings,
362 "known-players",
363 (const gchar * const *) array->pdata);
364 g_ptr_array_free (array, TRUE);
365 }
366
367 g_strfreev (known);
368}

429static void
430disconnect_input_stream_control_signals (GvcStatusIcon *status_icon)
431{
432 const GList *streams;
433
434 streams = mate_mixer_context_list_streams (status_icon->priv->context);
435 while (streams != NULL) {
436 MateMixerStream *stream = MATE_MIXER_STREAM (streams->data);
437
438 if (mate_mixer_stream_get_direction (stream) == MATE_MIXER_DIRECTION_INPUT) {
439 g_signal_handlers_disconnect_by_func (G_OBJECT (stream),
440 G_CALLBACK (on_input_stream_control_added),
441 status_icon);
442 g_signal_handlers_disconnect_by_func (G_OBJECT (stream),
443 G_CALLBACK (on_input_stream_control_removed),
444 status_icon);
445 }
446
447 streams = streams->next;
448 }
449}

451static void
452connect_input_stream_control_signals (GvcStatusIcon *status_icon)
453{
454 const GList *streams;
455
456 disconnect_input_stream_control_signals (status_icon);
457
458 streams = mate_mixer_context_list_streams (status_icon->priv->context);
459 while (streams != NULL) {
460 MateMixerStream *stream = MATE_MIXER_STREAM (streams->data);
461
462 if (mate_mixer_stream_get_direction (stream) == MATE_MIXER_DIRECTION_INPUT) {
463 g_signal_connect (G_OBJECT (stream),
464 "control-added",
465 G_CALLBACK (on_input_stream_control_added),
466 status_icon);
467 g_signal_connect (G_OBJECT (stream),
468 "control-removed",
469 G_CALLBACK (on_input_stream_control_removed),
470 status_icon);
471 }
472
473 streams = streams->next;
474 }
475}

477static gboolean
478update_default_output_stream (GvcStatusIcon *status_icon)
479{
480 MateMixerStream *stream;
481
482 stream = mate_mixer_context_get_default_output_stream (status_icon->priv->context);
483 if (stream == status_icon->priv->output)
484 return FALSE;
485
486 if (status_icon->priv->output != NULL) {
487 g_signal_handlers_disconnect_by_data (G_OBJECT (status_icon->priv->output), status_icon);
488 g_object_unref (status_icon->priv->output);
489 }
490
491 status_icon->priv->output = (stream == NULL) ? NULL : g_object_ref (stream);
492 if (status_icon->priv->output != NULL) {
493 g_signal_connect (G_OBJECT (status_icon->priv->output),
494 "control-added",
495 G_CALLBACK (on_output_stream_control_added),
496 status_icon);
497 g_signal_connect (G_OBJECT (status_icon->priv->output),
498 "control-removed",
499 G_CALLBACK (on_output_stream_control_removed),
500 status_icon);
501 }
502
503 return TRUE;
504}

524static void
525on_context_state_notify (MateMixerContext *context,
526 GParamSpec *pspec,
527 GvcStatusIcon *status_icon)
528{
529 MateMixerState state = mate_mixer_context_get_state (context);
530
531 switch (state) {
532 case MATE_MIXER_STATE_FAILED:
533 g_warning ("Failed to connect to a sound system");
534 break;
535
536 case MATE_MIXER_STATE_READY:
537 update_default_output_stream (status_icon);
538 update_default_input_stream (status_icon);
539 connect_input_stream_control_signals (status_icon);
540
541 update_icon_output (status_icon);
542 update_icon_input (status_icon);
543 break;
544 default:
545 break;
546 }
547}

594void
595gvc_status_icon_start (GvcStatusIcon *status_icon)
596{
597 g_return_if_fail (GVC_IS_STATUS_ICON (status_icon));
598
599 if (G_UNLIKELY (status_icon->priv->running == TRUE))
600 return;
601
602 if (G_UNLIKELY (mate_mixer_context_open (status_icon->priv->context) == FALSE))
603 g_warning ("Failed to connect to a sound system");
604
605 g_debug ("StatusIcon has been started");
606
607 status_icon->priv->running = TRUE;
608}

610static void
611gvc_status_icon_dispose (GObject *object)
612{
613 GvcStatusIcon *status_icon = GVC_STATUS_ICON (object);
614
615 if (status_icon->priv->input != NULL) {
616 g_signal_handlers_disconnect_by_data (G_OBJECT (status_icon->priv->input),
617 status_icon);
618 g_clear_object (&status_icon->priv->input);
619 }
620 if (status_icon->priv->output != NULL) {
621 g_signal_handlers_disconnect_by_data (G_OBJECT (status_icon->priv->output),
622 status_icon);
623 g_clear_object (&status_icon->priv->output);
624 }
625 if (status_icon->priv->context != NULL)
626 disconnect_input_stream_control_signals (status_icon);
627
628 if (status_icon->priv->mpris_manager != NULL) {
629 GList *players = gvc_mpris_manager_get_players (status_icon->priv->mpris_manager);
630 GList *l;
631
632 for (l = players; l != NULL; l = l->next)
633 g_signal_handlers_disconnect_by_data (l->data, status_icon);
634 g_list_free (players);
635 g_signal_handlers_disconnect_by_data (status_icon->priv->mpris_manager, status_icon);
636 }
637 if (status_icon->priv->applet_settings != NULL)
638 g_signal_handlers_disconnect_by_data (status_icon->priv->applet_settings, status_icon);
639 if (status_icon->priv->context != NULL)
640 g_signal_handlers_disconnect_by_data (status_icon->priv->context, status_icon);
641
642 g_clear_object (&status_icon->priv->context);
643 g_clear_object (&status_icon->priv->active_player);
644 g_clear_object (&status_icon->priv->selected_player);
645 g_clear_object (&status_icon->priv->mpris_manager);
646 g_clear_object (&status_icon->priv->applet_settings);
647 g_clear_object (&status_icon->priv->icon_input);
648 g_clear_object (&status_icon->priv->icon_output);
649
650 G_OBJECT_CLASS (gvc_status_icon_parent_class)->dispose (object);
651}

661static void
662gvc_status_icon_init (GvcStatusIcon *status_icon)
663{
664 status_icon->priv = gvc_status_icon_get_instance_private (status_icon);
665
666 status_icon->priv->icon_input = gvc_stream_status_icon_new (NULL, icon_names_input);
667 status_icon->priv->icon_output = gvc_stream_status_icon_new (NULL, icon_names_output);
668
669 status_icon->priv->applet_settings = new_applet_settings ();
670 gvc_stream_status_icon_set_applet_settings (status_icon->priv->icon_input,
671 status_icon->priv->applet_settings);
672 gvc_stream_status_icon_set_applet_settings (status_icon->priv->icon_output,
673 status_icon->priv->applet_settings);
674
675 status_icon->priv->player_widget = gvc_player_widget_new ();
676 gvc_stream_status_icon_set_player_widget (status_icon->priv->icon_output,
677 status_icon->priv->player_widget);
678 g_signal_connect (status_icon->priv->player_widget,
679 "player-selected",
680 G_CALLBACK (on_player_widget_player_selected),
681 status_icon);
682 g_signal_connect (status_icon->priv->icon_output,
683 "player-selected",
684 G_CALLBACK (on_status_icon_player_selected),
685 status_icon);
686 g_signal_connect (status_icon->priv->icon_input,
687 "player-selected",
688 G_CALLBACK (on_status_icon_player_selected),
689 status_icon);
690
691 gvc_stream_status_icon_set_display_name (status_icon->priv->icon_input, _("Input"));
692 gvc_stream_status_icon_set_display_name (status_icon->priv->icon_output, _("Output"));
693
