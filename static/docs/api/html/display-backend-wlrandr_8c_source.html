74static GArray *
75parse_heads (WlrandrBackend *self, GVariant *configuration)
76{
77 GVariantIter iter;
78 GArray *heads = g_array_new (FALSE, FALSE, sizeof (OutputHead));
79
80 g_variant_iter_init (&iter, configuration);
81
82 while (TRUE)
83 {
84 GVariant *head_var;
85 const char *name;
86 const char *description;
87 gboolean enabled;
88 gboolean adaptive_sync_supported;
89 gboolean adaptive_sync;
90 int x, y, transform;
91 double scale;
92 int cur_w, cur_h, cur_rate, cur_pref;
93 g_autoptr (GVariant) modes_var = NULL;
94 OutputHead head;
95 GVariantIter modes_iter;
96
97 head_var = g_variant_iter_next_value (&iter);
98 if (head_var == NULL)
99 break;
100
101 g_variant_get_child (head_var, HEAD_INFO_FIELD_NAME, "&s", &name);
102 g_variant_get_child (head_var, HEAD_INFO_FIELD_DESCRIPTION, "&s", &description);
103 g_variant_get_child (head_var, HEAD_INFO_FIELD_ENABLED, "b", &enabled);
104 g_variant_get_child (head_var, HEAD_INFO_FIELD_X, "i", &x);
105 g_variant_get_child (head_var, HEAD_INFO_FIELD_Y, "i", &y);
106 g_variant_get_child (head_var, HEAD_INFO_FIELD_TRANSFORM, "i", &transform);
107 g_variant_get_child (head_var, HEAD_INFO_FIELD_SCALE, "d", &scale);
108 g_variant_get_child (head_var, HEAD_INFO_FIELD_ADAPTIVE_SYNC_SUPPORTED,
109 "b", &adaptive_sync_supported);
110 g_variant_get_child (head_var, HEAD_INFO_FIELD_ADAPTIVE_SYNC, "b", &adaptive_sync);
111 g_variant_get_child (head_var, HEAD_INFO_FIELD_CURRENT_MODE,
112 MODE_FORMAT, &cur_w, &cur_h, &cur_rate, &cur_pref);
113 g_variant_get_child (head_var, HEAD_INFO_FIELD_MODES, "@a" MODE_FORMAT, &modes_var);
114
115 memset (&head, 0, sizeof (OutputHead));
116 head.name = g_strdup (name);
117 head.description = g_strdup (description);
118 head.enabled = enabled;
119 head.x = x;
120 head.y = y;
121 head.transform = transform & 0x3;
122 head.scale = scale;
123 head.adaptive_sync_supported = adaptive_sync_supported ? 1 : 0;
124 head.adaptive_sync = adaptive_sync;
125 head.width = cur_w;
126 head.height = cur_h;
127 head.rate = cur_rate;
128 head.preferred = cur_pref;
129 head.modes = g_array_new (FALSE, FALSE, sizeof (Mode));
130
131 if (has_primary (self))
132 g_variant_get_child (head_var, HEAD_INFO_FIELD_PRIMARY, "b", &head.primary);
133
134 if (cur_w > 0 && cur_h > 0)
135 {
136 Mode m;
137
138 m.width = cur_w;
139 m.height = cur_h;
140 m.rate = cur_rate;
141 m.preferred = cur_pref;
142 g_array_append_val (head.modes, m);
143 }
144
145 g_variant_iter_init (&modes_iter, modes_var);
146 while (TRUE)
147 {
148 int w, h, rate;
149 int pref;
150 Mode mode;
151
152 if (!g_variant_iter_next (&modes_iter, MODE_FORMAT, &w, &h, &rate, &pref))
153 break;
154
155 if (display_head_has_mode (&head, w, h, rate))
156 continue;
157
158 mode.width = w;
159 mode.height = h;
160 mode.rate = rate;
161 mode.preferred = pref;
162 g_array_append_val (head.modes, mode);
163 }
164
165 g_variant_unref (head_var);
166
167 g_array_append_val (heads, head);
168 }
169
170 return heads;
171}

173static void
174on_configuration_received (GObject *source_object, GAsyncResult *res, gpointer data)
175{
176 WlrandrBackend *self = data;
177 GError *error = NULL;
178 g_autoptr (GVariant) result = NULL;
179 g_autoptr (GVariant) configuration = NULL;
180
181 result = g_dbus_proxy_call_finish (G_DBUS_PROXY (source_object), res, &error);
182 if (result == NULL)
183 {
184 if (!g_error_matches (error, G_IO_ERROR, G_IO_ERROR_CANCELLED))
185 self->base.changed (NULL, error->message, self->base.changed_data);
186 g_error_free (error);
187 return;
188 }
189
190 configuration = g_variant_get_child_value (result, 0);
191 self->base.changed (parse_heads (self, configuration), NULL, self->base.changed_data);
192}

203static void
204on_set_configuration_returned (GObject *source_object, GAsyncResult *res, gpointer data)
205{
206 ApplyCall *call = data;

208 g_autoptr (GVariant) result = NULL;

211 if (result == NULL && g_error_matches (error, G_IO_ERROR, G_IO_ERROR_CANCELLED))

218static void
219wlrandr_apply (DisplayBackend *backend,
220 GArray *heads,
221 gboolean temporary,
222 DisplayBackendAppliedFunc done,
223 gpointer data)
224{
225 WlrandrBackend *self = (WlrandrBackend *) backend;
226 const char *format = has_primary (self) ? HEAD_SET_FORMAT_3 : HEAD_SET_FORMAT;
227 g_autofree char *type = g_strconcat ("a", format, NULL);
228 gboolean primary_set = FALSE;
229 GVariantBuilder builder;
230 ApplyCall *call;
231 guint i;
232
233 g_variant_builder_init (&builder, G_VARIANT_TYPE (type));
234
235 for (i = 0; i < heads->len; i++)
236 {
237 OutputHead *head = &g_array_index (heads, OutputHead, i);
238 int width = 0, height = 0;
239
240 if (head->enabled)
241 display_head_get_size (head, &width, &height);
242
243 if (has_primary (self))
244 {
245 gboolean primary = head->primary && head->enabled && !primary_set;
246
247 primary_set = primary_set || primary;
248 g_variant_builder_add (&builder, format,
249 head->name, head->enabled, width, height, head->rate,
250 head->x, head->y, head->scale, head->transform,
251 head->adaptive_sync, primary);
252 }
253 else
254 {
255 g_variant_builder_add (&builder, format,
256 head->name, head->enabled, width, height, head->rate,
257 head->x, head->y, head->scale, head->transform,
258 head->adaptive_sync);
259 }
260 }
261
262 call = g_new0 (ApplyCall, 1);
263 call->done = done;
264 call->data = data;
265 g_dbus_proxy_call (self->proxy, "SetConfiguration",
266 g_variant_new_tuple ((GVariant *[]) { g_variant_builder_end (&builder) }, 1),
267 G_DBUS_CALL_FLAGS_NONE, -1, NULL, on_set_configuration_returned, call);
268}

294static gboolean
295serves_interface (GDBusConnection *connection, const char *interface)
296{
297 g_autoptr (GVariant) result = NULL;
298 g_autoptr (GDBusNodeInfo) node = NULL;
299 const char *xml;
300
301 result = g_dbus_connection_call_sync (connection, "org.mate.SettingsDaemon", MSD_WLRANDR_DBUS_PATH,
302 "org.freedesktop.DBus.Introspectable", "Introspect", NULL,
303 G_VARIANT_TYPE ("(s)"), G_DBUS_CALL_FLAGS_NONE, -1, NULL, NULL);
304 if (result == NULL)
305 return FALSE;
306
307 g_variant_get (result, "(&s)", &xml);
308 node = g_dbus_node_info_new_for_xml (xml, NULL);
309 return node != NULL && g_dbus_node_info_lookup_interface (node, interface) != NULL;
310}

312DisplayBackend *
313display_backend_wlrandr_new (DisplayBackendChangedFunc changed, gpointer data, GError **error)
314{
315 WlrandrBackend *self;
316 GDBusConnection *connection;
317 GDBusProxy *proxy;
318 gboolean primary;
319 const char *interface;
320
321 connection = g_bus_get_sync (G_BUS_TYPE_SESSION, NULL, error);
322 if (connection == NULL)
323 return NULL;
324
325 primary = serves_interface (connection, MSD_WLRANDR_DBUS_NAME_3);
326 interface = primary ? MSD_WLRANDR_DBUS_NAME_3 : MSD_WLRANDR_DBUS_NAME;
327 proxy = g_dbus_proxy_new_sync (connection, G_DBUS_PROXY_FLAGS_NONE, NULL,
328 "org.mate.SettingsDaemon", MSD_WLRANDR_DBUS_PATH,
329 interface, NULL, error);
330 if (proxy == NULL)
331 {
332 g_object_unref (connection);
333 return NULL;
334 }
335
336 self = g_new0 (WlrandrBackend, 1);
337 self->base.name = "mate-settings-daemon";
338 self->base.features = primary ? DISPLAY_BACKEND_PRIMARY : 0;
339 self->base.changed = changed;
340 self->base.changed_data = data;
341 self->base.refresh = wlrandr_refresh;
342 self->base.apply = wlrandr_apply;
343 self->base.free = wlrandr_free;
344 self->connection = connection;
345 self->proxy = proxy;
346 self->signal_id = g_dbus_connection_signal_subscribe (connection, NULL, interface,
347 "ConfigurationChanged",
348 MSD_WLRANDR_DBUS_PATH, NULL,
349 G_DBUS_SIGNAL_FLAGS_NONE,
350 on_configuration_changed, self, NULL);
351 return &self->base;
352}
