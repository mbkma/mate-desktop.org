57char *
58caja_global_preferences_get_default_folder_viewer_preference_as_iid (void)
59{
60 int preference_value;
61 const char *viewer_iid;
62
63 preference_value =
64 g_settings_get_enum (caja_preferences, CAJA_PREFERENCES_DEFAULT_FOLDER_VIEWER);
65
66 if (preference_value == CAJA_DEFAULT_FOLDER_VIEWER_LIST_VIEW)
67 {
68 viewer_iid = CAJA_LIST_VIEW_IID;
69 }
70 else if (preference_value == CAJA_DEFAULT_FOLDER_VIEWER_COMPACT_VIEW)
71 {
72 viewer_iid = CAJA_COMPACT_VIEW_IID;
73 }
74 else
75 {
76 viewer_iid = CAJA_ICON_VIEW_IID;
77 }
78
79 return g_strdup (viewer_iid);
80}

108GSettings *
109caja_preferences_get_monitor_settings (GdkMonitor *monitor)
110{
111 static GHashTable *cache = NULL;
112 GSettings *settings;
113 gchar *id;
114
115 g_return_val_if_fail (GDK_IS_MONITOR (monitor), NULL);
116
117 if (cache == NULL)
118 {
119 cache = g_hash_table_new_full (g_str_hash, g_str_equal, g_free, g_object_unref);
120 }
121
122 id = mate_bg_get_monitor_id (monitor);
123 settings = g_hash_table_lookup (cache, id);
124 if (settings == NULL)
125 {
126 settings = mate_bg_get_monitor_settings (monitor);
127 if (settings != NULL)
128 {
129 g_hash_table_insert (cache, g_strdup (id), settings);
130 }
131 }
132 g_free (id);
133
134 return settings;
135}
