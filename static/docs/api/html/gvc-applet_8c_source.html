101static void
102gvc_applet_update_mute_action (GvcApplet *applet)
103{
104 GtkAction *action;
105 gboolean is_muted;
106
107 if (applet->priv->action_group == NULL)
108 return;
109
110 action = gtk_action_group_get_action (applet->priv->action_group, "MuteOutput");
111 if (applet->priv->output == NULL ||
112 mate_mixer_stream_get_default_control (applet->priv->output) == NULL) {
113 gtk_action_set_sensitive (action, FALSE);
114 return;
115 }
116
117 gtk_action_set_sensitive (action, TRUE);
118 is_muted = gvc_stream_applet_icon_get_mute (applet->priv->icon_output);
119 gtk_action_set_label (action, is_muted ? _("Unmute Output") : _("Mute Output"));
120 gtk_action_set_icon_name (action, is_muted ? "audio-volume-medium" : "audio-volume-muted");
121}

155static gboolean
156is_recording_application_control (MateMixerStreamControl *control)
157{
158 MateMixerAppInfo *app_info;
159 const gchar *app_id;
160
161 if (mate_mixer_stream_control_get_role (control) != MATE_MIXER_STREAM_CONTROL_ROLE_APPLICATION)
162 return FALSE;
163
164 app_info = mate_mixer_stream_control_get_app_info (control);
165 if (app_info == NULL)
166 return TRUE;
167
168 app_id = mate_mixer_app_info_get_id (app_info);
169 if (app_id == NULL)
170 return TRUE;
171
172 return strcmp (app_id, "org.mate.VolumeControl") != 0 &&
173 strcmp (app_id, "org.gnome.VolumeControl") != 0 &&
174 strcmp (app_id, "org.PulseAudio.pavucontrol") != 0;
175}

177static void
178update_icon_input (GvcApplet *applet)
179{
180 MateMixerStreamControl *control = NULL;
181 gboolean show = FALSE;
182
183 /* Enable the input icon in case there is an input stream present and there
184 * is a non-mixer application using the input */
185 if (applet->priv->input != NULL) {
186 const gchar *app_id = NULL;
187 const gchar *stream_name;
188 const GList *inputs = mate_mixer_stream_list_controls (applet->priv->input);
189
190 control = mate_mixer_stream_get_default_control (applet->priv->input);
191
192 stream_name = mate_mixer_stream_get_name (applet->priv->input);
193 g_debug ("Got stream name %s", stream_name);
194 if (g_str_has_suffix (stream_name, ".monitor")) {
195 inputs = NULL;
196 g_debug ("Stream is a monitor, ignoring");
197 }
198
199 while (inputs != NULL) {
200 MateMixerStreamControl *input = MATE_MIXER_STREAM_CONTROL (inputs->data);
201
202 if (is_recording_application_control (input)) {
203 MateMixerAppInfo *app_info;
204
205 app_info = mate_mixer_stream_control_get_app_info (input);
206 app_id = app_info != NULL ? mate_mixer_app_info_get_id (app_info) : NULL;
207 if (app_id != NULL)
208 g_debug ("Found a recording application %s", app_id);
209 else
210 g_debug ("Found a recording application control %s",
211 mate_mixer_stream_control_get_label (input));
212
213 if (G_UNLIKELY (control == NULL))
214 control = input;
215
216 show = TRUE;
217 break;
218 }
219 inputs = inputs->next;
220 }
221 }
222
223 if (!show && applet->priv->context != NULL) {
224 const GList *streams;
225
226 streams = mate_mixer_context_list_streams (applet->priv->context);
227 while (streams != NULL && !show) {
228 MateMixerStream *stream = MATE_MIXER_STREAM (streams->data);
229 const gchar *name;
230 const GList *inputs;
231
232 if (stream == applet->priv->input ||
233 mate_mixer_stream_get_direction (stream) != MATE_MIXER_DIRECTION_INPUT) {
234 streams = streams->next;
235 continue;
236 }
237
238 name = mate_mixer_stream_get_name (stream);
239 if (name != NULL && g_str_has_suffix (name, ".monitor")) {
240 streams = streams->next;
241 continue;
242 }
243
244 inputs = mate_mixer_stream_list_controls (stream);
245 while (inputs != NULL) {
246 MateMixerStreamControl *input = MATE_MIXER_STREAM_CONTROL (inputs->data);
247
248 if (is_recording_application_control (input)) {
249 control = mate_mixer_stream_get_default_control (stream);
250 if (control == NULL)
251 control = input;
252 show = TRUE;
253 break;
254 }
255
256 inputs = inputs->next;
257 }
258
259 streams = streams->next;
260 }
261 }
262
263 if (show == TRUE)
264 g_debug ("Input icon enabled");
265 else
266 g_debug ("There is no recording application, input icon disabled");
267
268 gvc_stream_applet_icon_set_control (applet->priv->icon_input, control);
269
270 gtk_widget_set_visible (GTK_WIDGET (applet->priv->icon_input), show);
271}

295static void
296update_track_label (GvcApplet *applet)
297{
298 MatePanelAppletOrient orient;
299 const gchar *title;
300 const gchar *artist;
301 gchar *text = NULL;
302
303 if (applet->priv->track_label == NULL)
304 return;
305
306 if (!settings_get_boolean (applet, "show-track-in-panel", FALSE) ||
307 applet->priv->active_player == NULL ||
308 applet->priv->applet == NULL) {
309 gtk_widget_hide (GTK_WIDGET (applet->priv->track_label));
310 return;
311 }
312
313 orient = mate_panel_applet_get_orient (applet->priv->applet);
314 if (orient == MATE_PANEL_APPLET_ORIENT_LEFT ||
315 orient == MATE_PANEL_APPLET_ORIENT_RIGHT) {
316 gtk_widget_hide (GTK_WIDGET (applet->priv->track_label));
317 return;
318 }
319
320 title = gvc_mpris_player_get_title (applet->priv->active_player);
321 artist = gvc_mpris_player_get_artist (applet->priv->active_player);
322
323 if (title != NULL && title[0] != '\0' &&
324 artist != NULL && artist[0] != '\0')
325 text = g_strdup_printf ("%s - %s", title, artist);
326 else if (title != NULL && title[0] != '\0')
327 text = g_strdup (title);
328 else if (artist != NULL && artist[0] != '\0')
329 text = g_strdup (artist);
330
331 if (text == NULL) {
332 gtk_widget_hide (GTK_WIDGET (applet->priv->track_label));
333 return;
334 }
335
336 gtk_label_set_text (applet->priv->track_label, text);
337 gtk_widget_set_tooltip_text (GTK_WIDGET (applet->priv->track_label), text);
338 gtk_widget_show (GTK_WIDGET (applet->priv->track_label));
339
340 g_free (text);
341}

374static void
375update_active_player (GvcApplet *applet)
376{
377 GvcMprisPlayer *player;
378
379 player = NULL;
380 if (applet->priv->selected_player != NULL &&
381 gvc_mpris_player_is_ready (applet->priv->selected_player))
382 player = applet->priv->selected_player;
383 if (player == NULL)
384 player = gvc_mpris_manager_get_best_player (applet->priv->mpris_manager);
385
386 if (player == applet->priv->active_player) {
387 if (player != NULL) {
388 if (settings_get_boolean (applet, "show-player-controls", TRUE))
389 gvc_player_widget_set_player (GVC_PLAYER_WIDGET (applet->priv->player_widget), player);
390 else
391 gvc_player_widget_set_player (GVC_PLAYER_WIDGET (applet->priv->player_widget), NULL);
392 gvc_stream_applet_icon_set_player_widget_visible (applet->priv->icon_output,
393 settings_get_boolean (applet, "show-player-controls", TRUE));
394 }
395 update_player_selector (applet);
396 update_track_label (applet);
397 return;
398 }
399
400 g_clear_object (&applet->priv->active_player);
401 if (player != NULL)
402 applet->priv->active_player = g_object_ref (player);
403
404 if (settings_get_boolean (applet, "show-player-controls", TRUE))
405 gvc_player_widget_set_player (GVC_PLAYER_WIDGET (applet->priv->player_widget),
406 applet->priv->active_player);
407 else
408 gvc_player_widget_set_player (GVC_PLAYER_WIDGET (applet->priv->player_widget), NULL);
409 gvc_stream_applet_icon_set_mpris_player (applet->priv->icon_output,
410 applet->priv->active_player);
411 gvc_stream_applet_icon_set_player_widget_visible (applet->priv->icon_output,
412 applet->priv->active_player != NULL &&
413 settings_get_boolean (applet, "show-player-controls", TRUE));
414 update_player_selector (applet);
415 update_track_label (applet);
416}

435static void
436on_stream_menu_item_toggled (GtkCheckMenuItem *item,
437 GvcApplet *applet)
438{
439 MateMixerStream *stream;
440 MateMixerDirection direction;
441
442 if (!gtk_check_menu_item_get_active (item))
443 return;
444

446 if (stream == NULL)
447 return;

451 mate_mixer_context_set_default_output_stream (applet->priv->context, stream);
452 else if (direction == MATE_MIXER_DIRECTION_INPUT)
453 mate_mixer_context_set_default_input_stream (applet->priv->context, stream);

456static void
457on_launch_player_menu_item_activate (GtkMenuItem *item,
458 GvcApplet *applet)
459{
460 const gchar *desktop_entry;
461 GDesktopAppInfo *app_info;
462 GError *error = NULL;
463
464 desktop_entry = g_object_get_data (G_OBJECT (item), "desktop-entry");
465 if (desktop_entry == NULL)
466 return;
467
468 app_info = g_desktop_app_info_new (desktop_entry);
469 if (app_info == NULL && !g_str_has_suffix (desktop_entry, ".desktop")) {
470 gchar *desktop_id;
471
472 desktop_id = g_strconcat (desktop_entry, ".desktop", NULL);
473 app_info = g_desktop_app_info_new (desktop_id);
474 g_free (desktop_id);
475 }
476
477 if (app_info == NULL)
478 return;
479
480 g_app_info_launch (G_APP_INFO (app_info), NULL, NULL, &error);
481 if (error != NULL)
482 g_error_free (error);
483
484 g_object_unref (app_info);
485}

487static void
488append_stream_menu_items (GvcApplet *applet,
489 GtkWidget *submenu,
490 MateMixerDirection direction)
491{
492 const GList *streams;
493 GSList *group = NULL;
494
495 streams = mate_mixer_context_list_streams (applet->priv->context);
496 while (streams != NULL) {
497 MateMixerStream *stream = MATE_MIXER_STREAM (streams->data);
498
499 if (mate_mixer_stream_get_direction (stream) == direction) {
500 GtkWidget *item;
501 MateMixerStream *active;
502 const gchar *label;
503
504 active = direction == MATE_MIXER_DIRECTION_OUTPUT ?
505 mate_mixer_context_get_default_output_stream (applet->priv->context) :
506 mate_mixer_context_get_default_input_stream (applet->priv->context);
507 label = mate_mixer_stream_get_label (stream);
508 if (label == NULL || label[0] == '\0')
509 label = mate_mixer_stream_get_name (stream);
510
511 item = gtk_radio_menu_item_new_with_label (group, label);
512 group = gtk_radio_menu_item_get_group (GTK_RADIO_MENU_ITEM (item));
513 gtk_check_menu_item_set_active (GTK_CHECK_MENU_ITEM (item), stream == active);
514 g_object_set_data_full (G_OBJECT (item), "gvc-stream", g_object_ref (stream), g_object_unref);
515 g_signal_connect_object (item, "toggled", G_CALLBACK (on_stream_menu_item_toggled), applet, 0);
516 gtk_menu_shell_append (GTK_MENU_SHELL (submenu), item);
517 }
518
519 streams = streams->next;
520 }
521}

523static GtkWidget *
524get_applet_submenu (GvcApplet *applet,
525 const gchar *action_name)
526{
527 GtkAction *action;
528 GSList *proxies;
529
530 action = gtk_action_group_get_action (applet->priv->action_group, action_name);
531 if (action == NULL)
532 return NULL;
533
534 for (proxies = gtk_action_get_proxies (action); proxies != NULL; proxies = proxies->next) {
535 GtkWidget *proxy = GTK_WIDGET (proxies->data);
536 GtkWidget *submenu;
537
538 if (!GTK_IS_MENU_ITEM (proxy))
539 continue;
540
541 submenu = gtk_menu_item_get_submenu (GTK_MENU_ITEM (proxy));
542 if (GTK_IS_MENU (submenu))
543 return submenu;
544 }
545
546 return NULL;
547}

561static void
562on_applet_menu_show (GtkWidget *menu,
563 GvcApplet *applet)
564{
565 GtkWidget *submenu;
566 GtkWidget *item;
567 GList *players;
568 GList *l;
569 GSList *player_group = NULL;
570
571 /* The panel owns and positions this menu. Refresh only our dynamic
572 * submenus when it is about to be shown, including from the Menu key. */
573 submenu = get_applet_submenu (applet, "MediaPlayers");
574 if (submenu != NULL) {
575 clear_applet_submenu (submenu);
576 players = gvc_mpris_manager_get_players (applet->priv->mpris_manager);
577 for (l = players; l != NULL; l = l->next) {
578 GvcMprisPlayer *player = GVC_MPRIS_PLAYER (l->data);
579 const gchar *label;
580
581 if (!gvc_mpris_player_is_ready (player))
582 continue;
583
584 label = gvc_mpris_player_get_identity (player);
585 item = gtk_radio_menu_item_new_with_label (player_group,
586 label != NULL && label[0] != '\0' ?
587 label :
588 gvc_mpris_player_get_bus_name (player));
589 player_group = gtk_radio_menu_item_get_group (GTK_RADIO_MENU_ITEM (item));
590 gtk_check_menu_item_set_active (GTK_CHECK_MENU_ITEM (item), player == applet->priv->active_player);
591 g_object_set_data_full (G_OBJECT (item), "gvc-player", g_object_ref (player), g_object_unref);
592 g_signal_connect_object (item, "toggled", G_CALLBACK (on_player_menu_item_toggled), applet, 0);
593 gtk_menu_shell_append (GTK_MENU_SHELL (submenu), item);
594 }
595 g_list_free (players);
596 gtk_widget_show_all (submenu);
597 }
598
599 submenu = get_applet_submenu (applet, "LaunchPlayer");
600 if (submenu != NULL && applet->priv->applet_settings != NULL) {
601 gchar **known;
602 gint i;
603
604 clear_applet_submenu (submenu);
605 known = g_settings_get_strv (applet->priv->applet_settings, "known-players");
606 for (i = 0; known != NULL && known[i] != NULL; i++) {
607 item = gtk_menu_item_new_with_label (known[i]);
608 g_object_set_data_full (G_OBJECT (item), "desktop-entry", g_strdup (known[i]), g_free);
609 g_signal_connect_object (item, "activate", G_CALLBACK (on_launch_player_menu_item_activate), applet, 0);
610 gtk_menu_shell_append (GTK_MENU_SHELL (submenu), item);
611 }
612 g_strfreev (known);
613 gtk_widget_show_all (submenu);
614 }
615
616 submenu = get_applet_submenu (applet, "OutputDevice");
617 if (submenu != NULL) {
618 clear_applet_submenu (submenu);
619 append_stream_menu_items (applet, submenu, MATE_MIXER_DIRECTION_OUTPUT);
620 gtk_widget_show_all (submenu);
621 }
622
623 submenu = get_applet_submenu (applet, "InputDevice");
624 if (submenu != NULL) {
625 clear_applet_submenu (submenu);
626 append_stream_menu_items (applet, submenu, MATE_MIXER_DIRECTION_INPUT);
627 gtk_widget_show_all (submenu);
628 }
629
630 gvc_applet_update_mute_action (applet);
631}

633static void
634remember_player (GvcApplet *applet,
635 GvcMprisPlayer *player)
636{
637 const gchar *desktop_entry;
638 gchar **known;
639 gboolean found = FALSE;
640 gint i;
641
642 if (applet->priv->applet_settings == NULL)
643 return;
644
645 desktop_entry = gvc_mpris_player_get_desktop_entry (player);
646 if (desktop_entry == NULL || desktop_entry[0] == '\0')
647 return;
648
649 known = g_settings_get_strv (applet->priv->applet_settings, "known-players");
650 for (i = 0; known != NULL && known[i] != NULL; i++) {
651 if (g_strcmp0 (known[i], desktop_entry) == 0) {
652 found = TRUE;
653 break;
654 }
655 }
656
657 if (!found) {
658 GPtrArray *array;
659
660 array = g_ptr_array_new_with_free_func (g_free);
661 for (i = 0; known != NULL && known[i] != NULL; i++)
662 g_ptr_array_add (array, g_strdup (known[i]));
663 g_ptr_array_add (array, g_strdup (desktop_entry));
664 g_ptr_array_add (array, NULL);
665 g_settings_set_strv (applet->priv->applet_settings,
666 "known-players",
667 (const gchar * const *) array->pdata);
668 g_ptr_array_free (array, TRUE);
669 }
670
671 g_strfreev (known);
672}

772static void
773disconnect_input_stream_control_signals (GvcApplet *applet)
774{
775 const GList *streams;
776
777 streams = mate_mixer_context_list_streams (applet->priv->context);
778 while (streams != NULL) {
779 MateMixerStream *stream = MATE_MIXER_STREAM (streams->data);
780
781 if (mate_mixer_stream_get_direction (stream) == MATE_MIXER_DIRECTION_INPUT) {
782 g_signal_handlers_disconnect_by_func (G_OBJECT (stream),
783 G_CALLBACK (on_input_stream_control_added),
784 applet);
785 g_signal_handlers_disconnect_by_func (G_OBJECT (stream),
786 G_CALLBACK (on_input_stream_control_removed),
787 applet);
788 }
789
790 streams = streams->next;
791 }
792}

794static void
795connect_input_stream_control_signals (GvcApplet *applet)
796{
797 const GList *streams;
798
799 disconnect_input_stream_control_signals (applet);
800
801 streams = mate_mixer_context_list_streams (applet->priv->context);
802 while (streams != NULL) {
803 MateMixerStream *stream = MATE_MIXER_STREAM (streams->data);
804
805 if (mate_mixer_stream_get_direction (stream) == MATE_MIXER_DIRECTION_INPUT) {
806 g_signal_connect (G_OBJECT (stream),
807 "control-added",
808 G_CALLBACK (on_input_stream_control_added),
809 applet);
810 g_signal_connect (G_OBJECT (stream),
811 "control-removed",
812 G_CALLBACK (on_input_stream_control_removed),
813 applet);
814 }
815
816 streams = streams->next;
817 }

820static gboolean
821update_default_output_stream (GvcApplet *applet)
822{
823 MateMixerStream *stream;
824
825 stream = mate_mixer_context_get_default_output_stream (applet->priv->context);
826 if (stream == applet->priv->output)
827 return FALSE;
828
829 /* The output stream has changed */
830 if (applet->priv->output != NULL) {
831 g_signal_handlers_disconnect_by_data (G_OBJECT (applet->priv->output), applet);
832 g_object_unref (applet->priv->output);
833 }
834
835 applet->priv->output = (stream == NULL) ? NULL : g_object_ref (stream);
836 if (applet->priv->output != NULL) {
837 g_signal_connect (G_OBJECT (applet->priv->output),
838 "control-added",
839 G_CALLBACK (on_output_stream_control_added),
840 applet);
841 g_signal_connect (G_OBJECT (applet->priv->output),
842 "control-removed",
843 G_CALLBACK (on_output_stream_control_removed),
844 applet);
845 }
846
847 /* Return TRUE if the default output stream has changed */
848 return TRUE;
849}

851static gboolean
852update_default_input_stream (GvcApplet *applet)
853{
854 MateMixerStream *stream;
855
856 stream = mate_mixer_context_get_default_input_stream (applet->priv->context);
857 if (stream == applet->priv->input)
858 return FALSE;
859
860 /* The input stream has changed */
861 if (applet->priv->input != NULL) {
862 g_signal_handlers_disconnect_by_data (G_OBJECT (applet->priv->input), applet);
863 g_object_unref (applet->priv->input);
864 }
865
866 applet->priv->input = (stream == NULL) ? NULL : g_object_ref (stream);
867 /* Return TRUE if the default input stream has changed */
868 return TRUE;
869}

871static void
872on_context_state_notify (MateMixerContext *context,
873 GParamSpec *pspec,
874 GvcApplet *applet)
875{
876 MateMixerState state = mate_mixer_context_get_state (context);
877
878 switch (state) {
879 case MATE_MIXER_STATE_FAILED:
880 g_warning ("Failed to connect to a sound system");
881 break;
882
883 case MATE_MIXER_STATE_READY:
884 update_default_output_stream (applet);
885 update_default_input_stream (applet);
886 connect_input_stream_control_signals (applet);
887
888 /* Each applet change may affect the visibility of the icons */
889 update_icon_output (applet);
890 update_icon_input (applet);
891 break;
892 default:
893 break;
894 }
895}

946{
947 g_return_if_fail (GVC_IS_APPLET (applet));

963static void
964gvc_applet_dispose (GObject *object)
965{
966 GvcApplet *applet = GVC_APPLET (object);
967
968 if (applet->priv->output != NULL) {
969 g_signal_handlers_disconnect_by_data (G_OBJECT (applet->priv->output), applet);
970 g_clear_object (&applet->priv->output);
971 }
972 if (applet->priv->input != NULL) {
973 g_signal_handlers_disconnect_by_data (G_OBJECT (applet->priv->input), applet);
974 g_clear_object (&applet->priv->input);
975 }
976 if (applet->priv->context != NULL)
977 disconnect_input_stream_control_signals (applet);
978
979 if (applet->priv->mpris_manager != NULL) {
980 GList *players = gvc_mpris_manager_get_players (applet->priv->mpris_manager);
981 GList *l;
982
983 for (l = players; l != NULL; l = l->next)
984 g_signal_handlers_disconnect_by_data (l->data, applet);
985 g_list_free (players);
986 g_signal_handlers_disconnect_by_data (applet->priv->mpris_manager, applet);
987 }
988 if (applet->priv->applet_settings != NULL)
989 g_signal_handlers_disconnect_by_data (applet->priv->applet_settings, applet);
990 if (applet->priv->context != NULL)
991 g_signal_handlers_disconnect_by_data (applet->priv->context, applet);
992
993 g_clear_object (&applet->priv->context);
994 g_clear_object (&applet->priv->active_player);
995 g_clear_object (&applet->priv->selected_player);
996 g_clear_object (&applet->priv->mpris_manager);
997 g_clear_object (&applet->priv->applet_settings);
998 if (applet->priv->applet != NULL)
999 g_signal_handlers_disconnect_by_data (applet->priv->applet, applet);
1000 applet->priv->applet = NULL;
1001 applet->priv->box = NULL;
1002 applet->priv->player_widget = NULL;
1003 g_clear_object (&applet->priv->action_group);
1004 if (applet->priv->track_label != NULL)
1005 gtk_widget_destroy (GTK_WIDGET (applet->priv->track_label));
1006 g_clear_object (&applet->priv->track_label);
1007 if (applet->priv->icon_input != NULL)
1008 gtk_widget_destroy (GTK_WIDGET (applet->priv->icon_input));
1009 if (applet->priv->icon_output != NULL)
1010 gtk_widget_destroy (GTK_WIDGET (applet->priv->icon_output));
1011 g_clear_object (&applet->priv->icon_input);
1012 g_clear_object (&applet->priv->icon_output);
1013
1014 G_OBJECT_CLASS (gvc_applet_parent_class)->dispose (object);
1015}
