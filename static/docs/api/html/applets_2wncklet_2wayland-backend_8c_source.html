94static void
95update_task_state (ToplevelTask *task)
96{
97 XfwWindowState state;
98
99 if (!task || !task->window)
100 return;
101
102 state = xfw_window_get_state (task->window);
103 task->active = (state & XFW_WINDOW_STATE_ACTIVE) != 0;
104 task->maximized = (state & XFW_WINDOW_STATE_MAXIMIZED) != 0;
105 task->minimized = (state & XFW_WINDOW_STATE_MINIMIZED) != 0;
106 task->fullscreen = (state & XFW_WINDOW_STATE_FULLSCREEN) != 0;
107 task->urgent = (state & XFW_WINDOW_STATE_URGENT) != 0;
108
109 if (task->button)
110 {
111 if (task->urgent)
112 gtk_style_context_add_class (gtk_widget_get_style_context (task->button), "urgent");
113 else
114 gtk_style_context_remove_class (gtk_widget_get_style_context (task->button), "urgent");
115
116 gtk_button_set_relief (GTK_BUTTON (task->button),
117 task->active ? GTK_RELIEF_NORMAL : GTK_RELIEF_NONE);
118 }
119}

121static void
122adjust_buttons (GtkContainer *tasklist_list, int button_space, int buttons, ToplevelTask *task)
123{
124 GtkWidget *widget, *button, *box;
125
126 /*catch the case of an added button that can be missed
127 *Note that button space can come up zero on a first button
128 */
129 if (buttons < 2)
130 {
131 if(task)
132 {
133 gtk_widget_set_size_request (task->button, full_button_width, -1);
134 }
135 }
136
137 if ((task) && (button_space > 0) && (button_space < icon_size * 3))
138 {
139 gtk_widget_hide (task->icon);
140 }
141 else if (task)
142 {
143 gtk_widget_show (task->icon);
144 }
145
146 if ((task) && (button_space > 0) && (button_space < icon_size))
147 {
148 gtk_widget_hide (task->label);
149 }
150 else if (task)
151 {
152 gtk_widget_show (task->label);
153 }
154
155 GList* children = gtk_container_get_children (GTK_CONTAINER (tasklist_list));
156
157 while (children != NULL)
158 {
159 button = GTK_WIDGET (children->data);
160 box = gtk_bin_get_child (GTK_BIN (button));
161
162 if ((buttons < 2) || (buttons * full_button_width < tasklist_width * 0.75))
163 {
164 gtk_widget_set_size_request (button, full_button_width, -1);
165 gtk_widget_show_all (button);
166 return;
167 }
168 else
169 {
170 gtk_widget_set_size_request (button, MIN(button_space, full_button_width), -1);
171 }
172
173 /* if the number of buttons forces width to less than 3x the icon size, hide the icons
174 * if the number of buttons forces width to less than the icon size, hide the labels too.
175 * This is roughy the same behavior as on x11
176 * To find the icon and label we must iterate through the children of the box we packed
177 * into the button, there are two of them
178 */
179
180 GList* contents = gtk_container_get_children (GTK_CONTAINER (box));
181 while (contents != NULL)
182 {
183 widget = GTK_WIDGET (contents->data);
184 /*Show or hide the icon*/
185 if (GTK_IS_IMAGE (widget))
186 {
187 if ((button_space < icon_size * 3) && (button_space > 1))
188 gtk_widget_hide (widget);
189
190 else
191 gtk_widget_show (widget);
192
193 }
194
195 /*Show or hide the label*/
196 if (GTK_IS_LABEL (widget))
197 {
198 if ((button_space < icon_size) && (button_space > 1))
199 {
200 gtk_widget_hide (widget);
201 /*We can go a little wider for empty buttons*/
202 gtk_widget_set_size_request (button, tasklist_width / buttons * 0.9, -1);
203 if (task)
204 gtk_widget_hide (task->label);
205
206 }
207 else
208 {
209 gtk_widget_show (widget);
210 }
211 }
212 contents = contents->next;
213 }
214 children = children->next;
215 }
216 return;
217}

284static ContextMenu *
285context_menu_new ()
286{
287 ContextMenu *menu = g_new0 (ContextMenu, 1);
288 menu->menu = gtk_menu_new ();
289 menu->maximize = gtk_menu_item_new ();
290 menu->minimize = gtk_menu_item_new ();
291 menu->on_top = gtk_check_menu_item_new_with_label ("Always On Top");
292 menu->close = gtk_menu_item_new_with_label ("Close");
293
294 gtk_menu_shell_append (GTK_MENU_SHELL (menu->menu), menu->maximize);
295 gtk_menu_shell_append (GTK_MENU_SHELL (menu->menu), menu->minimize);
296 gtk_menu_shell_append (GTK_MENU_SHELL (menu->menu), gtk_separator_menu_item_new ());
297 gtk_menu_shell_append (GTK_MENU_SHELL (menu->menu), menu->on_top);
298 gtk_menu_shell_append (GTK_MENU_SHELL (menu->menu), gtk_separator_menu_item_new ());
299 gtk_menu_shell_append (GTK_MENU_SHELL (menu->menu), menu->close);
300
301 gtk_widget_show_all (menu->menu);
302
303 g_signal_connect (menu->maximize, "activate", G_CALLBACK (menu_on_maximize), NULL);
304 g_signal_connect (menu->minimize, "activate", G_CALLBACK (menu_on_minimize), NULL);
305 g_signal_connect (menu->close, "activate", G_CALLBACK (menu_on_close), NULL);
306 gtk_widget_set_sensitive (menu->on_top, FALSE);
307 return menu;
308}

336static void
337screen_window_closed (XfwScreen *screen, XfwWindow *window, TasklistManager *tasklist)
338{
339 GList *l;
340
341 for (l = tasklist->tasks; l != NULL; l = l->next)
342 {
343 ToplevelTask *task = l->data;
344 if (task->window == window)
345 {
346 GtkOrientation orient;
347 GtkWidget *outer_box, *parent_box;
348 int button_space;
349
350 buttons = buttons -1;
351
352 outer_box = tasklist->outer_box;
353 /* removes the task from tasklist->tasks and destroys the button */
354 tasklist->tasks = g_list_remove (tasklist->tasks, task);
355 gtk_widget_destroy (task->button);
356
357 if (buttons == 0)
358 return;
359
360 /* We don't need to modify button size on a vertical panel*/
361 orient = gtk_orientable_get_orientation (GTK_ORIENTABLE (outer_box));
362 if (orient == GTK_ORIENTATION_VERTICAL)
363 return;
364
365 /*Get the box the tasklist outer box sits in
366 *and leave a little space so the buttons don't push other applets off the panel
367 */
368 parent_box = gtk_widget_get_ancestor ((outer_box), GTK_TYPE_BOX);
369 tasklist_width = MAX(gtk_widget_get_allocated_width (parent_box), tasklist_width) ;
370 button_space = (tasklist_width / buttons) * 0.75;
371 button_space = MIN(button_space, full_button_width);
372 adjust_buttons (GTK_CONTAINER (tasklist->list), button_space, buttons, NULL);
373
374 return;
375 }
376 }

389static void
390tasklist_manager_disconnected_from_widget (TasklistManager *tasklist)
391{
392 if (tasklist->list)
393 {
394 GList *children = gtk_container_get_children (GTK_CONTAINER (tasklist->list));
395 for (GList *iter = children; iter != NULL; iter = g_list_next (iter))
396 gtk_widget_destroy (GTK_WIDGET (iter->data));
397 g_list_free (children);
398 tasklist->list = NULL;
399 }
400
401 /* free any remaining tasks not removed by window-closed */
402 while (tasklist->tasks != NULL)
403 {
404 ToplevelTask *task = tasklist->tasks->data;
405 tasklist->tasks = g_list_delete_link (tasklist->tasks, tasklist->tasks);
406 if (task->window)
407 {
408 g_signal_handlers_disconnect_by_data (task->window, task);
409 g_object_unref (task->window);
410 }
411 g_free (task);
412 }
413
414 if (tasklist->outer_box)
415 tasklist->outer_box = NULL;
416
417 if (tasklist->screen)
418 {
419 g_signal_handlers_disconnect_by_data (tasklist->screen, tasklist);
420 g_object_unref (tasklist->screen);
421 tasklist->screen = NULL;
422 }
423
424 if (tasklist->context_menu)
425 {
426 gtk_widget_destroy (tasklist->context_menu->menu);
427 g_free (tasklist->context_menu);
428 tasklist->context_menu = NULL;
429 }
430
431 g_free (tasklist);
432}

434static TasklistManager *
435tasklist_manager_new (void)
436{
437 TasklistManager *tasklist;
438 GList *windows, *l;
439 XfwScreen *screen = xfw_screen_get_default ();
440
441 if (screen == NULL)
442 return NULL;
443
444 tasklist = g_new0 (TasklistManager, 1);
445 tasklist->screen = screen;
446 tasklist->list = gtk_box_new (GTK_ORIENTATION_HORIZONTAL, 0);
447 gtk_box_set_homogeneous (GTK_BOX (tasklist->list), TRUE);
448 tasklist->outer_box = gtk_box_new (GTK_ORIENTATION_HORIZONTAL, 0);
449 gtk_box_pack_start (GTK_BOX (tasklist->outer_box), tasklist->list, FALSE, FALSE, 0);
450 gtk_widget_show (tasklist->list);
451 g_object_set_data_full (G_OBJECT (tasklist->outer_box),
452 tasklist_manager_key,
453 tasklist,
454 (GDestroyNotify)tasklist_manager_disconnected_from_widget);
455 tasklist->context_menu = context_menu_new ();
456
457 /* mouse scroll to switch windows */
458 gtk_widget_add_events (tasklist->list, GDK_SCROLL_MASK | GDK_SMOOTH_SCROLL_MASK);
459 g_signal_connect (tasklist->list, "scroll-event",
460 G_CALLBACK (tasklist_scroll_event), tasklist);
461
462 /* add all existing windows on this screen */
463 windows = xfw_screen_get_windows (screen);
464 for (l = windows; l != NULL; l = l->next)
465 {
466 XfwWindow *window = l->data;
467 ToplevelTask *task = toplevel_task_new (tasklist, window);
468 if (task != NULL)
469 gtk_box_pack_start (GTK_BOX (tasklist->list), task->button, TRUE, TRUE, 0);
470 }
471
472 /* monitor window changes */
473 g_signal_connect (screen, "window-opened",
474 G_CALLBACK (screen_window_opened), tasklist);
475 g_signal_connect (screen, "window-closed",
476 G_CALLBACK (screen_window_closed), tasklist);
477 g_signal_connect (screen, "active-window-changed",
478 G_CALLBACK (screen_active_window_changed), tasklist);
479
480 return tasklist;
481}

541static gboolean
542tasklist_button_drag_motion (GtkWidget *widget,
543 GdkDragContext *context,
544 gint x, gint y, guint time,
545 TasklistManager *tasklist)
546{
547 GtkWidget *source_widget;
548 ToplevelTask *source, *target;
549 GList *source_node, *target_node;
550 GtkAllocation allocation;
551 gboolean second_half;
552
553 source_widget = gtk_drag_get_source_widget (context);
554 if (source_widget == NULL)
555 return FALSE;
556
557 /* only reorder among our own tasklist buttons */
558 if (gtk_widget_get_parent (source_widget) != GTK_WIDGET (tasklist->list))
559 return FALSE;
560
561 source = g_object_get_data (G_OBJECT (source_widget), toplevel_task_key);
562 target = g_object_get_data (G_OBJECT (widget), toplevel_task_key);
563 if (source == NULL || target == NULL || source == target)
564 return FALSE;
565
566 /* drop on the right/bottom half -> insert after the target button */
567 gtk_widget_get_allocation (widget, &allocation);
568 second_half = tasklist_horizontal (tasklist)
569 ? x >= allocation.width / 2
570 : y >= allocation.height / 2;
571
572 source_node = g_list_find (tasklist->tasks, source);
573 target_node = g_list_find (tasklist->tasks, target);
574 if (source_node == NULL || target_node == NULL)
575 return FALSE;
576 if (second_half)
577 target_node = g_list_next (target_node);
578
579 /* no change needed if already at the insertion point */
580 if (source_node == target_node || g_list_next (source_node) == target_node)
581 {
582 gdk_drag_status (context, GDK_ACTION_MOVE, time);
583 return TRUE;
584 }
585
586 /* live-reorder the list and the widgets */
587 tasklist->tasks = g_list_remove (tasklist->tasks, source);
588 tasklist->tasks = g_list_insert_before (tasklist->tasks, target_node, source);
589 gtk_box_reorder_child (GTK_BOX (tasklist->list), source->button,
590 g_list_index (tasklist->tasks, source));
591 gtk_widget_queue_resize (tasklist->list);
592
593 gdk_drag_status (context, GDK_ACTION_MOVE, time);
594 return TRUE;
595}

626static gboolean on_toplevel_button_press (GtkWidget *button, GdkEvent *event, TasklistManager *tasklist)
627{
628 /* Assume event is a button press */
629
630 if (((GdkEventButton*)event)->button == GDK_BUTTON_MIDDLE &&
631 tasklist->middle_click_close)
632 {
633 ToplevelTask *task = g_object_get_data (G_OBJECT (button), toplevel_task_key);
634 if (task && task->window)
635 xfw_window_close (task->window, ((GdkEventButton*)event)->time, NULL);
636 return TRUE;
637 }
638
639 if (((GdkEventButton*)event)->button == GDK_BUTTON_SECONDARY)
640 {
641 ContextMenu *menu = tasklist->context_menu;
642 ToplevelTask *task = g_object_get_data (G_OBJECT (button), toplevel_task_key);
643
644 g_object_set_data (G_OBJECT (menu->maximize), toplevel_task_key, task);
645 g_object_set_data (G_OBJECT (menu->minimize), toplevel_task_key, task);
646 g_object_set_data (G_OBJECT (menu->close), toplevel_task_key, task);
647
648 gtk_menu_item_set_label (GTK_MENU_ITEM (menu->minimize),
649 task->minimized ? "Unminimize" : "Minimize");
650 gtk_menu_item_set_label (GTK_MENU_ITEM (menu->maximize),
651 task->maximized ? "Unmaximize" : "Maximize");
652
653 gtk_menu_popup_at_widget (GTK_MENU (menu->menu), button,
654 GDK_GRAVITY_NORTH_WEST, GDK_GRAVITY_SOUTH_WEST, event);
655 return TRUE;
656 }
657 else
658 {
659 return FALSE;
660 }
661}

663static gboolean
664tasklist_scroll_event (GtkWidget *widget, GdkEventScroll *event, TasklistManager *tasklist)
665{
666 ToplevelTask *child;
667 GList *li, *lnew = NULL;
668 GdkScrollDirection direction;
669 gboolean wrap_windows = TRUE;
670
671 if (!tasklist->scroll_enabled)
672 return TRUE;
673
674 /* get the current active window button */
675 for (li = tasklist->tasks; li != NULL; li = li->next)
676 {
677 child = li->data;
678 if (child->window && xfw_window_is_active (child->window))
679 break;
680 }
681
682 if (li == NULL)
683 return TRUE;
684
685 if (event->direction != GDK_SCROLL_SMOOTH)
686 direction = event->direction;
687 else if (event->delta_y < 0)
688 direction = GDK_SCROLL_UP;
689 else if (event->delta_y > 0)
690 direction = GDK_SCROLL_DOWN;
691 else if (event->delta_x < 0)
692 direction = GDK_SCROLL_LEFT;
693 else if (event->delta_x > 0)
694 direction = GDK_SCROLL_RIGHT;
695 else
696 return TRUE;
697
698 switch (direction)
699 {
700 case GDK_SCROLL_UP:
701 /* find the previous button on the tasklist */
702 for (lnew = g_list_previous (li);; lnew = lnew->prev)
703 {
704 if (lnew == NULL)
705 {
706 if (wrap_windows)
707 {
708 lnew = g_list_last (li);
709 wrap_windows = FALSE;
710 if (lnew == NULL)
711 break;
712 }
713 else
714 break;
715 }
716
717 child = lnew->data;
718 if (child->window != NULL)
719 break;
720 }
721 break;
722
723 case GDK_SCROLL_DOWN:
724 /* find the next button on the tasklist */
725 for (lnew = g_list_next (li);; lnew = lnew->next)
726 {
727 if (lnew == NULL)
728 {
729 if (wrap_windows)
730 {
731 lnew = g_list_first (li);
732 wrap_windows = FALSE;
733 if (lnew == NULL)
734 break;
735 }
736 else
737 break;
738 }
739
740 child = lnew->data;
741 if (child->window != NULL)
742 break;
743 }
744 break;
745
746 case GDK_SCROLL_LEFT:
747 case GDK_SCROLL_RIGHT:
748 default:
749 return TRUE;
750 }
751
752 if (lnew != NULL)
753 {
754 child = lnew->data;
755 xfw_window_activate (child->window, NULL, event->time, NULL);
756 }
757
758 return TRUE;
759}

841 /* Buttons on a vertical panel are not affected by how many are needed
842 * GTK handles compressing contents as needed as the window width tells
843 * GTK how much space to allocate the label and icon. Buttons will use
844 * the full width of a vertical panel without any special attention
845 * so break out here instead of breaking the vertical panel case
846 */
847
848 if (orient == GTK_ORIENTATION_VERTICAL)
849 return task;
850
851 /* On horizontal panels, GTK does not by default limit the width of the tasklist
852 * as it does not run out of space in the window until the entire panel is used,
853 * leaving buttons at full width until then and overflowing all other applets
854 *
855 * Thus we must get the tasklist's allocated width when extra space remains,
856 * which will be most of the distance between the handle and the next applet
857 * From there, we can expand buttons and/or hide elements as needed
858 * For some reason this function always gets called twice, so use half the value of buttons
859 * but do not attempt to adjust the global value as it would get adjusted twice
860 * Since we are adding a button here the true value cannot be zero
861 */
862 whole_panel_box = gtk_widget_get_toplevel(GTK_WIDGET (tasklist->outer_box));
863 parent_box = gtk_widget_get_ancestor(GTK_WIDGET (tasklist->outer_box), GTK_TYPE_BOX);
864 if (gtk_widget_get_allocated_width (parent_box) > 1)
865 {
866 tasklist_width = gtk_widget_get_allocated_width (parent_box);
867 }
868 else
869 {
870 tasklist_width = MAX(gtk_widget_get_allocated_width (parent_box), tasklist_width);
871 }
872
873 panel_width = gtk_widget_get_allocated_width (whole_panel_box);
874
875 /*on startup we get an allocated with of zero, so start with 1/3 the panel width
876 *as a sane default
877 *This may overflow on very crowded panels where the tasklist is less than 1/3ed the
878 *panel witth but will self-correct on opening or closing a few windows
879 */
880
881 if (tasklist_width <= 2)
882 tasklist_width = panel_width / 3;
883
884 /*Do not allow buttons to equal zero or the division below is a crasher*/
885 buttons = MAX ((buttons), 1);
886
887 /*always allow at least three buttons to fit without adjustment
888 *so short window lists don't overflow
889 */
890 if (tasklist_width > 0)
891 {
892 full_button_width = MIN(max_button_width, tasklist_width / 3);
893 }
894
895 /*Leave a little space so the buttons don't push other applets off the panel*/
896 button_space = (tasklist_width / buttons) * 0.75;
897 button_space = MIN(button_space, full_button_width);
898
899 /* iterate over all the buttons*/
900 adjust_buttons (GTK_CONTAINER (tasklist->list), button_space, buttons, task);
901
902 /*Reset the tasklist width after button adjustments*/
903 if (gtk_widget_get_allocated_width (parent_box) > 1)
904 {
905 tasklist_width = gtk_widget_get_allocated_width (parent_box);
906 }
907 else
908 {
909 tasklist_width = MAX(gtk_widget_get_allocated_width (parent_box), tasklist_width);
910 }
911 return task;
912}
