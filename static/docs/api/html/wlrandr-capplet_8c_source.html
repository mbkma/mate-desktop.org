274static void
275add_key (GtkWidget *widget,
276 const char *text,
277 guint width,
278 guint height,
279 int rate,
280 int rotation)
281{
282 ForeachInfo info;
283 GtkComboBox *box = GTK_COMBO_BOX (widget);
284 GtkTreeModel *model = gtk_combo_box_get_model (box);
285 GtkListStore *store = GTK_LIST_STORE (model);
286
287 info.text = text;
288 info.found = FALSE;
289
290 gtk_tree_model_foreach (model, foreach, &info);
291
292 if (!info.found)
293 {
294 GtkTreeIter iter;
295 gtk_list_store_insert_with_values (store, &iter, -1,
296 0, text,
297 1, width,
298 2, height,
299 3, rate,
300 4, width * height,
301 5, rotation,
302 -1);
303 }
304}

397static gboolean
398reported_mirror (App *app)
399{
400 const char *source = NULL;
401 guint i, enabled = 0;
402
403 for (i = 0; i < app->heads->len; i++)
404 {
405 OutputHead *head = get_nth_head (app, i);
406
407 if (!head->enabled)
408 continue;
409
410 if (head->mirror_of == NULL || head->mirror_of[0] == '\0' ||
411 (source != NULL && g_strcmp0 (source, head->mirror_of) != 0))
412 return FALSE;
413
414 source = head->mirror_of;
415 enabled++;
416 }
417
418 return enabled >= 2;
419}

421static void
422on_heads_changed (GArray *heads, const char *error, gpointer data)
423{
424 App *app = data;
425 char *selected = NULL;
426 guint i;
427
428 if (heads == NULL)
429 {
430 error_message (app, _("Could not get display configuration"), error);
431 return;
432 }
433
434 if (app->current_head)
435 selected = g_strdup (app->current_head->name);
436
437 free_heads (app);
438 app->current_head = NULL;
439 app->heads = heads;
440
441 if (app->pending == NULL)
442 {
443 display_heads_free (app->reported);
444 app->reported = display_heads_copy (heads);
445 }
446
447 for (i = 0; selected && i < heads->len; i++)
448 {
449 if (g_strcmp0 (get_nth_head (app, i)->name, selected) == 0)
450 app->current_head = get_nth_head (app, i);
451 }
452 g_free (selected);
453
454 app->mirror = has_mirror (app) && reported_mirror (app);
455 if (app->mirror)
456 app->current_head = first_enabled_head (app);
457 if (app->labeler)
458 display_labeler_show (app->labeler, heads, app->mirror);
459 select_default_head (app);
460
461 rebuild_gui (app);
462 foo_scroll_area_invalidate (FOO_SCROLL_AREA (app->area));
463}

506static gboolean
507ask_to_keep (App *app)
508{
509 Countdown countdown;
510 int response;
511
512 countdown.dialog = gtk_message_dialog_new (GTK_WINDOW (app->dialog),
513 GTK_DIALOG_MODAL | GTK_DIALOG_DESTROY_WITH_PARENT,
514 GTK_MESSAGE_QUESTION, GTK_BUTTONS_NONE,
515 "%s", g_dgettext (MSD_DOMAIN, "Does the display look OK?"));
516 gtk_dialog_add_buttons (GTK_DIALOG (countdown.dialog),
517 g_dgettext (MSD_DOMAIN, "_Restore Previous Configuration"), GTK_RESPONSE_CANCEL,
518 g_dgettext (MSD_DOMAIN, "_Keep This Configuration"), GTK_RESPONSE_ACCEPT,
519 NULL);
520 gtk_dialog_set_default_response (GTK_DIALOG (countdown.dialog), GTK_RESPONSE_ACCEPT);
521
522 countdown.left = CONFIRM_SECONDS;
523 countdown_show (&countdown);
524 countdown.source = g_timeout_add_seconds (1, countdown_tick, &countdown);
525
526 response = gtk_dialog_run (GTK_DIALOG (countdown.dialog));
527
528 if (countdown.source)
529 g_source_remove (countdown.source);
530 gtk_widget_destroy (countdown.dialog);
531
532 return response == GTK_RESPONSE_ACCEPT;
533}

537static gboolean
538confirm_configuration (gpointer data)
539{
540 App *app = data;
541 GArray *pending = app->pending;
542 GArray *previous = app->previous;
543 gboolean keep;
544
545 keep = ask_to_keep (app) || previous == NULL;
546
547 app->pending = NULL;
548 app->previous = NULL;
549
550 app->applying = TRUE;
551 gtk_widget_set_sensitive (app->dialog, FALSE);
552 display_backend_apply (app->backend, keep ? pending : previous, FALSE, on_configuration_applied, app);
553
554 display_heads_free (pending);
555 display_heads_free (previous);
556
557 return G_SOURCE_REMOVE;
558}

560static void
561on_configuration_applied (const char *error, gpointer data)
562{
563 App *app = data;
564
565 app->applying = FALSE;
566
567 if (error)
568 {
569 display_heads_free (app->pending);
570 display_heads_free (app->previous);
571 app->pending = NULL;
572 app->previous = NULL;
573 error_message (app, _("Could not apply the display configuration"), error);
574 }
575 else if (app->pending)
576 {
577 g_idle_add (confirm_configuration, app);
578 return;
579 }
580
581 gtk_widget_set_sensitive (app->dialog, TRUE);
582}

584static void
585apply_configuration (App *app)
586{
587 gboolean temporary;
588
589 g_assert (app->backend != NULL);
590 g_assert (app->applying == FALSE);
591
592 temporary = (display_backend_get_features (app->backend) & DISPLAY_BACKEND_TEMPORARY) != 0;
593
594 app->applying = TRUE;
595 display_layout_normalize (app->heads);
596 gtk_widget_set_sensitive (app->dialog, FALSE);
597
598 if (temporary)
599 {
600 app->pending = display_heads_copy (app->heads);
601 app->previous = app->reported ? display_heads_copy (app->reported) : NULL;
602 }
603
604 display_backend_apply (app->backend, app->heads, temporary, on_configuration_applied, app);
605}

728static void
729rebuild_resolution_combo (App *app)
730{
731 guint i;
732 int width, height;
733
734 clear_combo (app->resolution_combo);
735
736 if (!app->current_head || !app->current_head->enabled)
737 {
738 gtk_widget_set_sensitive (app->resolution_combo, FALSE);
739 return;
740 }
741
742 gtk_widget_set_sensitive (app->resolution_combo, TRUE);
743
744 for (i = 0; i < app->current_head->modes->len; i++)
745 {
746 Mode *mode = &g_array_index (app->current_head->modes, Mode, i);
747
748 if (app->mirror && !display_heads_have_size (app->heads, mode->width, mode->height))
749 continue;
750
751 add_key (app->resolution_combo,
752 make_resolution_string (mode->width, mode->height),
753 mode->width, mode->height, 0, -1);
754 }
755
756 display_head_get_size (app->current_head, &width, &height);
757
758 if (!combo_select (app->resolution_combo, make_resolution_string (width, height)))
759 {
760 int best_w = 0, best_h = 0;
761
762 for (i = 0; i < app->current_head->modes->len; i++)
763 {
764 Mode *mode = &g_array_index (app->current_head->modes, Mode, i);
765
766 if (mode->width * mode->height > best_w * best_h)
767 {
768 best_w = mode->width;
769 best_h = mode->height;
770 }
771 }
772
773 combo_select (app->resolution_combo, make_resolution_string (best_w, best_h));
774 }
