71static gboolean
72gpm_button_emit_type (GpmButton *button, const gchar *type)
73{
74 g_return_val_if_fail (GPM_IS_BUTTON (button), FALSE);
75
76 /* did we just have this button before the timeout? */
77 if (g_strcmp0 (type, button->priv->last_button) == 0 &&
78 g_timer_elapsed (button->priv->timer, NULL) < GPM_BUTTON_DUPLICATE_TIMEOUT) {
79 g_debug ("ignoring duplicate button %s", type);
80 return FALSE;
81 }
82
83 g_debug ("emitting button-pressed : %s", type);
84 g_signal_emit (button, signals [BUTTON_PRESSED], 0, type);
85
86 /* save type and last size */
87 g_free (button->priv->last_button);
88 button->priv->last_button = g_strdup (type);
89 g_timer_reset (button->priv->timer);
90
91 return TRUE;
92}

99static GdkFilterReturn
100gpm_button_filter_x_events (GdkXEvent *xevent, GdkEvent *event, gpointer data)
101{
102 GpmButton *button = (GpmButton *) data;
103 XEvent *xev = (XEvent *) xevent;
104 guint keycode;
105 const gchar *key;
106 gchar *keycode_str;
107
108 if (xev->type != KeyPress)
109 return GDK_FILTER_CONTINUE;
110
111 keycode = xev->xkey.keycode;
112
113 /* is the key string already in our DB? */
114 keycode_str = g_strdup_printf ("0x%x", keycode);
115 key = g_hash_table_lookup (button->priv->keysym_to_name_hash, (gpointer) keycode_str);
116 g_free (keycode_str);
117
118 /* found anything? */
119 if (key == NULL) {
120 g_debug ("Key %u not found in hash", keycode);
121 /* pass normal keypresses on, which might help with accessibility access */
122 return GDK_FILTER_CONTINUE;
123 }
124
125 g_debug ("Key %u mapped to key %s", keycode, key);
126 gpm_button_emit_type (button, key);
127
128 return GDK_FILTER_REMOVE;
129}

143static gboolean
144gpm_button_grab_keystring (GpmButton *button, guint64 keycode)
145{
146 guint modmask = AnyModifier;
147 Display *display;
148 GdkDisplay *gdkdisplay;
149 gint ret;
150
151 /* get the current X Display */
152 display = GDK_DISPLAY_XDISPLAY (gdk_display_get_default());
153
154 /* don't abort on error */
155 gdkdisplay = gdk_display_get_default ();
156 gdk_x11_display_error_trap_push (gdkdisplay);
157
158 /* grab the key if possible */
159 ret = XGrabKey (display, keycode, modmask,
160 GDK_WINDOW_XID (button->priv->window), True,
161 GrabModeAsync, GrabModeAsync);
162 if (ret == BadAccess) {
163 g_warning ("Failed to grab modmask=%u, keycode=%li",
164 modmask, (long int) keycode);
165 return FALSE;
166 }
167
168 /* grab the lock key if possible */
169 ret = XGrabKey (display, keycode, LockMask | modmask,
170 GDK_WINDOW_XID (button->priv->window), True,
171 GrabModeAsync, GrabModeAsync);
172 if (ret == BadAccess) {
173 g_warning ("Failed to grab modmask=%u, keycode=%li",
174 LockMask | modmask, (long int) keycode);
175 return FALSE;
176 }
177
178 /* we are not processing the error */
179 gdk_display_flush (gdkdisplay);
180 gdk_x11_display_error_trap_pop_ignored (gdkdisplay);
181
182 g_debug ("Grabbed modmask=%x, keycode=%li", modmask, (long int) keycode);
183 return TRUE;
184}

198static gboolean
199gpm_button_xevent_key (GpmButton *button, guint keysym, const gchar *key_name)
200{
201 gchar *key = NULL;
202 gboolean ret;
203 gchar *keycode_str;
204 guint keycode;
205
206 /* convert from keysym to keycode */
207 keycode = XKeysymToKeycode (GDK_DISPLAY_XDISPLAY (gdk_display_get_default()), keysym);
208 if (keycode == 0) {
209 g_warning ("could not map keysym %x to keycode", keysym);
210 return FALSE;
211 }
212
213 /* is the key string already in our DB? */
214 keycode_str = g_strdup_printf ("0x%x", keycode);
215 key = g_hash_table_lookup (button->priv->keysym_to_name_hash, (gpointer) keycode_str);
216 if (key != NULL) {
217 g_warning ("found in hash %u", keycode);
218 g_free (keycode_str);
219 return FALSE;
220 }
221
222 /* try to register X event */
223 ret = gpm_button_grab_keystring (button, keycode);
224 if (!ret) {
225 g_warning ("Failed to grab %u", keycode);
226 g_free (keycode_str);
227 return FALSE;
228 }
229
230 /* add to hash table */
231 g_hash_table_insert (button->priv->keysym_to_name_hash, (gpointer) keycode_str, (gpointer) g_strdup (key_name));
232
233 /* the key is freed in the hash function unref */
234 return TRUE;
235}

262gboolean
263gpm_button_is_lid_closed (GpmButton *button)
264{
265 GDBusProxy *proxy;
266 GVariant *res, *inner;
267 gboolean lid;
268 GError *error = NULL;
269
270 g_return_val_if_fail (GPM_IS_BUTTON (button), FALSE);
271
272 if (LOGIND_RUNNING()) {
273 proxy = g_dbus_proxy_new_for_bus_sync (G_BUS_TYPE_SYSTEM,
274 G_DBUS_PROXY_FLAGS_DO_NOT_LOAD_PROPERTIES,
275 NULL,
276 "org.freedesktop.UPower",
277 "/org/freedesktop/UPower",
278 "org.freedesktop.DBus.Properties",
279 NULL,
280 &error );
281 if (proxy == NULL) {
282 g_error("Error connecting to dbus - %s", error->message);
283 g_error_free (error);
284 return -1;
285 }
286
287 res = g_dbus_proxy_call_sync (proxy, "Get",
288 g_variant_new( "(ss)",
289 "org.freedesktop.UPower",
290 "LidIsClosed"),
291 G_DBUS_CALL_FLAGS_NONE,
292 -1,
293 NULL,
294 &error
295 );
296 if (error == NULL && res != NULL) {
297 g_variant_get(res, "(v)", &inner );
298 lid = g_variant_get_boolean(inner);
299 g_variant_unref (inner);
300 g_variant_unref (res);
301 return lid;
302 } else if (error != NULL ) {
303 g_error ("Error in dbus - %s", error->message);
304 g_error_free (error);
305 }
306 g_object_unref(proxy);
307
308 return FALSE;
309 }
310 else {
311 return up_client_get_lid_is_closed (button->priv->client);
312 }
313}

332static void
333gpm_button_client_changed_cb (UpClient *client, GParamSpec *pspec, GpmButton *button)
334{
335 gboolean lid_is_closed;
336
337 /* get new state */
338 lid_is_closed = gpm_button_is_lid_closed(button);
339
340 /* same state */
341 if (button->priv->lid_is_closed == lid_is_closed)
342 return;
343
344 /* save state */
345 button->priv->lid_is_closed = lid_is_closed;
346
347 /* sent correct event */
348 if (lid_is_closed)
349 gpm_button_emit_type (button, GPM_BUTTON_LID_CLOSED);
350 else
351 gpm_button_emit_type (button, GPM_BUTTON_LID_OPEN);
352}

358static void
359gpm_button_init (GpmButton *button)
360{
361 button->priv = gpm_button_get_instance_private (button);
362
363 button->priv->last_button = NULL;
364 button->priv->timer = g_timer_new ();
365
366 button->priv->client = up_client_new ();
367 button->priv->lid_is_closed = up_client_get_lid_is_closed (button->priv->client);
368 g_signal_connect (button->priv->client, "notify",
369 G_CALLBACK (gpm_button_client_changed_cb), button);
370
371#ifdef HAVE_X11
372 if (gdk_display_get_default () != NULL &&
373 GDK_IS_X11_DISPLAY (gdk_display_get_default ())) {
374 button->priv->screen = gdk_screen_get_default ();
375 button->priv->window = gdk_screen_get_root_window (button->priv->screen);
376 button->priv->keysym_to_name_hash = g_hash_table_new_full (g_str_hash, g_str_equal, g_free, g_free);
377
378 gpm_button_xevent_key (button, XF86XK_PowerOff, GPM_BUTTON_POWER);
379
380 gpm_button_xevent_key (button, XF86XK_Suspend, GPM_BUTTON_HIBERNATE);
381 gpm_button_xevent_key (button, XF86XK_Sleep, GPM_BUTTON_SUSPEND);
382 gpm_button_xevent_key (button, XF86XK_Hibernate, GPM_BUTTON_HIBERNATE);
383 gpm_button_xevent_key (button, XF86XK_MonBrightnessUp, GPM_BUTTON_BRIGHT_UP);
384 gpm_button_xevent_key (button, XF86XK_MonBrightnessDown, GPM_BUTTON_BRIGHT_DOWN);
385 gpm_button_xevent_key (button, XF86XK_ScreenSaver, GPM_BUTTON_LOCK);
386 gpm_button_xevent_key (button, XF86XK_Battery, GPM_BUTTON_BATTERY);
387 gpm_button_xevent_key (button, XF86XK_KbdBrightnessUp, GPM_BUTTON_KBD_BRIGHT_UP);
388 gpm_button_xevent_key (button, XF86XK_KbdBrightnessDown, GPM_BUTTON_KBD_BRIGHT_DOWN);
389 gpm_button_xevent_key (button, XF86XK_KbdLightOnOff, GPM_BUTTON_KBD_BRIGHT_TOGGLE);
390
391 gdk_window_add_filter (button->priv->window,
392 gpm_button_filter_x_events, (gpointer) button);
393 return;
394 }
395#endif
396 g_debug ("No X11 backend for button key grabbing; compositor handles keys");
397}

403static void
404gpm_button_finalize (GObject *object)
405{
406 GpmButton *button;
407 g_return_if_fail (object != NULL);
408 g_return_if_fail (GPM_IS_BUTTON (object));
409
410 button = GPM_BUTTON (object);
411 button->priv = gpm_button_get_instance_private (button);
412
413 g_object_unref (button->priv->client);
414 g_free (button->priv->last_button);
415 g_timer_destroy (button->priv->timer);
416
417#ifdef HAVE_X11
418 if (button->priv->keysym_to_name_hash)
419 g_hash_table_unref (button->priv->keysym_to_name_hash);
420#endif
421
422 G_OBJECT_CLASS (gpm_button_parent_class)->finalize (object);
423}
